    68→    68→export function ChatView({ business, userId, onNavigate }: ChatViewProps) {
    69→    69→  const vertical = getVertical(business.type)

    75→    75→  const { toast } = useToast()
    76→    76→
    77→    77→  const [turns, setTurns] = useState<ChatTurn[]>([])
    78→    78→  // the conversation ACTUALLY displayed (from GET / start event) — distinct
    79→    79→  // from the requested one so the query key only re-keys on explicit intent
    80→    80→  const [loadedConvId, setLoadedConvId] = useState<string | null>(null)
    81→    81→  const [sending, setSending] = useState(false)
    82→    82→  const [busyOpId, setBusyOpId] = useState<string | null>(null)
    83→    83→  const [revealId, setRevealId] = useState<string | null>(null)
    84→    84→  const [failedId, setFailedId] = useState<string | null>(null)
    85→    85→  const [nearBottom, setNearBottom] = useState(true)
    86→    86→  const [drawerOpen, setDrawerOpen] = useState(false)
    87→    87→  const [creating, setCreating] = useState(false)
    88→    88→
    89→    89→  const scrollRef = useRef<HTMLDivElement>(null)
    90→    90→  const nearBottomRef = useRef(true)
    91→    91→  const sendingRef = useRef(false)
    92→    92→  const retryRef = useRef<string | null>(null)
    93→    93→  const creatingRef = useRef(false)
    94→    94→
    95→    95→  // ---- load conversation history (keyed to the ACTIVE conversation) ----
    96→    96→  const { data, isLoading } = useQuery({
    97→    97→    queryKey: QUERY_KEYS.chatConversation(business.id, userId, activeConversationId),
    98→    98→    queryFn: () => api.chat(business.id, userId, activeConversationId),
    99→    99→    refetchOnWindowFocus: false,
   100→   100→  })
   101→   101→  // One effect owns the transcript: it applies the fetched history whenever
   102→   102→  // the data or the target conversation changes, and drops the old transcript
   103→   103→  // while a NEW key has no cached data yet (so stale messages never linger).
   104→   104→  // (deps include the conversation key — a cache-hit key switch can return a
   105→   105→  // deep-equal object via structural sharing, which alone wouldn't re-fire.)
   106→   106→  useEffect(() => {
   107→   107→    if (!data) {
   108→   108→      setTurns([])
   109→   109→      setRevealId(null)
   110→   110→      setFailedId(null)
   111→   111→      return
   112→   112→    }
   113→   113→    if (sendingRef.current) return
   114→   114→    setTurns(data.messages.map(messageToTurn))
   115→   115→    setLoadedConvId(data.conversation.id)
   116→   116→    setRevealId(null)
   117→   117→    setFailedId(null)
   118→   118→  }, [data, business.id, userId, activeConversationId])
   119→   119→
   120→   120→  // ---- smart scroll: follow only while the user is near the bottom ----
   121→   121→  const scrollToBottom = useCallback((smooth = true) => {
   122→   122→    const el = scrollRef.current
   123→   123→    if (!el) return
   124→   124→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   125→   125→    nearBottomRef.current = true
   126→   126→  }, [])
   127→   127→
   128→   128→  const onScroll = useCallback(() => {
   129→   129→    const el = scrollRef.current
   130→   130→    if (!el) return
   131→   131→    const near = el.scrollHeight - el.scrollTop - el.clientHeight < 140
   132→   132→    nearBottomRef.current = near
   133→   133→    setNearBottom((prev) => (prev === near ? prev : near))
   134→   134→  }, [])
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    if (nearBottomRef.current) scrollToBottom(false)
   138→   138→  }, [turns, scrollToBottom])
   139→   139→
   140→   140→  // ---- providers for the model picker ----
   141→   141→  const { data: providersData } = useQuery({
   142→   142→    queryKey: QUERY_KEYS.providers,
   143→   143→    queryFn: () => api.providers(),
   144→   144→  })
   145→   145→  const currentLabel = useMemo(() => {
   146→   146→    if (modelKey === 'builtin-deep') return 'عميق مفكّر'
   147→   147→    if (modelKey === 'builtin-air') return 'خفيف سريع'
   148→   148→    const ext = providersData?.providers.flatMap((p) => p.models).find((m) => `ext:${m.id}` === modelKey)
   149→   149→    return ext?.label ?? 'عميق مفكّر'
   150→   150→  }, [modelKey, providersData])
   151→   151→
   152→   152→  // carry the provider name so two providers exposing the same model label
   153→   153→  // (e.g. مزوّد تجريبي + مزوّد الاختبار → وكيل خفيف ×2) stay distinguishable
   154→   154→  const enabledExternal = (providersData?.providers ?? [])
   155→   155→    .flatMap((p) => p.models.map((m) => ({ ...m, providerName: p.name })))
   156→   156→    .filter((m) => m.enabled)
   157→   157→
   158→   158→  // ---- dynamic suggestions (the adaptive demos — بتتغير مع شغلك) ----
   159→   159→  const suggestionsQ = useQuery({
   160→   160→    queryKey: QUERY_KEYS.suggestions(business.id, userId),
   161→   161→    queryFn: () => api.suggestions(business.id, userId),
   162→   162→    refetchOnMount: true,
   163→   163→    retry: 1,
   164→   164→  })
   165→   165→
   166→   166→  // ---- the user's own quick tools (أدواتك السريعة — always visible) ----
   167→   167→  const commandsQ = useQuery({
   168→   168→    queryKey: QUERY_KEYS.quickCommands(business.id),
   169→   169→    queryFn: () => api.quickCommands(business.id),
   170→   170→  })
   171→   171→
   172→   172→  // ---- send a message → stream the turn (ChatGPT order guaranteed) ----
   173→   173→  const send = useCallback(
   174→   174→    async (text: string, opts?: { suggestionId?: string }) => {
   175→   175→      if (sendingRef.current || !text.trim()) return
   176→   176→      const trimmed = text.trim()
   177→   177→      sendingRef.current = true
   178→   178→      setSending(true)
   179→   179→      setFailedId(null)
   180→   180→      setRevealId(null)
   181→   181→      retryRef.current = trimmed
   182→   182→
   183→   183→      const userLocalId = `local-user-${Date.now()}`
   184→   184→      const liveId = `local-live-${Date.now()}`
   185→   185→
   186→   186→      setTurns((prev) => [
   187→   187→        ...prev,
   188→   188→        { id: userLocalId, role: 'user', text: trimmed, createdAt: new Date().toISOString() },
   189→   189→        {
   190→   190→          id: liveId,
   191→   191→          role: 'assistant',
   192→   192→          text: '',
   193→   193→          thinking: [],
   194→   194→          createdAt: new Date().toISOString(),
   195→   195→          streaming: true,
   196→   196→        },
   197→   197→      ])
   198→   198→      nearBottomRef.current = true
   199→   199→      setNearBottom(true)
   200→   200→
   201→   201→      // BUFFERS: card/cards/actions stay out of render until the stream ends
   202→   202→      const buf = {
   203→   203→        card: null as ActionCardDTO | null,
   204→   204→        cards: null as ResultCardDTO[] | null,
   205→   205→        actions: null as ChatActionDTO[] | null,
   206→   206→      }
   207→   207→      let errored = false
   208→   208→      let doneId: string | null = null
   209→   209→      // IMPORTANT: capture the target id at HANDLER-CALL time — closures that
   210→   210→      // resolve it inside the setTurns updater would read `doneId` at flush
   211→   211→      // time and silently drop the final deltas that arrive in the same batch
   212→   212→      // as the done event.
   213→   213→      const patchTurn = (patch: Partial<ChatTurn>) => {
   214→   214→        const id = doneId ?? liveId
   215→   215→        setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, ...patch } : t)))
   216→   216→      }
   217→   217→
   218→   218→      try {
   219→   219→        await streamChat(
   220→   220→          {
   221→   221→            businessId: business.id,
   222→   222→            userId,
   223→   223→            text: trimmed,
   224→   224→            modelKey,
   225→   225→            conversationId: activeConversationId ?? loadedConvId,
   226→   226→            suggestionId: opts?.suggestionId ?? null,
   227→   227→          },
   228→   228→          {
   229→   229→            onStart: (cid, userMessageId) => {
   230→   230→              // a new conversation may be created server-side when none was
   231→   231→              // passed — track it so later sends stay in the same thread
   232→   232→              setLoadedConvId(cid)
   233→   233→              // swap optimistic user id → the real persisted id so any later
   234→   234→              // refetch reconciles by key (no remount → no flicker)
   235→   235→              setTurns((prev) => prev.map((t) => (t.id === userLocalId ? { ...t, id: userMessageId } : t)))
   236→   236→            },
   237→   237→            onThinking: (steps) => patchTurn({ thinking: steps }),
   238→   238→            onStep: (label, state) => {
   239→   239→              const id = doneId ?? liveId
   240→   240→              setTurns((prev) =>
   241→   241→                prev.map((t) =>
   242→   242→                  t.id === id && t.thinking
   243→   243→                    ? { ...t, thinking: t.thinking.map((s) => (s.label === label ? { ...s, state } : s)) }
   244→   244→                    : t,
   245→   245→                ),
   246→   246→              )
   247→   247→            },
   248→   248→            onCard: (card) => {
   249→   249→              buf.card = card
   250→   250→            },
   251→   251→            onCards: (cards) => {
   252→   252→              buf.cards = cards
   253→   253→            },
   254→   254→            onActions: (actions) => {
   255→   255→              buf.actions = actions
   256→   256→            },
   257→   257→            onDelta: (d) => {
   258→   258→              const id = doneId ?? liveId
   259→   259→              setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, text: t.text + d } : t)))
   260→   260→            },
   261→   261→            onDone: (messageId, modelLabel) => {
   262→   262→              doneId = messageId
   263→   263→              // finalize IN PLACE (same DOM node) + reveal the buffers below text
   264→   264→              setTurns((prev) =>
   265→   265→                prev.map((t) =>
   266→   266→                  t.id === liveId
   267→   267→                    ? {
   268→   268→                        ...t,
   269→   269→                        id: messageId,
   270→   270→                        streaming: false,
   271→   271→                        modelLabel,
   272→   272→                        card: buf.card,
   273→   273→                        cards: buf.cards,
   274→   274→                        actions: buf.actions,
   275→   275→                      }
   276→   276→                    : t,
   277→   277→                ),
   278→   278→              )
   279→   279→              setRevealId(messageId)
   280→   280→            },
   281→   281→            onError: (message) => {
   282→   282→              errored = true
   283→   283→              patchTurn({ streaming: false })
   284→   284→              setFailedId(doneId ?? liveId)
   285→   285→              toast({ title: 'مشكلة', description: message, variant: 'destructive' })
   286→   286→            },
   287→   287→          },
   288→   288→        )
   289→   289→      } catch {
   290→   290→        errored = true
   291→   291→        patchTurn({ streaming: false })
   292→   292→        setFailedId(doneId ?? liveId)
   293→   293→        toast({ title: 'مشكلة', description: 'الاتصال اتقطع — جرّب تاني', variant: 'destructive' })
   294→   294→      } finally {
   295→   295→        sendingRef.current = false
   296→   296→        setSending(false)
   297→   297→        if (errored) {
   298→   298→          // keep the failed bubble (retry button) — the server never persisted
   299→   299→          // a reply, so a refetch here would wipe the retry UI
   300→   300→        } else if (doneId) {
   301→   301→          // chat state is already exact locally — refresh the app-wide caches
   302→   302→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   303→   303→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.operations(business.id) })
   304→   304→          // the conversation list (title/preview/counts) + suggestions follow
   305→   305→          // the user's work and update live after every turn
   306→   306→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.conversations(business.id, userId) })
   307→   307→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.suggestions(business.id, userId) })
   308→   308→        } else {
   309→   309→          // stream closed without done/error — treat as a failed turn
   310→   310→          patchTurn({ streaming: false })
   311→   311→          setFailedId(doneId ?? liveId)
   312→   312→          toast({ title: 'مشكلة', description: 'الرد مشكتلش — جرّب تاني', variant: 'destructive' })
   313→   313→        }
   314→   314→      }
   315→   315→    },
   316→   316→    [business.id, userId, modelKey, activeConversationId, loadedConvId, toast, queryClient],
   317→   317→  )
   318→   318→
   319→   319→  const retry = useCallback(() => {
   320→   320→    if (retryRef.current) void send(retryRef.current)
   321→   321→  }, [send])
   322→   322→
   323→   323→  // ---- suggestion / quick-command chips send their prompt directly ----
   324→   324→  const sendSuggestion = useCallback(
   325→   325→    (s: SuggestionDTO) => {
   326→   326→      void send(s.prompt, { suggestionId: s.id })
   327→   327→    },
   328→   328→    [send],
   329→   329→  )
   330→   330→
   331→   331→  const handleCommandTap = useCallback(
   332→   332→    (cmd: QuickCommandDTO) => {
   333→   333→      void send(cmd.prompt)
   334→   334→      // silent usage ping — updates «استخدمته X مرة» in the manager
   335→   335→      api
   336→   336→        .touchQuickCommand(cmd.id)
   337→   337→        .then((updated) => {
   338→   338→          queryClient.setQueryData<QuickCommandDTO[]>(QUERY_KEYS.quickCommands(business.id), (old) =>
   339→   339→            old?.map((c) => (c.id === updated.id ? updated : c)),
   340→   340→          )
   341→   341→        })
   342→   342→        .catch(() => {})
   343→   343→    },
   344→   344→    [send, business.id, queryClient],
   345→   345→  )
   346→   346→
   347→   347→  // ---- «محادثة جديدة» → create + activate + empty hero ----
   348→   348→  const startNewChat = useCallback(async () => {
   349→   349→    if (creatingRef.current) return
   350→   350→    creatingRef.current = true
   351→   351→    setCreating(true)
   352→   352→    try {
   353→   353→      const summary = await api.createConversation(business.id, userId)
   354→   354→      setActiveConversation(summary.id)
   355→   355→      setTurns([])
   356→   356→      setRevealId(null)
   357→   357→      setFailedId(null)
   358→   358→      setDrawerOpen(false)
   359→   359→      void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.conversations(business.id, userId) })
   360→   360→    } catch (err) {
   361→   361→      toast({
   362→   362→        title: 'مشكلة',
   363→   363→        description: err instanceof Error ? err.message : 'جرّب تاني',
   364→   364→        variant: 'destructive',
   365→   365→      })
   366→   366→    } finally {
   367→   367→      creatingRef.current = false
   368→   368→      setCreating(false)
   369→   369→    }
   370→   370→  }, [business.id, userId, setActiveConversation, queryClient, toast])
   371→   371→
   372→   372→  // ---- approvals (R2+ / R3 reason flow) ----
   373→   373→  const approve = useCallback(
   374→   374→    async (card: ActionCardDTO, reason?: string) => {
   375→   375→      if (!card.opId) return
   376→   376→      setBusyOpId(card.opId)
   377→   377→      try {
   378→   378→        const res = await api.confirmOperation(card.opId, 'approve', userId, reason)
   379→   379→        setTurns((prev) =>
   380→   380→          prev.map((t) =>
   381→   381→            t.card?.opId === card.opId
   382→   382→              ? { ...t, card: { ...t.card!, status: res.operation.status }, outcome: res.outcome ?? undefined }
   383→   383→              : t,
   384→   384→          ),
   385→   385→        )
   386→   386→        toast({
   387→   387→          title: res.outcome?.ok ? 'تم — موثّق ومتحقق' : 'العملية فشلت بأمان',
   388→   388→          description: res.outcome?.ok
   389→   389→            ? `${res.operation.ref} — ${res.operation.summary}`
   390→   390→            : (res.outcome?.failReason ?? 'راجع التفاصيل'),
   391→   391→          variant: res.outcome?.ok ? 'default' : 'destructive',
   392→   392→        })
   393→   393→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   394→   394→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.operations(business.id) })
   395→   395→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.items(business.id) })
   396→   396→      } catch (err) {
   397→   397→        toast({
   398→   398→          title: 'مشكلة في الاعتماد',
   399→   399→          description: err instanceof Error ? err.message : 'جرّب تاني',
   400→   400→          variant: 'destructive',
   401→   401→        })
   402→   402→      } finally {
   403→   403→        setBusyOpId(null)
   404→   404→      }
   405→   405→    },
   406→   406→    [userId, business.id, toast, queryClient],
   407→   407→  )
   408→   408→
   409→   409→  const reject = useCallback(
   410→   410→    async (card: ActionCardDTO) => {
   411→   411→      if (!card.opId) return
   412→   412→      setBusyOpId(card.opId)
   413→   413→      try {
   414→   414→        const res = await api.confirmOperation(card.opId, 'reject', userId)
   415→   415→        setTurns((prev) =>
   416→   416→          prev.map((t) => (t.card?.opId === card.opId ? { ...t, card: { ...t.card!, status: 'CANCELLED' } } : t)),
   417→   417→        )
   418→   418→        toast({ title: 'اتلغت المسودة', description: `${res.operation.ref} اتلغت بلا أي أثر` })
   419→   419→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   420→   420→      } catch (err) {
   421→   421→        toast({ title: 'مشكلة', description: err instanceof Error ? err.message : 'جرّب تاني', variant: 'destructive' })
   422→   422→      } finally {
   423→   423→        setBusyOpId(null)
   424→   424→      }
   425→   425→    },
   426→   426→    [userId, business.id, toast, queryClient],
   427→   427→  )
