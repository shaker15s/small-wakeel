    51→// ---------------------------------------------------------------------------
    52→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
    53→// and a hard floor so a slow provider never gets delay stacked on top of it.
    54→// ---------------------------------------------------------------------------
    55→const REPLY_PACING_BUDGET_MS = 1500
    56→const PIECE_DELAY_MIN_MS = 3
    57→const PIECE_DELAY_MAX_MS = 26
    58→
    59→interface PacerState {
    60→  t0: number | null
    61→  emitted: number
    62→}
    63→
    64→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
    65→ * The moment elapsed time eats the whole budget (the provider itself is slow),
    66→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
    67→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
    68→  const now = Date.now()
    69→  if (p.t0 == null) p.t0 = now
    70→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
    71→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
    72→  // Estimate the pieces still coming: at least the rest of this chunk, and at
    73→  // least as many as we've already emitted (replies rarely stop abruptly).
    74→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
    75→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
    76→}

    78→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
    79→ * (identity/memory replies: whole text known upfront, exact remaining count). */
    80→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
    81→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
    82→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
    83→  // project targets ES2017, where the `s` flag is a tsc error).
    84→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
    85→  const t0 = Date.now()
    86→  for (let i = 0; i < pieces.length; i++) {
    87→    emit({ event: 'delta', text: pieces[i] })
    88→    const remainingPieces = pieces.length - i - 1
    89→    if (remainingPieces <= 0) break
    90→    const remainingBudget = budgetMs - (Date.now() - t0)
    91→    if (remainingBudget <= 0) {
    92→      await sleep(PIECE_DELAY_MIN_MS)
    93→      continue
    94→    }
    95→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
    96→  }
    97→}

   129→/** «انت مين» / «اسمك ايه» / «بتعمل ايه»… (run on normalizeAr'd text).

   131→const IDENTITY_QUESTION_RES: RegExp[] = [
   132→  /(^|\s)(انت|انتا|انتي)\s*(مين|من)\s*(انت|انتي|يعني)?(\s|$|؟|\?)/,
   133→  /(^|\s)مين\s*(انت|انتا|انتي)(\s|$|؟|\?)/,
   134→  /(^|\s)اسمك(\s|$)/,
   135→  /بتاع\s*(مين|ايه)/,
   136→  /عرفني\s*بنفسك/,
   137→  /بتعمل\s*(ايه|اد\s*ايه)/,
   138→  /(^|\s)انت\s*(ايه|مين)(\s|$|؟|\?)/,
   139→]

   141→/** Strong business keywords mean the message is NOT about the agent's identity. */
   142→const IDENTITY_BLOCK_RE =
   143→  /(بعت|بيع|شرا|اشتري|خزن|كاش|مخزون|ناقص|ملخص|صلاحيه|مديون|مورد|تحصيل|مصروف|سدد|اشتراك|حضور|تسويه|عكس|احفظ|تسجل|سجل بيع)/

   145→function isIdentityQuestion(norm: string): boolean {
   146→  if (norm.length > 80 || IDENTITY_BLOCK_RE.test(norm)) return false
   147→  return IDENTITY_QUESTION_RES.some((re) => re.test(norm))
   148→}

   150→const MEMORY_SAVE_RE = /(احفظ|خليك فاكر|افتكر اني|انا بفضل|بستحب)/
   151→const DIACRITICS = '[\\u064B-\\u0652\\u0670\\u0640]*'
   152→const CHAR_ALTS: Record<string, string> = { ا: '[اأإآٱ]', ي: '[يى]', ه: '[هة]' }
   153→
   154→/** Keyword → regex that tolerates diacritics/hamza variants in the ORIGINAL text. */
   155→function flexiblePattern(kw: string): RegExp {
   156→  const parts = Array.from(kw).map((c) => CHAR_ALTS[c] ?? c.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
   157→  return new RegExp(parts.join(DIACRITICS))
   158→}

   160→/** Extract the to-remember text from «احفظ اني…» style requests, keeping the
   161→ * user's own words. Returns null when nothing memorable follows the keyword. */
   162→function extractMemorySave(text: string, norm: string): string | null {
   163→  if (!MEMORY_SAVE_RE.test(norm) || norm.length > 260) return null
   164→  for (const kw of ['خليك فاكر اني', 'خليك فاكر', 'افتكر اني', 'احفظ', 'انا بفضل', 'بستحب']) {
   165→    const m = flexiblePattern(kw).exec(text)
   166→    if (!m) continue
   167→    const raw = text.slice(m.index + m[0].length)
   168→    const cleaned = raw
   169→      .replace(/^[،,.\s]+/, '')
   170→      .replace(/^(?:لي\s+)?(?:اني|إني|أني|ان|إن)\s+/, '')
   171→      .replace(/[،,.!؟?\s]+$/, '')
   172→      .trim()
   173→      .slice(0, 200)
   174→    return cleaned.length >= 3 ? cleaned : null
   175→  }
   176→  return null
   177→}

   222→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   223→  const s = normalizeAr(text)
   224→  const isGym = businessType === 'gym'
   225→
   226→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   227→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   228→  }
   229→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   230→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   231→  if (/(صلاحيه|منتهيه|قربت)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   232→  if (/(مديون|اللي ليا|تحصيلات|عليهم|العملاء|الاجل)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   233→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   234→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   235→  if (/حضور/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   236→  if (/(اشتراك|تجديد|جديد لـ)/.test(s) && isGym) return { intent: 'write', opType: 'membership', fields: {} }
   237→  if (/(مصروف|دفعت|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   238→  if (/(تحصيل|خلص|سدد|دفع)/.test(s) && /(من|لـ)/.test(s)) {
   239→    return { intent: 'write', opType: isGym ? 'collection' : 'collection', fields: {} }
   240→  }
   241→  if (/(بعت|بيع|خرج|سلفت)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   242→  if (/(اشتريت|شرا|جبت|دخلت)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   243→  return { intent: 'chat', fields: {} }
   244→}

   246→function parseIntentJson(raw: string): ExtractedIntent | null {
   247→  const cleaned = raw.replace(/```json|```/g, '').trim()
   248→  const start = cleaned.indexOf('{')
   249→  const end = cleaned.lastIndexOf('}')
   250→  if (start < 0 || end <= start) return null
   251→  try {
   252→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent>
   253→    if (!json.intent) return null
   254→    return {
   255→      intent: json.intent === 'read' || json.intent === 'write' || json.intent === 'chat' ? json.intent : 'chat',
   256→      tool: json.tool,
   257→      opType: json.opType as OpType | undefined,
   258→      fields: (json.fields as Record<string, unknown>) ?? {},
   259→      question: json.question,
   260→      replyHint: json.replyHint,
   261→    }
   262→  } catch {
   263→    return null
   264→  }
   265→}

   271→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   272→  for (const c of cards) {
   273→    if (c.kind !== 'table' || !c.footer) continue
   274→    for (const f of c.footer) {
   275→      const m = String(f.value).match(/[\d٠-٩]+/)
   276→      if (m) {
   277→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   278→        const n = Number(west)
   279→        if (Number.isFinite(n)) return n
   280→      }
   281→    }
   282→  }
   283→  return null
   284→}

   332→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit): Promise<ResolvedWrite> {
   333→  const { business, user } = ctx
   334→  const type = (intent.opType ?? 'sale') as OpType
   335→  const f = intent.fields ?? {}
   336→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   337→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   338→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   339→
   340→  const text = ctx.text
   341→  const norm = normalizeAr(text)
   342→
   343→  switch (type) {
   344→    case 'sale':
   345→    case 'purchase': {
   346→      // Item
   347→      const rawItemName = String(f.itemName ?? '')
   348→      const itemMatch = rawItemName ? matchEntity(rawItemName, items) : null
   349→      const item = itemMatch?.entity ?? matchEntity(text, items)?.entity ?? null
   350→      if (!item) {
   351→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: `مش لاقي الصنف — ممكن تكتبه بالاسم اللي في الكتالوج؟` }
   352→      }
   353→      // Quantity + unit
   354→      let { qty, unit } = parseQuantity(`${f.qty ?? ''} ${f.unit ?? ''}`.trim() || text, null)
   355→      if (qty == null) {
   356→        const q = parseQuantity(text, null)
   357→        qty = q.qty
   358→        unit = unit ?? q.unit
   359→      }
   360→      if (qty == null || qty <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: 'الكمية قد إيه بالظبط؟' }
   361→      // Arabic unit words from the LLM (e.g. «جرام») → canonical codes
   362→      const rawUnit = unitToCode(unit) ?? unitToCode(f.unit as string) ?? (unit ?? null)
   363→      if (!rawUnit) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الوحدة', failMessage: 'الوحدة إيه؟ (جرام / كيلو / مللي / علبة...)' }
   364→      const baseQty = convertToBaseUnit(qty, rawUnit, item.unit)
   365→      if (baseQty == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})` }
   366→      const qtyN = Math.round(baseQty * 1000) / 1000
   367→      if (type === 'sale' && qtyN > item.stockQty) {
   368→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}` }
   369→      }
   370→      // Total from REAL price
   371→      const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   372→      const total = Math.round(qtyN * unitPrice)
   373→      // Paid
   374→      let paid = total
   375→      const paidWord = norm.match(/(?:دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   376→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
   377→      else if (paidWord?.[1]) {
   378→        const n = parseQuantity(paidWord[1], null).qty
   379→        if (n != null && n > 0) paid = Math.round(n * 100)
   380→      }
   381→      paid = Math.min(paid, total)
   382→      // Party
   383→      const customerName = String(f.customerName ?? '')
   384→      const customer = customerName ? matchEntity(customerName, customers)?.entity : null
   385→      const supplierName = String(f.supplierName ?? '')
   386→      const supplier = supplierName ? matchEntity(supplierName, suppliers)?.entity : null
   387→
   388→      const payload: Record<string, unknown> = {
   389→        itemId: item.id,
   390→        itemName: item.name,
   391→        qty: qtyN,
   392→        unit: item.unit,
   393→        unitPrice,
   394→        total,
   395→        paid,
   396→        ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   397→        ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   398→      }
   399→      const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   400→      const risk = computeRisk(type, total)
   401→      const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${fmtQty(qtyN, arUnit(item.unit))} ${item.name} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   402→      const card = fieldsToCard(ref, type, risk, summary, [
   403→        { label: 'الصنف', value: item.name },
   404→        { label: 'الكمية', value: fmtQty(qtyN, arUnit(item.unit)) },
   405→        { label: `سعر ${type === 'sale' ? 'البيع' : 'الشرا'} (للوحدة)`, value: egp(unitPrice) },
   406→        { label: 'الإجمالي', value: egp(total) },
   407→        { label: 'مدفوع', value: egp(paid) },
   408→        ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   409→        ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   410→        ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   411→      ], [
   412→        { label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + (type === 'sale' ? -qtyN : qtyN), arUnit(item.unit))}`, tone: 'neutral' },
   413→        { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' : 'bad' },
   414→      ], RISK_META[risk].hint)
   415→      return { ok: true, opType: type, payload, summary, card }
   416→    }
   417→
   418→    case 'expense': {
   419→      const amountRaw = f.amount as number | undefined
   420→      let amount = amountRaw && amountRaw > 0 ? Math.round(amountRaw * 100) : null
   421→      if (amount == null) {
   422→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   423→        if (m) {
   424→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d))).replace(',', '.')
   425→          const parsed = parseFloat(western)
   426→          if (!Number.isNaN(parsed)) amount = Math.round(parsed * 100)
   427→        }
   428→      }
   429→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: 'المبلغ قد إيه؟' }
   430→      const label = String(f.label ?? '') || (norm.includes('كهربا') ? 'كهربا' : norm.includes('ايجار') ? 'إيجار' : norm.includes('تليفون') ? 'تليفون' : 'مصروف')
   431→      const payload = { label, total: amount }
   432→      const risk = computeRisk(type, amount)
   433→      const summary = `مصروف ${label} بـ${egp(amount)}`
   434→      const card = fieldsToCard('', type, risk, summary, [
   435→        { label: 'البند', value: label },
   436→        { label: 'المبلغ', value: egp(amount) },
   437→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
   438→      return { ok: true, opType: type, payload, summary, card }
   439→    }
   440→
   441→    case 'collection': {
   442→      const customersAll = customers.length > 0 ? customers : []
   443→      const customerName = String(f.customerName ?? '')
   444→      const customer = matchEntity(customerName || text, customersAll)?.entity ?? null
   445→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العميل', failMessage: 'تحصيل من مين بالظبط؟' }
   446→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
   447→      if (amount == null) {
   448→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   449→        if (m) {
   450→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   451→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
   452→        }
   453→      }
   454→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هتحصّل قد إيه من ${customer.name}؟` }
   455→      if (amount > customer.balance) {
   456→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `${customer.name} عليه ${egp(customer.balance)} بس — مش ممكن تحصّل ${egp(amount)}` }
   457→      }
   458→      const payload = { customerId: customer.id, customerName: customer.name, amount }
   459→      const risk = computeRisk(type, amount)
   460→      const summary = `تحصيل ${egp(amount)} من ${customer.name}`
   461→      const card = fieldsToCard('', type, risk, summary, [
   462→        { label: 'العميل', value: customer.name },
   463→        { label: 'المبلغ', value: egp(amount) },
   464→        { label: 'المديونية الحالية', value: egp(customer.balance) },
   465→        { label: 'بعد التحصيل', value: egp(customer.balance - amount) },
   466→      ], [{ label: 'الخزنة', value: egp(amount), tone: 'good' }], RISK_META[risk].hint)
   467→      return { ok: true, opType: type, payload, summary, card }
   468→    }
   469→
   470→    case 'payment': {
   471→      const supplierName = String(f.supplierName ?? '')
   472→      const supplier = matchEntity(supplierName || text, suppliers)?.entity ?? null
   473→      if (!supplier) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المورد', failMessage: 'هسدد لمين؟' }
   474→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
   475→      if (amount == null) {
   476→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   477→        if (m) {
   478→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   479→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
   480→        }
   481→      }
   482→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هسدد قد إيه لـ${supplier.name}؟` }
   483→      if (amount > supplier.balance) {
   484→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المستحق لـ${supplier.name} ${egp(supplier.balance)} بس` }
   485→      }
   486→      const payload = { supplierId: supplier.id, supplierName: supplier.name, amount }
   487→      const risk = computeRisk(type, amount)
   488→      const summary = `سداد ${egp(amount)} لـ${supplier.name}`
   489→      const card = fieldsToCard('', type, risk, summary, [
   490→        { label: 'المورد', value: supplier.name },
   491→        { label: 'المبلغ', value: egp(amount) },
   492→        { label: 'المستفق الحالي', value: egp(supplier.balance) },
   493→        { label: 'بعد السداد', value: egp(supplier.balance - amount) },
   494→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
   495→      return { ok: true, opType: type, payload, summary, card }
   496→    }
   497→
   498→    case 'membership': {
   499→      const isRenewal = norm.includes('تجديد') || norm.includes('جدد')
   500→      const customerName = String(f.customerName ?? '')
   501→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
   502→      if (!customer && isRenewal) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'تجديد لمين؟' }
   503→      const plan = String(f.plan ?? 'شهرية')
   504→      const months = Math.max(1, Math.min(12, Math.round(Number(f.months ?? 1))))
   505→      // plan price from items (gym plans are items) or fallback 35000
   506→      const planItem = items.find((i) => normalizeAr(i.name).includes(normalizeAr(plan)))
   507→      const total = Math.round((planItem?.sellPrice ?? 35000) * months)
   508→      let paid = total
   509→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
   510→      paid = Math.min(paid, total)
   511→      const name = customer?.name ?? String(f.customerName ?? 'عضو جديد')
   512→      const payload = {
   513→        ...(customer ? { customerId: customer.id } : { customerName: name }),
   514→        plan,
   515→        months,
   516→        total,
   517→        paid,
   518→      }
   519→      const risk = computeRisk(type, total)
   520→      const summary = `${isRenewal ? 'تجديد' : 'اشتراك جديد'} ${plan} لـ${name} — ${egp(total)}`
   521→      const card = fieldsToCard('', type, risk, summary, [
   522→        { label: 'العضو', value: name },
   523→        { label: 'الخطة', value: plan },
   524→        { label: 'المدة', value: `${months} شهر` },
   525→        { label: 'الإجمالي', value: egp(total) },
   526→        { label: 'مدفوع', value: egp(paid) },
   527→      ], [
   528→        { label: 'الخزنة', value: egp(paid), tone: 'good' },
   529→        ...(total - paid > 0 ? [{ label: 'باقي على العضو', value: egp(total - paid), tone: 'bad' as const }] : []),
   530→      ], RISK_META[risk].hint)
   531→      return { ok: true, opType: type, payload, summary, card }
   532→    }
   533→
   534→    case 'attendance': {
   535→      const customerName = String(f.customerName ?? '')
   536→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
   537→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'حضور مين؟' }
   538→      const payload = { customerId: customer.id, customerName: customer.name }
   539→      const risk = computeRisk(type)
   540→      const summary = `حضور ${customer.name}`
   541→      const card = fieldsToCard('', type, risk, summary, [
   542→        { label: 'العضو', value: customer.name },
   543→        { label: 'اليوم', value: 'النهاردة' },
   544→        ...(customer.expiresAt ? [{ label: 'الاشتراك لحد', value: customer.expiresAt.toISOString().slice(0, 10) }] : []),
   545→      ], [], RISK_META[risk].hint)
   546→      return { ok: true, opType: type, payload, summary, card }
   547→    }
   548→
   549→    case 'stock_adjustment': {
   550→      const rawItemName = String(f.itemName ?? '')
   551→      const item = matchEntity(rawItemName || text, items)?.entity ?? null
   552→      if (!item) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: 'تسوية مخزون لأنهي صنف؟' }
   553→      let delta = (f.delta as number | undefined) ?? null
   554→      if (delta == null && typeof f.qty === 'number') delta = f.qty - item.stockQty
   555→      if (delta == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية الجديدة/الفرق', failMessage: `المخزون الحالي ${fmtQty(item.stockQty, arUnit(item.unit))} — عايزه يبقى قد إيه؟` }
   556→      const reason = String(f.reason ?? '') || 'تسوية جرد'
   557→      const payload = { itemId: item.id, itemName: item.name, delta, reason, unit: item.unit }
   558→      const risk = computeRisk(type)
   559→      const summary = `تسوية مخزون ${item.name}: ${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`
   560→      const card = fieldsToCard('', type, risk, summary, [
   561→        { label: 'الصنف', value: item.name },
   562→        { label: 'الحالي', value: fmtQty(item.stockQty, arUnit(item.unit)) },
   563→        { label: 'الجديد', value: fmtQty(item.stockQty + delta, arUnit(item.unit)) },
   564→        { label: 'السبب', value: reason },
   565→      ], [{ label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`, tone: delta >= 0 ? 'good' : 'bad' }], 'تسوية المخزون عملية حساسة (R3) — محتاجة سبب واعتماد')
   566→      return { ok: true, opType: type, payload, summary, card }
   567→    }
   568→
   569→    case 'reversal': {
   570→      const ref = String(f.ref ?? '') || (norm.match(/WKL-\d+/)?.[0] ?? '')
   571→      if (!ref) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'رقم العملية', failMessage: 'عايز تعكس أنهي عملية؟ اكتب رقمها (زي WKL-1006)' }
   572→      const original = await db.operation.findFirst({ where: { businessId: business.id, ref } })
   573→      if (!original) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `ملقيتش عملية رقم ${ref}` }
   574→      if (original.reversedByRef) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `العملية ${ref} معكوسة قبل كده بـ${original.reversedByRef}` }
   575→      const reason = String(f.reason ?? '') || 'تصحيح خطأ'
   576→      const payload = { originalRef: ref, reason }
   577→      const risk = computeRisk(type)
   578→      const summary = `عكس العملية ${ref} (${original.summary})`
   579→      const card = fieldsToCard('', type, risk, summary, [
   580→        { label: 'العملية الأصلية', value: `${ref} — ${TYPE_AR[original.type] ?? original.type}` },
   581→        { label: 'ملخصها', value: original.summary },
   582→        { label: 'السبب', value: reason },
   583→      ], [{ label: 'الأثر', value: 'هيتلغى بالكامل بعملية عكسية موثقة', tone: 'neutral' }], 'عكس عملية مالية حساس (R3) — التاريخ مش بيتمسح، بيتصحح')
   584→      return { ok: true, opType: type, payload, summary, card }
   585→    }
   586→
   587→    default:
   588→      return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: 'نوع العملية مش مدعوم' }
   589→  }
   590→}

   592→// ---------------------------------------------------------------------------
   593→// The turn runner
   594→// ---------------------------------------------------------------------------
   595→export interface TurnResult {
   596→  replyText: string
   597→  card?: ActionCardDTO | null
   598→  cards?: ResultCardDTO[] | null
   599→  thinking: ThinkingStep[]
   600→  modelLabel: string
   601→  opId?: string
   602→  actions?: ChatActionDTO[] | null
   603→}
   604→
   605→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
   606→  const { business, user, text } = ctx
   607→  const v = getVertical(business.type)
   608→  const model = await resolveModel(ctx.modelKey)
   609→  const modelLabel = modelLabelFor(model)
   610→
   611→  // Load context once (memories raised to 8; pending ops count for the persona)
   612→  const [items, customers, suppliers, memories, pendingCount] = await Promise.all([
   613→    db.item.findMany({ where: { businessId: business.id } }),
   614→    db.customer.findMany({ where: { businessId: business.id } }),
   615→    db.supplier.findMany({ where: { businessId: business.id } }),
   616→    db.agentMemory.findMany({ where: { businessId: business.id, userId: user.id }, orderBy: { createdAt: 'desc' }, take: 8 }),
   617→    db.operation.count({ where: { businessId: business.id, status: 'AWAITING_APPROVAL' } }),
   618→  ])
   619→
   620→  // Thinking steps — with a «فتكرت من ذاكرتي» recall step injected right after
   621→  // «بفهم طلبك» whenever the agent actually remembers something about this user.
   622→  const memStep: ThinkingStep[] =
   623→    memories.length > 0
   624→      ? [
   625→          {
   626→            label: 'فتكرت من ذاكرتي',
   627→            detail: memories[0].text.slice(0, 60) + (memories[0].text.length > 60 ? '…' : ''),
   628→            state: 'done',
   629→            icon: 'verify',
   630→          },
   631→        ]
   632→      : []
   633→  const steps: ThinkingStep[] = [
   634→    { label: 'بفهم طلبك', detail: 'بحلل الجملة وأستخرج القصد والحقول', state: 'active', icon: 'understand' },
   635→    ...memStep,
   636→    { label: 'بدوّر في سجلك', detail: 'بدوّر على الأصناف والأسماء الحقيقية', state: 'pending', icon: 'search' },
   637→    { label: 'بتحقق من الأرقام', detail: 'بحسب من الأسعار الفعلية مش من دماغي', state: 'pending', icon: 'verify' },
   638→    { label: 'بجهّز الرد', detail: '', state: 'pending', icon: 'plan' },
   639→  ]
   640→  const IDX = { understand: 0, search: 1 + memStep.length, verify: 2 + memStep.length, reply: 3 + memStep.length }
   641→  const pushThinking = () => emit({ event: 'thinking', steps: steps.map((s) => ({ ...s })) })
   642→  const done = (i: number, detail?: string) => {
   643→    if (steps[i]) {
   644→      steps[i].state = 'done'
   645→      if (detail) steps[i].detail = detail
   646→    }
   647→    pushThinking()
   648→  }
   649→  pushThinking()
   650→
   651→  const norm = normalizeAr(text)
   652→
   653→  // ---- PASS 0a: Identity question → instant, deterministic, zero LLM ----
   654→  if (isIdentityQuestion(norm)) {
   655→    const lowStockCount = items.filter((i) => i.stockQty <= i.lowStockThreshold).length
   656→    const receivablesTotal = customers.reduce((s, c) => s + Math.max(0, c.balance), 0)
   657→    done(IDX.understand, 'سؤال عن هويتي — بجاوب من سجلك من غير موديل')
   658→    done(IDX.search, 'جبت بياناتي من سجلك الحقيقي')
   659→    done(IDX.verify, 'عدّيت النواقص والعمليات المستنية من قاعدة البيانات')
   660→    emit({ event: 'step', label: 'بتعرّف بنفسي', state: 'done' })
   661→    done(IDX.reply)
   662→    const replyText = buildIdentityReply({
   663→      business,
   664→      user,
   665→      verticalLabel: v.label,
   666→      memoryText: memories[0]?.text ?? null,
   667→      lowStockCount,
   668→      pendingCount,
   669→      receivablesTotal,
   670→    })
   671→    const actions: ChatActionDTO[] = [
   672→      { id: 'act-today', label: 'لوحة اليوم', view: 'today', icon: 'today', tone: 'primary' },
   673→      ...(lowStockCount > 0
   674→        ? [{ id: 'act-inventory', label: 'شوف المخزون', view: 'inventory' as const, icon: 'boxes' as const, tone: 'outline' as const }]
   675→        : []),
   676→    ]
   677→    await streamDeterministic(replyText, emit, 1200)
   678→    emit({ event: 'actions', actions })
   679→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions }
   680→  }
   681→
   682→  // ---- PASS 0b: Memory save → instant, persisted, zero LLM ----
   683→  const memoryText = extractMemorySave(text, norm)
   684→  if (memoryText) {
   685→    await db.agentMemory.create({
   686→      data: { businessId: business.id, userId: user.id, kind: 'preference', text: memoryText },
   687→    })
   688→    done(IDX.understand, 'طلب حفظ في الذاكرة — مش محتاج موديل')
   689→    done(IDX.search, 'بتخزنها في ذاكرتي عنك')
   690→    done(IDX.verify, 'اتحفظت فعليًا في قاعدة البيانات')
   691→    emit({ event: 'step', label: 'بحفظ في ذاكرتي', state: 'done' })
   692→    done(IDX.reply)
   693→    const replyText = `اتحفظت في ذاكرتي ✅ — هفتكرها: «${memoryText}». لما الموقف ده يتكرر هرفعها على طول.`
   694→    await streamDeterministic(replyText, emit, 1000)
   695→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions: null }
   696→  }
   697→
   698→  // ---- PASS 1: Understand (LLM with rule-based fallback) ----
   699→  let intent: ExtractedIntent
   700→  try {
   701→    const system = buildSystemPrompt({ business, user, items, customers, suppliers, memories, history: ctx.history })
   702→    const raw = await completeChat(model, [
   703→      { role: 'system', content: system },
   704→      { role: 'user', content: text },
   705→    ])
   706→    intent = parseIntentJson(raw) ?? ruleBasedIntent(text, business.type)
   707→    done(IDX.understand, intent.replyHint ? `فهمت: ${intent.replyHint}` : 'فهمت الطلب')
   708→  } catch (err) {
   709→    console.error('[harness] intent pass failed → rule-based fallback:', err)
   710→    intent = ruleBasedIntent(text, business.type)
   711→    done(IDX.understand, 'فهمت الطلب بالقواعد المحلية (الموديل مش متاح)')
   712→  }
   713→
   714→  // ---- PASS 2+3: Act ----
   715→  let situation = ''
   716→  let card: ActionCardDTO | null = null
   717→  let cards: ResultCardDTO[] | null = null
   718→  let opId: string | undefined
   719→  let replyInstruction = ''
   720→  let questionMode = false
   721→  let actions: ChatActionDTO[] | null = null
   722→
   723→  if (intent.intent === 'read' && intent.tool) {
   724→    emit({ event: 'step', label: `بشغّل أداة ${intent.tool}`, state: 'active' })
   725→    done(IDX.search, 'لقيت البيانات في السجل')
   726→    cards = await runReadTool(intent.tool as ReadTool, business.id, intent.fields ?? {})
   727→    emit({ event: 'cards', cards })
   728→    emit({ event: 'step', label: `بشغّل أداة ${intent.tool}`, state: 'done' })
   729→    done(IDX.verify, 'الأرقام جاية من قاعدة البيانات مباشرة')
   730→    situation = `شغّلت أداة القراءة ${intent.tool} ودي النتايج:\n${JSON.stringify(cards, null, 1).slice(0, 1600)}`
   731→    replyInstruction = 'لخّص النتيجة للمستخدم بالمصري بأرقامها الصحيحة من الجدول فوق. لو مفيش بيانات قول كده بصراحة.'
   732→    const hint = proactiveHintFor(intent.tool, cards)
   733→    if (hint) replyInstruction += ` ${hint}`
   734→    actions = TOOL_ACTIONS[intent.tool] ?? null
   735→  } else if (intent.intent === 'write' && intent.opType) {
   736→    const opType = intent.opType as OpType
   737→    if (!canDraft(user.role, opType)) {
   738→      situation = `المستخدم (${user.role}) مش مسموحله يسجّل ${TYPE_AR[opType] ?? opType}.`
   739→      replyInstruction = 'وضّح بذوق إن العملية دي محتاجة صلاحية أعلى (المالك أو المشرف) ومتشغلش أي أداة.'
   740→    } else {
   741→      done(IDX.search, 'بمشي على الأصناف والأسماء الحقيقية')
   742→      const resolved = await resolveWrite(ctx, intent, emit)
   743→      if (!resolved.ok) {
   744→        done(IDX.verify, resolved.failMessage ?? '')
   745→        situation = `معلومات ناقصة: ${resolved.missing ?? ''} — ${resolved.failMessage ?? ''}`
   746→        replyInstruction = 'اسأل سؤال واحد قصير يطلب أهم حقل ناقص. متاخدش أي قرار.'
   747→        questionMode = true
   748→      } else {
   749→        done(IDX.verify, `حسبت الإجمالي من السعر الحقيقي: ${resolved.summary}`)
   750→        emit({ event: 'step', label: `بجهّز مسودة ${TYPE_AR[resolved.opType]}`, state: 'active' })
   751→        const idem = `${business.id}:${user.id}:${Date.now()}:${Math.random().toString(36).slice(2, 8)}`
   752→        const draft = await createDraft({
   753→          businessId: business.id,
   754→          userId: user.id,
   755→          type: resolved.opType,
   756→          summary: resolved.summary,
   757→          payload: resolved.payload,
   758→          conversationId: ctx.conversationId,
   759→          idempotencyKey: idem,
   760→        })
   761→        opId = draft.id
   762→        card = { ...resolved.card, ref: draft.ref, opId: draft.id }
   763→        emit({ event: 'card', card })
   764→        emit({ event: 'step', label: `بجهّز مسودة ${TYPE_AR[resolved.opType]}`, state: 'done' })
   765→        situation = `اتعملت مسودة عملة رقم ${draft.ref} (${resolved.summary}) مستنية اعتماد المستخدم. درجة الخطورة ${draft.risk}. مفيش حاجة اتنفذت في الخزنة أو المخزون لسه.`
   766→        replyInstruction = `اكتب تأكيد قصير إن المسودة جاهزة بالأرقام الصح وإنها مستنية اعتماده${draft.risk === 'R3' ? '، وذكّره إنها عملية حساسة محتاجة سبب' : ''}. خليك مختصر.`
   767→        actions = DRAFT_ACTIONS
   768→      }
   769→    }
   770→  } else {
   771→    // chat
   772→    done(IDX.search, 'كلام عام')
   773→    done(IDX.verify, '')
   774→    situation = `رسالة عامة (تحية/سؤال خارج الأدوات). سياق الشغل: ${v.label} — ${business.name}.`
   775→    replyInstruction = `ردّ بالمصري بلطف وقصر. لو مناسب اعرض 2-3 من الأوامر اللي بتنفع ${v.shortLabel}: ${v.chips.slice(0, 3).join(' · ')}`
   776→  }
   777→
   778→  done(IDX.reply)
   779→
   780→  // ---- PASS 4: Explain (letter-by-letter, ADAPTIVE pacing) ----
   781→  let replyText = ''
   782→  const pacer: PacerState = { t0: null, emitted: 0 }
   783→  // NOTE: the reply pass MUST carry a trailing user turn — providers reject a
   784→  // conversation that is only a system/assistant prompt (z-ai: code 1214).
   785→  const messages: LLMMessage[] = [
   786→    { role: 'system', content: buildReplyPrompt({ business: { name: business.name, type: business.type }, userName: user.name, situation, instruction: replyInstruction }) },
   787→    { role: 'user', content: 'اكتب ردك النهائي دلوقتي بناءً على الوضع الحالي والمطلوب فوق.' },
   788→  ]
   789→  try {
   790→    for await (const chunk of streamChat(model, messages)) {
   791→      if (chunk.type !== 'content') continue
   792→      replyText += chunk.text
   793→      // Re-pace into tiny letter-by-letter deltas — adaptively: total pacing
   794→      // is capped by the budget; a slow provider gets the 3ms floor instead.
   795→      // [\s\S] keeps the model's own line breaks alive in the stream
   796→      // (plain `.` drops \n; ES2017 target forbids the dotAll `s` flag).
   797→      const pieces = chunk.text.match(/[\s\S]{1,2}/g) ?? []
   798→      for (let i = 0; i < pieces.length; i++) {
   799→        emit({ event: 'delta', text: pieces[i] })
   800→        pacer.emitted++
   801→        await sleep(adaptivePieceDelay(pacer, pieces.length - i - 1))
   802→      }
   803→    }
   804→  } catch (err) {
   805→    console.error('[harness] reply stream failed:', err)
   806→    if (!replyText) {
   807→      const fallback = card
   808→        ? `جهزتلك مسودة «${card.title}» — ${card.summary}. راجعها واعتمدها لو تمام.`
   809→        : cards
   810→          ? 'دي النتايج من السجل فوق.'
   811→          : questionMode
   812→            ? (situation.match(/— (.+)$/)?.[1] ?? 'محتاج تفاصيل أكتر لو سمحت.')
   813→            : 'تمام، تحت أمرك.'
   814→      await streamDeterministic(fallback, emit, 900)
   815→      replyText = fallback
   816→    }
   817→  }
   818→
   819→  if (!replyText.trim()) {
   820→    const fallback = card ? `جهزت المسودة — ${card.summary}` : 'تمام.'
   821→    await streamDeterministic(fallback, emit, 900)
   822→    replyText = fallback
   823→  }
   824→
   825→  // ---- PASS 5: Deep-link action buttons (after deltas, before done) ----
   826→  if (actions && actions.length > 0) {
   827→    emit({ event: 'actions', actions })
   828→  }
   829→
   830→  return { replyText, card, cards, thinking: steps, modelLabel, opId, actions: actions && actions.length > 0 ? actions : null }
   831→}
