    85→// ---------------------------------------------------------------------------
    86→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
    87→// and a hard floor so a slow provider never gets delay stacked on top of it.
    88→// ---------------------------------------------------------------------------
    89→const REPLY_PACING_BUDGET_MS = 1500
    90→const PIECE_DELAY_MIN_MS = 3
    91→const PIECE_DELAY_MAX_MS = 26
    92→
    93→interface PacerState {
    94→  t0: number | null
    95→  emitted: number
    96→}
    97→
    98→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
    99→ * The moment elapsed time eats the whole budget (the provider itself is slow),
   100→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
   101→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
   102→  const now = Date.now()
   103→  if (p.t0 == null) p.t0 = now
   104→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
   105→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
   106→  // Estimate the pieces still coming: at least the rest of this chunk, and at
   107→  // least as many as we've already emitted (replies rarely stop abruptly).
   108→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
   109→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
   110→}

   112→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
   113→ * (identity/memory replies: whole text known upfront, exact remaining count). */
   114→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
   115→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
   116→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
   117→  // project targets ES2017, where the `s` flag is a tsc error).
   118→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
   119→  const t0 = Date.now()
   120→  for (let i = 0; i < pieces.length; i++) {
   121→    emit({ event: 'delta', text: pieces[i] })
   122→    const remainingPieces = pieces.length - i - 1
   123→    if (remainingPieces <= 0) break
   124→    const remainingBudget = budgetMs - (Date.now() - t0)
   125→    if (remainingBudget <= 0) {
   126→      await sleep(PIECE_DELAY_MIN_MS)
   127→      continue
   128→    }
   129→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
   130→  }
   131→}

   270→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   271→  const s = normalizeAr(text)
   272→  const isGym = businessType === 'gym'
   273→
   274→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   275→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   276→  }
   277→  // Opinion/decision-seeking → ADVISORY free-chat (the real-employee mode).
   278→  // Must beat the price-query rule below so «رأيك أزود سعر الكمون؟» is a
   279→  // conversation about the decision, never a bare catalog scan.
   280→  if (/(رايك|رأيك|ايه رايك|تنصحني|نصيحه|اعمل ايه لو|تشوف ايه|رايك ايه|استشاره)/.test(s) && !/(بعت|بيع|سجل بيع|تحصيل|اشتري)/.test(s)) {
   281→    return { intent: 'chat', fields: {}, replyHint: 'استشارة شغل' }
   282→  }
   283→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   284→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   285→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   286→  }
   287→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   288→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   289→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   290→  if (/(مديون|اللي ليا|تحصيلات|مين عليها|عليها فلوس|عليهم فلوس|العملاء)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   291→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   292→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   293→  // Price query («ايه سعر عود كمبودي؟» / «بكام المسك؟») → catalog search (R0 read)
   294→  if (/(سعر|بكام)/.test(s) && !/(بعت|بيع|بع |خرج|سلف|دفع|سجل)/.test(s)) {
   295→    const q = s
   296→      .replace(/^.*?(?:سعر|بكام)\s*/, '')
   297→      .replace(/[؟?].*$/, '')
   298→      .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   299→      .replace(/^\s*(?:في|من)\s*/, '')
   300→      .trim()
   301→    if (q.length >= 2) return { intent: 'read', tool: 'search_catalog', fields: { query: q.slice(0, 40) } }
   302→  }
   303→  // Attendance (gym): noun «حضور» + Egyptian verb forms («كريم حضر»,
   304→  // «لعب حصة PT», «خلص تدريب») — a member WORKED OUT today.
   305→  if (/(حضور|(^| )حضر( |$)|لعب حصه?|خلص (تدريب|حصه?)|اتمرن|اتدرب)/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   306→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   307→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   308→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   309→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   310→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   311→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   312→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   313→    if (name && (isQuestion || /عضويه/.test(s))) {
   314→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   315→    }
   316→  }
   317→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   318→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   319→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   320→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   321→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   322→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   323→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   324→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   325→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   326→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   327→  // fire a collection (caught by the expiring rule above instead).
   328→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   329→    return { intent: 'write', opType: 'collection', fields: {} }
   330→  }
   331→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   332→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   333→  return { intent: 'chat', fields: {} }
   334→}

   364→/** Parse the LLM JSON contract: single intent OR the "intents" array (multi). */
   365→function parseIntentJsonMulti(raw: string): { intents: RawIntentJson[]; single: ExtractedIntent | null } {
   366→  const cleaned = raw.replace(/```json|```/g, '').trim()
   367→  const start = cleaned.indexOf('{')
   368→  const end = cleaned.lastIndexOf('}')
   369→  if (start < 0 || end <= start) return { intents: [], single: null }
   370→  try {
   371→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent> & { intents?: RawIntentJson[] }
   372→    if (Array.isArray(json.intents)) {
   373→      const valid = json.intents.filter((r) => r && typeof r.intent === 'string')
   374→      if (valid.length >= 2) return { intents: valid, single: null }
   375→      if (valid.length === 1) return { intents: [], single: rawToSingle(valid[0]) }
   376→      return { intents: [], single: null }
   377→    }
   378→    if (!json.intent) return { intents: [], single: null }
   379→    return { intents: [], single: rawToSingle(json as RawIntentJson) }
   380→  } catch {
   381→    return { intents: [], single: null }
   382→  }
   383→}

   389→    intent,

   393→    question: json.question,

   398→/** Read-request trigger keywords → tool (used by the deterministic splitter). */
   399→const READ_SPLITTERS: [RegExp, string][] = [
   400→  [/ملخص|حال اليوم|كويس اليوم|عملت ايه النهارده|اقفل الورديه/, 'get_daily_summary'],
   401→  [/وضع الخزنه|رصيد الخزنه|الخزنه|الكاش/, 'get_cash_position'],
   402→  [/النواقص|المخزون الناقص|الناقص|المخزون|استوك/, 'list_low_stock'],
   403→  [/صلاحيه?|منتهي|قربت تخلص|تنتهي/, 'list_expiring'],
   404→  [/مين عليها|عليها فلوس|عليهم فلوس|مديون|تحصيلات|اللي ليا/, 'list_receivables'],
   405→  [/اللي علينا|للموردين|الموردين|مستحقات/, 'list_payables'],
   406→  [/آخر الحركات|اخر الحركات|سجل العمليات|آخر العمليات|اخر العمليات|العمليات المستنيه/, 'get_activity_history'],
   407→  [/سعر|بكام/, 'search_catalog'],
   408→]
   409→
   410→/** Sequence verbs that START a new request mid-message (never paid-clauses). */
   411→const WRITE_CUT_RE = /بعت|اشتريت|اشتري|سجل|حصلت|قبضت|استلمت|سدد|صرفت|مصروف|تسويه|عكسها|اعكس/
   412→
   413→/**
   414→ * Deterministic compound splitter — «ملخص اليوم ومين عليها فلوس وايه الناقص
   415→ * في المخزون» → 3 read sub-intents. Cuts at read-trigger keywords, sequence
   416→ * verbs (بعت/سجل/…) and connectors (و بعدين/كمان) — never inside a multi-item
   417→ * basket («بعت ٢ لبن و ١٠ بيض» stays ONE write). Zero LLM.
   418→ */
   419→function splitSubIntents(text: string, businessType: string): SubIntent[] {
   420→  const norm = toWesternDigits(normalizeAr(text))
   421→  if (norm.length < 8) return []
   422→  type Hit = { at: number; len: number; tool: string | null }
   423→  const hits: Hit[] = []
   424→  for (const [re, tool] of READ_SPLITTERS) {
   425→    const g = new RegExp(re.source, 'g')
   426→    let m: RegExpExecArray | null
   427→    while ((m = g.exec(norm)) != null) {
   428→      hits.push({ at: m.index, len: m[0].length, tool })
   429→      if (m.index === g.lastIndex) g.lastIndex++
   430→    }
   431→  }
   432→  // sequence verbs — only when a و/، actually precedes them (mid-message cut)
   433→  {
   434→    const g = new RegExp(WRITE_CUT_RE.source, 'g')
   435→    let m: RegExpExecArray | null
   436→    while ((m = g.exec(norm)) != null) {
   437→      const at = m.index
   438→      const prev = at > 0 ? norm[at - 1] : ''
   439→      const prev2 = at > 1 ? norm[at - 2] : ''
   440→      const prev3 = at > 2 ? norm[at - 3] : ''
   441→      const isBoundary =
   442→        prev === 'و' || prev === '،' || prev === ',' || prev === '؛' || (prev === ' ' && (prev2 === 'و' || prev2 === '،')) || (prev === ' ' && prev2 === ' ' && prev3 === 'و')
   443→      if (isBoundary) hits.push({ at, len: m[0].length, tool: null })
   444→      if (m.index === g.lastIndex) g.lastIndex++
   445→    }
   446→  }
   447→  // connectors «و بعدين» — cut at the connector itself
   448→  {
   449→    const g = /(بعدين|بعد كده|كمان)/g
   450→    let m: RegExpExecArray | null
   451→    while ((m = g.exec(norm)) != null) {
   452→      hits.push({ at: m.index, len: m[0].length, tool: null })
   453→      if (m.index === g.lastIndex) g.lastIndex++
   454→    }
   455→  }
   456→  hits.sort((a, b) => a.at - b.at)
   457→  if (hits.length === 0) return []
   458→
   459→  // dedupe overlaps + same-tool continuations, build cut points
   460→  const cuts: { at: number; tool: string | null }[] = []
   461→  let coveredEnd = -1
   462→  let currentTool: string | null = null
   463→  for (const h of hits) {
   464→    if (h.at < coveredEnd) continue // overlapping keyword (الخزنه inside وضع الخزنه)
   465→    if (h.tool != null && h.tool === currentTool) {
   466→      // same tool still talking — «سعر العنبر بكام؟» is ONE search, and
   467→      // «سعر العود وبكام المسك» is one question about two items (the catalog
   468→      // matcher is bidirectional-includes, so both still show in one card)
   469→      coveredEnd = Math.max(coveredEnd, h.at + h.len)
   470→      continue
   471→    }
   472→    // cut position: include a glued و in the PREVIOUS segment
   473→    const cutAt = h.at > 0 && norm[h.at - 1] === 'و' ? h.at - 1 : h.at
   474→    cuts.push({ at: cutAt, tool: h.tool })
   475→    coveredEnd = h.at + h.len
   476→    currentTool = h.tool
   477→  }
   478→  if (cuts.length === 0) return []
   479→
   480→  // build segments
   481→  const segments: { text: string; tool: string | null }[] = []
   482→  const seg0 = norm.slice(0, cuts[0].at).replace(/[\s،,؛]+$/, '').trim()
   483→  if (seg0.length > 0) segments.push({ text: seg0, tool: null })
   484→  for (let i = 0; i < cuts.length; i++) {
   485→    const end = i + 1 < cuts.length ? cuts[i + 1].at : norm.length
   486→    const seg = norm.slice(cuts[i].at, end).replace(/^[\s،,؛و]+/, '').replace(/[\s،,؛]+$/, '').trim()
   487→    if (seg.length > 0) segments.push({ text: seg, tool: cuts[i].tool })
   488→  }
   489→
   490→  // map segments → sub-intents
   491→  const subs: SubIntent[] = []
   492→  for (let i = 0; i < segments.length; i++) {
   493→    const seg = segments[i]
   494→    // politeness prefixes («قوللي», «اط») — skip chat-y tiny segments
   495→    if (seg.tool == null) {
   496→      if (i === 0 && seg.text.length <= 6) continue
   497→      const mapped = ruleBasedIntent(seg.text, businessType)
   498→      if (mapped.intent === 'read' && mapped.tool) subs.push({ ...mapped, intent: 'read', tool: mapped.tool, sourceText: seg.text })
   499→      else if (mapped.intent === 'write' && mapped.opType) subs.push({ ...mapped, intent: 'write', sourceText: seg.text })
   500→      continue
   501→    }
   502→    // tool-owned segment
   503→    if (seg.tool === 'search_catalog') {
   504→      const q = seg.text
   505→        .replace(/^(?:ايه|ايه هو|هو)?\s*(?:سعر|بكام)\s*/, '')
   506→        .replace(/[؟?].*$/, '')
   507→        .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   508→        .replace(/^\s*(?:في|من)\s*/, '')
   509→        .trim()
   510→      subs.push({ intent: 'read', tool: 'search_catalog', fields: q.length >= 2 ? { query: q.slice(0, 40) } : {}, sourceText: seg.text })
   511→    } else {
   512→      subs.push({ intent: 'read', tool: seg.tool, fields: {}, sourceText: seg.text })
   513→    }
   514→  }
   515→
   516→  // merge CONSECUTIVE same-opType writes into one basket (multi-item scan)
   517→  const merged: SubIntent[] = []
   518→  for (const s of subs) {
   519→    const last = merged[merged.length - 1]
   520→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   521→      last.sourceText = `${last.sourceText} و ${s.sourceText}`
   522→      last.fields = { ...last.fields, ...s.fields }
   523→      continue
   524→    }
   525→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   526→      continue // duplicate read
   527→    }
   528→    merged.push({ ...s })
   529→  }
   530→  return merged
   531→}
   532→
   533→/** Normalize LLM multi-intents: validate, merge consecutive same-opType writes. */
   534→function normalizeMultiIntents(raws: RawIntentJson[], fullText: string): SubIntent[] {
   535→  const subs: SubIntent[] = []
   536→  for (const r of raws) {
   537→    if (r.intent === 'read' && r.tool && READ_TOOLS.has(r.tool)) {
   538→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   539→      subs.push({ intent: 'read', tool: r.tool as ReadTool, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   540→    } else if (r.intent === 'write' && r.opType && VALID_OP_TYPES.has(r.opType)) {
   541→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   542→      subs.push({ intent: 'write', opType: r.opType as OpType, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   543→    }
   544→  }
   545→  const merged: SubIntent[] = []
   546→  for (const s of subs) {
   547→    const last = merged[merged.length - 1]
   548→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   549→      // consecutive same-type writes = ONE basket; full-text scan finds the lines
   550→      last.sourceText = fullText
   551→      last.fields = { ...last.fields, ...s.fields }
   552→      continue
   553→    }
   554→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   555→      continue // duplicate read
   556→    }
   557→    merged.push(s)
   558→  }
   559→  return merged
   560→}

   566→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   567→  for (const c of cards) {
   568→    if (c.kind !== 'table' || !c.footer) continue
   569→    for (const f of c.footer) {
   570→      const m = String(f.value).match(/[\d٠-٩]+/)
   571→      if (m) {
   572→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   573→        const n = Number(west)
   574→        if (Number.isFinite(n)) return n
   575→      }
   576→    }
   577→  }
   578→  return null
   579→}

   581→function proactiveHintFor(tool: string, cards: ResultCardDTO[]): string {
   582→  if (tool === 'list_low_stock') {
   583→    const n = tableFooterCount(cards)
   584→    if (n != null && n > 0) return `قفل جملة قصيرة تنبّه على ${n} صنف ناقص من غير ما تكرر الأرقام كلها.`

   592→      .flatMap((c) => (c.kind === 'table' ? c.rows : []))

   595→  }
   596→  return ''

   599→/**
   600→ * Deterministic reply for read turns when the LLM reply stream fails
   601→ * (rate-limit / network). NEVER generic — quotes the REAL numbers the tools
   602→ * just fetched, so the owner still gets a useful answer with the data below.
   603→ */
   604→function deterministicReadReply(cards: ResultCardDTO[]): string {
   605→  for (const c of cards) {
   606→    if (c.kind === 'kv' && c.rows.length > 0) {
   607→      const parts = c.rows.slice(0, 4).map((r) => `${r.label} ${r.value}`).join(' · ')
   608→      return `دي أرقام «${c.title}» من سجلك: ${parts}.`
   609→    }
   610→  }
   611→  const table = cards.find((c) => c.kind === 'table') as Extract<ResultCardDTO, { kind: 'table' }> | undefined
   612→  if (table) {
   613→    const n = table.rows.length
   614→    return `دي «${table.title}» من سجلك — ${toArabicNum(n)} ${n === 1 ? 'صف' : n === 2 ? 'صفين' : 'صفوف'}، التفاصيل في الكارت تحت.`
   615→  }
   616→  const text = cards.find((c) => c.kind === 'text') as Extract<ResultCardDTO, { kind: 'text' }> | undefined
   617→  if (text && text.body.trim()) return text.body.trim().slice(0, 220)
   618→  return 'دي النتايج من سجل الشغل، التفاصيل في الكارت تحت.'
   619→}

   621→/** Deterministic numbered multi-intent reply — answers EVERY sub-request, no LLM. */
   622→function deterministicMultiReply(parts: string[], overflow: number): string {
   623→  const sections = parts.map((p, i) => `${AR_ORDINALS[i] ?? i + 1}. ${p}`)
   624→  const overflowNote = overflow > 0 ? `\n\n${toArabicNum(overflow)} ${overflow === 1 ? 'طلب' : 'طلبات'} تانية بعد دول — قولي نكمل فيهم؟` : ''

   626→    ? '\n\nالمسودات تحت مستنية اعتمادك.'
   627→    : '\n\nتحب تفاصيل أكتر على أي جزء؟'
   628→  return sections.join('\n\n') + overflowNote + closer

   659→/**
   660→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   661→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   662→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   663→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   664→ * customer — the ل/من marker is what marks the ROLE.
   665→ */
   666→function findPartyMention<T extends { name: string }>(
   667→  text: string,
   668→  parties: T[],
   669→  marker: 'ل' | 'من' = 'ل',
   670→): T | null {
   671→  const norm = normalizeAr(text)
   672→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   673→  for (const p of parties) {
   674→    const words = normalizeAr(p.name)
   675→      .split(' ')
   676→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   677→      .filter((w) => w.length >= 2)
   678→    for (const w of words) {
   679→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   680→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   681→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   682→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   683→    }
   684→  }
   685→  return null
   686→}

   703→/** عطارة «البيع بالمبلغ»: «بـ20 جنيه فلفل أسود» → (amount, in EGP) or null.
   704→ * Runs on WESTERN-digit normalized text (normalizeAr already ate the ـ tatweel
   705→ * in «بـ20»). Requires a real money word (جنيه/ج) after the number so a bare
   706→ * quantity («بعت 20 فلفل») can never fire it. */
   707→function matchReverseSellAmount(normWestern: string): number | null {
   708→  const m = normWestern.match(/(?:^|\s)(?:بال|بمبلغ|ب)([0-9]+(?:[.,][0-9]+)?)\s*(?:جنيه|جنيهات|ج)(?=\s|$)/)
   709→  if (!m) return null
   710→  const amount = parseFloat(m[1].replace(',', '.'))
   711→  return Number.isFinite(amount) && amount > 0 ? amount : null
   712→}
   713→
   714→/** Remove the «بـN جنيه» clause so the money number can never masquerade as a
   715→ * weight in ANY quantity parser downstream (segment scan / parseQuantity). */
   716→function stripReverseSellAmount(normWestern: string): string {
   717→  return normWestern.replace(/(?:^|\s)(?:بال|بمبلغ|ب)[0-9]+(?:[.,][0-9]+)?\s*(?:جنيه|جنيهات|ج)(?=\s|$)/g, ' ')
   718→}
   719→
   720→/** Paid amount: LLM field → explicit «دفع 20» clause → آجل = 0 → default cash (total). */
   721→function resolvePaid(norm: string, f: Record<string, unknown>, total: number): number {
   722→  if (typeof f.paid === 'number' && f.paid > 0) return Math.min(Math.round(f.paid * 100), total)
   723→  const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   724→  if (paidWord?.[1]) {
   725→    const n = parseQuantity(paidWord[1], null).qty
   726→    if (n != null && n > 0) return Math.min(Math.round(n * 100), total)
   727→  }
   728→  if (/(?:بال)?اجل|علي الحساب|نسيون/.test(norm)) return 0
   729→  return total
   730→}

   732→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit, memCtx: MemoryContext): Promise<ResolvedWrite> {
   733→  const { business, user } = ctx
   734→  const type = (intent.opType ?? 'sale') as OpType
   735→  const f = intent.fields ?? {}
   736→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   737→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   738→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   739→
   740→  const text = ctx.text
   741→  const norm = normalizeAr(text)
   742→  const normW = toWesternDigits(norm)
   743→
   744→  switch (type) {
   745→    case 'sale':
   746→    case 'purchase': {
   747→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   748→      // «دفع ٢٠ بس» can never be misread as the sold quantity.
   749→      const textNoPaid = norm.replace(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?[٠-٩0-9][٠-٩0-9.,]*\s*(?:بس|فقط|تمام)?/g, ' ')
   750→      // ---- عطارة reverse-selling («بعت بـ20 جنيه فلفل») ----
   751→      // The money clause MUST be detected and stripped BEFORE any quantity
   752→      // parsing: a bare «20» in «بـ20 جنيه» is the AMOUNT, not a weight. The
   753→      // deterministic scanners and the LLM both tend to read it as qty=20 —
   754→      // the pattern match on «بـN جنيه» is the ground truth.
   755→      const reverseAmountEgp = business.type === 'attar' && type === 'sale' ? matchReverseSellAmount(normW) : null
   756→      const scanText = reverseAmountEgp != null ? stripReverseSellAmount(toWesternDigits(textNoPaid)) : textNoPaid
   757→      // ---- Multi-item scan (deterministic — works with or without the LLM) ----
   758→      // «سجّل بيع ٢ لبن و ١٠ بيض» → segments → per-item quantity. When TWO or
   759→      // more distinct catalog items are mentioned we draft ONE multi-line
   760→      // operation (a single basket the cashier approves once), like the real POS.
   761→      // Longest-alias-first + overlap rejection: «بانادول» alone can never fan
   762→      // out into «بانادول أكتيف» + «بانادول شرائط» — the more specific name wins.
   763→      const segments = scanText.split(/\s*(?:،|,|\+|؛|;)\s*|\s+و\s+|\s+و(?=[٠-٩0-9])/).filter(Boolean)
   764→      const byItem = new Map<string, { item: Item; qty: number | null; unit: string | null }>()
   765→      for (const seg of segments) {
   766→        const segNorm = normalizeAr(seg)
   767→        const hits: { item: Item; token: string; at: number }[] = []
   768→        for (const it of items) {
   769→          // canonical names + column aliases + LEXICON dialect aliases (boost)
   770→          const lexAl = memCtx.lexicon.filter((l) => l.entityRef === it.id).map((l) => l.alias)
   771→          const names = [it.name, ...(it.aliases ? it.aliases.split(',').map((a) => a.trim()) : []), ...lexAl]
   772→          let best: { token: string; at: number } | null = null
   773→          for (const n of names) {
   774→            const nn = normalizeAr(n)
   775→            if (nn.length >= 3) {
   776→              const at = segNorm.indexOf(nn)
   777→              if (at >= 0 && (!best || nn.length > best.token.length)) best = { token: nn, at }
   778→            }
   779→          }
   780→          if (best) hits.push({ item: it, ...best })
   781→        }
   782→        hits.sort((a, b) => b.token.length - a.token.length)
   783→        const taken: Array<[number, number]> = []
   784→        for (const h of hits) {
   785→          if (taken.some(([s, e]) => h.at < e && h.at + h.token.length > s)) continue
   786→          taken.push([h.at, h.at + h.token.length])
   787→          const q = parseQuantity(seg, null)
   788→          const prev = byItem.get(h.item.id)
   789→          byItem.set(h.item.id, {
   790→            item: h.item,
   791→            qty: q.qty != null && q.qty > 0 ? (prev?.qty ?? 0) + q.qty : prev?.qty ?? null,
   792→            unit: prev?.unit ?? unitToCode(q.unit) ?? null,
   793→          })
   794→        }
   795→      }
   796→      const multi = byItem.size >= 2
   797→
   798→      if (multi) {
   799→        // ---- Multi-line draft: one basket, real prices per line ----
   800→        const lines: { item: Item; qty: number; unitPrice: number; total: number }[] = []
   801→        const issues: string[] = []

   803→          const rawUnit = unit ?? item.unit // default = the item's own canonical unit
   804→          const baseQty = convertToBaseUnit(qty ?? 1, rawUnit, item.unit)
   805→          if (baseQty == null || !(baseQty > 0)) {
   806→            issues.push(`الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})`)
   807→            continue
   808→          }
   809→          const qtyN = Math.round(baseQty * 1000) / 1000
   810→          if (type === 'sale' && qtyN > item.stockQty) {
   811→            issues.push(`المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}`)
   812→            continue
   813→          }
   814→          const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   815→          lines.push({ item, qty: qtyN, unitPrice, total: Math.round(qtyN * unitPrice) })
   816→        }
   817→        if (lines.length === 0) {
   818→          return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: issues.join('؛ ') || 'الكمية قد إيه لكل صنف؟', issues }
   819→        }
   820→        const total = lines.reduce((s, l) => s + l.total, 0)
   821→        // Paid (clamped to the basket total; آجل = 0)
   822→        const paid = resolvePaid(norm, f, total)
   823→        // Party (lexicon first, then LLM field, then deterministic phrasing)
   824→        const lexCustomer = type === 'sale' ? lexiconResolve(norm, memCtx.lexicon, customers)?.entity : null
   825→        if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
   826→        const customer = lexCustomer ?? (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   827→        const supplier = (f.supplierName ? matchEntity(String(f.supplierName), suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   828→
   829→        const payload: Record<string, unknown> = {
   830→          lines: lines.map((l) => ({ itemId: l.item.id, name: l.item.name, qty: l.qty, unit: l.item.unit, unitPrice: l.unitPrice, total: l.total })),
   831→          total,
   832→          paid,
   833→          ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   834→          ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   835→        }
   836→        const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   837→        const risk = computeRisk(type, total)
   838→        const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${lines.map((l) => `${fmtQty(l.qty, arUnit(l.item.unit))} ${l.item.name}`).join(' + ')} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   839→        // BUSINESS_DNA credit-limit enforcement (آجل on a capped party)
   840→        let dangerNote: string | undefined
   841→        let creditField: { label: string; value: string; hint?: string } | undefined
   842→        if (type === 'sale' && customer && total - paid > 0 && memCtx.rules.length > 0) {
   843→          const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
   844→          if (check) {
   845→            dangerNote = check.warning

   847→            memCtx.referencedIds.push(check.rule.id)
   848→          }
   849→        }
   850→        const card = fieldsToCard(ref, type, risk, summary, [
   851→          ...lines.map((l) => ({ label: l.item.name, value: `${fmtQty(l.qty, arUnit(l.item.unit))} × ${egp(l.unitPrice)} = ${egp(l.total)}` })),
   852→          { label: 'الإجمالي', value: egp(total) },
   853→          { label: 'مدفوع', value: egp(paid) },
   854→          ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   855→          ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   856→          ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   857→          ...(creditField ? [creditField] : []),

   859→        ], [
   860→          ...lines.map((l) => ({ label: `مخزون ${l.item.name}`, value: `${fmtQty(l.item.stockQty, arUnit(l.item.unit))} ← ${fmtQty(l.item.stockQty + (type === 'sale' ? -l.qty : l.qty), arUnit(l.item.unit))}`, tone: 'neutral' as const })),
   861→          { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' as const : 'bad' as const },
   862→        ], dangerNote ?? RISK_META[risk].hint)
   863→        return { ok: true, opType: type, payload, summary, card, issues: issues.length > 0 ? issues : undefined }
   864→      }
   865→
   866→      // ---- Single-item draft (original path) ----
   867→      // Item: lexicon (dialect alias) → LLM field → text mention → fuzzy whole-text match
   868→      const rawItemName = String(f.itemName ?? '')

   872→      if (lexHit) {
   873→        memCtx.referencedIds.push(lexHit.entry.id)

   878→      }

   881→      }
   882→

   885→        // scanText already has the money clause stripped, so ANY number still

   914→              itemId: item.id,
   915→              itemName: item.name,

  1017→        ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
  1018→        ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
  1019→        ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
  1020→        ...(creditField ? [creditField] : []),
  1021→      ], [

  1023→        { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' : 'bad' },
  1024→      ], dangerNote ?? RISK_META[risk].hint)
  1025→      return { ok: true, opType: type, payload, summary, card }
  1026→    }
  1027→
  1028→    case 'expense': {
  1029→      const amountRaw = f.amount as number | undefined
  1030→      let amount = amountRaw && amountRaw > 0 ? Math.round(amountRaw * 100) : null
  1031→      if (amount == null) {
  1032→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1033→        if (m) {
  1034→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d))).replace(',', '.')
  1035→          const parsed = parseFloat(western)
  1036→          if (!Number.isNaN(parsed)) amount = Math.round(parsed * 100)
  1037→        }
  1038→      }
  1039→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: 'المبلغ قد إيه؟' }
  1040→      const label = String(f.label ?? '') || (norm.includes('كهربا') ? 'كهربا' : norm.includes('ايجار') ? 'إيجار' : norm.includes('تليفون') ? 'تليفون' : 'مصروف')
  1041→      const payload = { label, total: amount }
  1042→      const risk = computeRisk(type, amount)
  1043→      const summary = `مصروف ${label} بـ${egp(amount)}`

  1045→        { label: 'البند', value: label },

  1054→      const lexCustomer = lexiconResolve(norm, memCtx.lexicon, customersAll)?.entity ?? null
  1055→      if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
  1056→      const customer = lexCustomer ?? matchEntity(customerName || text, customersAll)?.entity ?? null
  1057→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العميل', failMessage: 'تحصيل من مين بالظبط؟' }
  1058→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null

  1060→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)

  1074→        { label: 'العميل', value: customer.name },
  1075→        { label: 'المبلغ', value: egp(amount) },
  1076→        { label: 'المديونية الحالية', value: egp(customer.balance) },
  1077→        { label: 'بعد التحصيل', value: egp(customer.balance - amount) },
  1078→      ], [{ label: 'الخزنة', value: egp(amount), tone: 'good' }], RISK_META[risk].hint)

  1081→

  1085→      if (!supplier) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المورد', failMessage: 'هسدد لمين؟' }
  1086→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
  1087→      if (amount == null) {
  1088→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1089→        if (m) {
  1090→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
  1091→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)

  1095→      if (amount > supplier.balance) {

  1099→      const risk = computeRisk(type, amount)
  1100→      const summary = `سداد ${egp(amount)} لـ${supplier.name}`

  1102→        { label: 'المورد', value: supplier.name },
  1103→        { label: 'المبلغ', value: egp(amount) },
  1104→        { label: 'المستفق الحالي', value: egp(supplier.balance) },

  1106→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
  1107→      return { ok: true, opType: type, payload, summary, card }
  1108→    }
  1109→
  1110→    case 'membership': {
  1111→      const isRenewal = norm.includes('تجديد') || norm.includes('جدد')
  1112→      const explicitName = String(f.customerName ?? '').trim()
  1113→      const customer = explicitName ? matchEntity(explicitName, customers)?.entity ?? null : matchEntity(text, customers)?.entity ?? null
  1114→      if (!customer && isRenewal) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'تجديد لمين؟' }
  1115→      // A NEW subscription with NO name at all (rule path, question-like text)
  1116→      // must never draft «عضو جديد» — ask for the name first.
  1117→      if (!customer && !explicitName) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'اشتراك جديد لمين؟ اكتبلي اسم العضو' }
  1118→      const plan = String(f.plan ?? 'شهرية')
  1119→      const months = Math.max(1, Math.min(12, Math.round(Number(f.months ?? 1))))
  1120→      // plan price from items (gym plans are items) or fallback 35000
  1121→      const planItem = items.find((i) => normalizeAr(i.name).includes(normalizeAr(plan)))
  1122→      const total = Math.round((planItem?.sellPrice ?? 35000) * months)
  1123→      let paid = total
  1124→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
  1125→      paid = Math.min(paid, total)
  1126→      const name = customer?.name ?? explicitName
  1127→      const payload = {
  1128→        ...(customer ? { customerId: customer.id } : { customerName: name }),
  1129→        plan,
  1130→        months,
  1131→        total,
  1132→        paid,
  1133→      }
  1134→      const risk = computeRisk(type, total)
  1135→      const summary = `${isRenewal ? 'تجديد' : 'اشتراك جديد'} ${plan} لـ${name} — ${egp(total)}`
  1136→      const card = fieldsToCard('', type, risk, summary, [
  1137→        { label: 'العضو', value: name },
  1138→        { label: 'الخطة', value: plan },
  1139→        { label: 'المدة', value: fmtQty(months, 'شهر') },
  1140→        { label: 'الإجمالي', value: egp(total) },
  1141→        { label: 'مدفوع', value: egp(paid) },
  1142→      ], [
  1143→        { label: 'الخزنة', value: egp(paid), tone: 'good' },
  1144→        ...(total - paid > 0 ? [{ label: 'باقي على العضو', value: egp(total - paid), tone: 'bad' as const }] : []),
  1145→      ], RISK_META[risk].hint)
  1146→      return { ok: true, opType: type, payload, summary, card }
  1147→    }
  1148→
  1149→    case 'attendance': {
  1150→      const customerName = String(f.customerName ?? '')
  1151→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
  1152→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'حضور مين؟' }
  1153→      const payload = { customerId: customer.id, customerName: customer.name }
  1154→      const risk = computeRisk(type)
  1155→      const summary = `حضور ${customer.name}`
  1156→      const card = fieldsToCard('', type, risk, summary, [
  1157→        { label: 'العضو', value: customer.name },
  1158→        { label: 'اليوم', value: 'النهاردة' },
  1159→        ...(customer.expiresAt ? [{ label: 'الاشتراك لحد', value: arDate(customer.expiresAt) }] : []),
  1160→      ], [], RISK_META[risk].hint)
  1161→      return { ok: true, opType: type, payload, summary, card }
  1162→    }
  1163→
  1164→    case 'stock_adjustment': {
  1165→      const rawItemName = String(f.itemName ?? '')
  1166→      const item = matchEntity(rawItemName || text, items)?.entity ?? null
  1167→      if (!item) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: 'تسوية مخزون لأنهي صنف؟' }
  1168→      let delta = (f.delta as number | undefined) ?? null

  1170→      if (delta == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية الجديدة/الفرق', failMessage: `المخزون الحالي ${fmtQty(item.stockQty, arUnit(item.unit))} — عايزه يبقى قد إيه؟` }

  1173→      const risk = computeRisk(type)
  1174→      const summary = `تسوية مخزون ${item.name}: ${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`

  1177→        { label: 'الحالي', value: fmtQty(item.stockQty, arUnit(item.unit)) },

  1179→        { label: 'السبب', value: reason },
  1180→      ], [{ label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`, tone: delta >= 0 ? 'good' : 'bad' }], 'تسوية المخزون عملية حساسة (R3) — محتاجة سبب واعتماد')

  1182→    }
  1183→
  1184→    case 'reversal': {
  1185→      const ref = String(f.ref ?? '') || (norm.match(/WKL-\d+/)?.[0] ?? '')
  1186→      if (!ref) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'رقم العملية', failMessage: 'عايز تعكس أنهي عملية؟ اكتب رقمها (زي WKL-1006)' }

  1188→      if (!original) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `ملقيتش عملية رقم ${ref}` }

  1191→      const payload = { originalRef: ref, reason }
  1192→      const risk = computeRisk(type)
  1193→      const summary = `عكس العملية ${ref} (${original.summary})`
  1194→      const card = fieldsToCard('', type, risk, summary, [
  1195→        { label: 'العملية الأصلية', value: `${ref} — ${TYPE_AR[original.type] ?? original.type}` },

  1197→        { label: 'السبب', value: reason },
  1198→      ], [{ label: 'الأثر', value: 'هيتلغى بالكامل بعملية عكسية موثقة', tone: 'neutral' }], 'عكس عملية مالية حساس (R3) — التاريخ مش بيتمسح، بيتصحح')
  1199→      return { ok: true, opType: type, payload, summary, card }
  1200→    }
  1201→
  1202→    default:
  1203→      return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: 'نوع العملية مش مدعوم' }
  1204→  }

  1210→export interface TurnResult {
  1211→  replyText: string

  1213→  cards?: ResultCardDTO[] | null

  1219→
  1220→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
  1221→  const { business, user, text } = ctx
  1222→  const v = getVertical(business.type)
  1223→  const model = await resolveModel(ctx.modelKey)
  1224→  const modelLabel = modelLabelFor(model)
  1225→
  1226→  // Load context once (memories raised to 8; pending ops count for the persona)
  1227→  const [items, customers, suppliers, memories, pendingCount] = await Promise.all([
  1228→    db.item.findMany({ where: { businessId: business.id } }),
  1229→    db.customer.findMany({ where: { businessId: business.id } }),
  1230→    db.supplier.findMany({ where: { businessId: business.id } }),
  1231→    db.agentMemory.findMany({ where: { businessId: business.id, userId: user.id, active: true }, orderBy: { createdAt: 'desc' }, take: 12 }),
  1232→    db.operation.count({ where: { businessId: business.id, status: 'AWAITING_APPROVAL' } }),
  1233→  ])
  1234→  const rules = memories.filter((m) => m.kind === 'rule')
  1235→  const memCtx: MemoryContext = {
  1236→    lexicon: parseLexicon(memories),
  1237→    rules,
  1238→    referencedIds: [],
  1239→  }
  1240→  // Prompt memories: preferences + facts (rules are injected in their own block)
  1241→  const promptMemories = memories.filter((m) => m.kind === 'preference' || m.kind === 'fact')
  1242→
  1243→  // Thinking steps — with a «فتكرت من ذاكرتي» recall step injected right after
  1244→  // «بفهم طلبك» whenever the agent actually remembers something about this user.
  1245→  const memStep: ThinkingStep[] =
  1246→    memories.length > 0
  1247→      ? [
  1248→          {
  1249→            label: 'فتكرت من ذاكرتي',
  1250→            detail: memories[0].text.slice(0, 60) + (memories[0].text.length > 60 ? '…' : ''),
  1251→            state: 'done',
  1252→            icon: 'verify',
  1253→          },
  1254→        ]
  1255→      : []
  1256→  const steps: ThinkingStep[] = [
  1257→    { label: 'بفهم طلبك', detail: 'بحلل الجملة وأستخرج القصد والحقول', state: 'active', icon: 'understand' },
  1258→    ...memStep,
  1259→    { label: 'بدوّر في سجلك', detail: 'بدوّر على الأصناف والأسماء الحقيقية', state: 'pending', icon: 'search' },
  1260→    { label: 'بتحقق من الأرقام', detail: 'بحسب من الأسعار الفعلية مش من دماغي', state: 'pending', icon: 'verify' },
  1261→    { label: 'بجهّز الرد', detail: '', state: 'pending', icon: 'plan' },
  1262→  ]
  1263→  const IDX = { understand: 0, search: 1 + memStep.length, verify: 2 + memStep.length, reply: 3 + memStep.length }
  1264→  const pushThinking = () => emit({ event: 'thinking', steps: steps.map((s) => ({ ...s })) })
  1265→  const done = (i: number, detail?: string) => {
  1266→    if (steps[i]) {
  1267→      steps[i].state = 'done'
  1268→      if (detail) steps[i].detail = detail
  1269→    }
  1270→    pushThinking()
  1271→  }
  1272→  pushThinking()
  1273→
  1274→  const norm = normalizeAr(text)
  1275→
  1276→  // ---- PASS 0a: Identity question → instant, deterministic, zero LLM ----
  1277→  if (isIdentityQuestion(norm)) {
  1278→    const lowStockCount = items.filter((i) => i.stockQty <= i.lowStockThreshold).length
  1279→    const receivablesTotal = customers.reduce((s, c) => s + Math.max(0, c.balance), 0)
  1280→    done(IDX.understand, 'سؤال عن هويتي — بجاوب من سجلك من غير موديل')
  1281→    done(IDX.search, 'جبت بياناتي من سجلك الحقيقي')
  1282→    done(IDX.verify, 'عدّيت النواقص والعمليات المستنية من قاعدة البيانات')
  1283→    emit({ event: 'step', label: 'بتعرّف بنفسي', state: 'done' })
  1284→    done(IDX.reply)
  1285→    const identityMemory = memories.find((m) => m.kind === 'preference' || m.kind === 'rule' || m.kind === 'fact') ?? null
  1286→    if (identityMemory) memCtx.referencedIds.push(identityMemory.id)
  1287→    const replyText = buildIdentityReply({
  1288→      business,
  1289→      user,
  1290→      verticalLabel: v.label,
  1291→      memoryText: identityMemory?.text ?? null,
  1292→      lowStockCount,
  1293→      pendingCount,
  1294→      receivablesTotal,
  1295→    })
  1296→    const actions: ChatActionDTO[] = [
  1297→      { id: 'act-today', label: 'لوحة اليوم', view: 'today', icon: 'today', tone: 'primary' },
  1298→      ...(lowStockCount > 0
  1299→        ? [{ id: 'act-inventory', label: 'شوف المخزون', view: 'inventory' as const, icon: 'boxes' as const, tone: 'outline' as const }]
  1300→        : []),
  1301→    ]
  1302→    await streamDeterministic(replyText, emit, 1200)
  1303→    emit({ event: 'actions', actions })
  1304→    await trackUsage(business.id, user.id, 'chat', 'chat')
  1305→    await bumpMemoryUsage(memCtx.referencedIds)
  1306→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions }
  1307→  }
  1308→
  1309→  // ---- PASS 0b: Memory save (preference OR BUSINESS_DNA rule) → instant, persisted, zero LLM ----
  1310→  const saveRequest = extractSaveRequest(text, norm) ?? extractRuleRequest(text, norm)
  1311→  if (saveRequest) {
  1312→    // Dedupe: re-saving the exact same memory must NOT clutter the memory
  1313→    // list — the owner repeating himself deserves a calm «already saved».
  1314→    const isDuplicate = memories.some((m) => normalizeAr(m.text) === normalizeAr(saveRequest.text))
  1315→    if (!isDuplicate) {
  1316→      await db.agentMemory.create({
  1317→        data: {
  1318→          businessId: business.id,
  1319→          userId: user.id,
  1320→          kind: saveRequest.kind,
  1321→          category: saveRequest.category,
  1322→          text: saveRequest.text,
  1323→          confidence: saveRequest.confidence,
  1324→        },
  1325→      })
  1326→    }
  1327→    const isRule = saveRequest.kind === 'rule'
  1328→    done(IDX.understand, isRule ? 'قانون من صاحب الشغل — مش محتاج موديل' : 'طلب حفظ في الذاكرة — مش محتاج موديل')
  1329→    done(IDX.search, 'بتخزنها في ذاكرتي عنك')
  1330→    done(IDX.verify, isDuplicate ? 'موجودة أصلًا — من غير تكرار' : 'اتحفظت فعليًا في قاعدة البيانات')
  1331→    emit({ event: 'step', label: 'بحفظ في ذاكرتي', state: 'done' })
  1332→    done(IDX.reply)
  1333→    const replyText = isDuplicate

  1636→    // back to the deterministic numbered answer — live == persisted == كامل.
  1637→    let collected = ''
  1638→    try {
  1639→      for await (const chunk of streamChat(model, messages)) {
  1640→        if (chunk.type !== 'content') continue
  1641→        collected += chunk.text
  1642→      }
  1643→    } catch (err) {
  1644→      console.error('[harness] multi-intent reply stream failed → deterministic numbered reply:', err)
  1645→    }
  1646→    const acceptable = collected.trim().length > 0 && replyAnswersAllParts(collected, Math.min(subs.length, 4))
  1647→    replyText = acceptable ? collected.trim() : deterministicMultiReply(multiReplyParts, multiOverflow)
  1648→    await streamDeterministic(replyText, emit, REPLY_PACING_BUDGET_MS)
  1649→  } else {
  1650→    try {
  1651→      for await (const chunk of streamChat(model, messages)) {
  1652→        if (chunk.type !== 'content') continue
  1653→        replyText += chunk.text
  1654→        // Re-pace into tiny letter-by-letter deltas — adaptively: total pacing
  1655→        // is capped by the budget; a slow provider gets the 3ms floor instead.
  1656→        // [\s\S] keeps the model's own line breaks alive in the stream
  1657→        // (plain `.` drops \n; ES2017 target forbids the dotAll `s` flag).
  1658→        const pieces = chunk.text.match(/[\s\S]{1,2}/g) ?? []
  1659→        for (let i = 0; i < pieces.length; i++) {
  1660→          emit({ event: 'delta', text: pieces[i] })
  1661→          pacer.emitted++
  1662→          await sleep(adaptivePieceDelay(pacer, pieces.length - i - 1))
  1663→        }
  1664→      }
  1665→    } catch (err) {
  1666→      console.error('[harness] reply stream failed:', err)
  1667→      if (!replyText) {
  1668→        const fallback = advisoryMode
  1669→          ? advisoryFallback(v)
  1670→          : card
  1671→            ? `جهزتلك مسودة «${card.title}» — ${card.summary}. راجعها واعتمدها لو تمام.${card.dangerNote ? `\n${card.dangerNote}` : ''}${draftIssues ? ` بس خد بالك: ${draftIssues.join('؛ ')}.` : ''}`
  1672→            : cards
  1673→              ? deterministicReadReply(cards)
  1674→              : questionMode
  1675→                ? (situation.match(/— (.+)$/)?.[1] ?? 'محتاج تفاصيل أكتر لو سمحت.')
  1676→                : permissionDenied
  1677→                  // The cashier must NEVER read «تمام» on a denied operation —
  1678→                  // that reads like «it's done». Say the permission truth instead.
  1679→                  ? permissionMsg
  1680→                  : chatFallbackWithChips(v)
  1681→        await streamDeterministic(fallback, emit, 900)
  1682→        replyText = fallback
  1683→      }
  1684→    }
  1685→  }
  1686→
  1687→  if (!replyText.trim()) {
  1688→    const fallback = advisoryMode
  1689→      ? advisoryFallback(v)
  1690→      : multiReplyParts
  1691→        ? deterministicMultiReply(multiReplyParts, multiOverflow)
  1692→        : card ? `جهزت المسودة — ${card.summary}` : permissionDenied ? permissionMsg : chatFallbackWithChips(v)
  1693→    await streamDeterministic(fallback, emit, 900)
  1694→    replyText = fallback
  1695→  }
  1696→
  1697→  // ---- PASS 5: Deep-link action buttons (after deltas, before done) ----
  1698→  if (actions && actions.length > 0) {
  1699→    emit({ event: 'actions', actions })
  1700→  }
  1701→
  1702→  await bumpMemoryUsage(memCtx.referencedIds)
  1703→  return { replyText, card, cards, thinking: steps, modelLabel, opId, actions: actions && actions.length > 0 ? actions : null }
  1704→}
