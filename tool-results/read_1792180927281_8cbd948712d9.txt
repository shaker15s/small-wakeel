   114→// ---------------------------------------------------------------------------
   115→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
   116→// and a hard floor so a slow provider never gets delay stacked on top of it.
   117→// ---------------------------------------------------------------------------
   118→const REPLY_PACING_BUDGET_MS = 1500
   119→const PIECE_DELAY_MIN_MS = 3
   120→const PIECE_DELAY_MAX_MS = 26
   121→
   122→interface PacerState {
   123→  t0: number | null
   124→  emitted: number
   125→}
   126→
   127→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
   128→ * The moment elapsed time eats the whole budget (the provider itself is slow),
   129→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
   130→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
   131→  const now = Date.now()
   132→  if (p.t0 == null) p.t0 = now
   133→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
   134→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
   135→  // Estimate the pieces still coming: at least the rest of this chunk, and at
   136→  // least as many as we've already emitted (replies rarely stop abruptly).
   137→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
   138→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
   139→}

   141→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
   142→ * (identity/memory replies: whole text known upfront, exact remaining count). */
   143→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
   144→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
   145→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
   146→  // project targets ES2017, where the `s` flag is a tsc error).
   147→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
   148→  const t0 = Date.now()
   149→  for (let i = 0; i < pieces.length; i++) {
   150→    emit({ event: 'delta', text: pieces[i] })
   151→    const remainingPieces = pieces.length - i - 1
   152→    if (remainingPieces <= 0) break
   153→    const remainingBudget = budgetMs - (Date.now() - t0)
   154→    if (remainingBudget <= 0) {
   155→      await sleep(PIECE_DELAY_MIN_MS)
   156→      continue
   157→    }
   158→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
   159→  }
   160→}

   345→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   346→  const s = normalizeAr(text)
   347→  const isGym = businessType === 'gym'
   348→
   349→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   350→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   351→  }
   352→  // Opinion/decision-seeking → ADVISORY free-chat (the real-employee mode).
   353→  // Must beat the price-query rule below so «رأيك أزود سعر الكمون؟» is a
   354→  // conversation about the decision, never a bare catalog scan.
   355→  if (/(رايك|رأيك|ايه رايك|تنصحني|نصيحه|اعمل ايه لو|تشوف ايه|رايك ايه|استشاره)/.test(s) && !/(بعت|بيع|سجل بيع|تحصيل|اشتري)/.test(s)) {
   356→    return { intent: 'chat', fields: {}, replyHint: 'استشارة شغل' }
   357→  }
   358→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   359→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   360→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   361→  }
   362→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   363→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   364→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   365→  if (/(مديون|اللي ليا|تحصيلات|مين عليها|عليها فلوس|عليهم فلوس|العملاء)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   366→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   367→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   368→  // Price query («ايه سعر عود كمبودي؟» / «بكام المسك؟») → catalog search (R0 read)
   369→  if (/(سعر|بكام)/.test(s) && !/(بعت|بيع|بع |خرج|سلف|دفع|سجل)/.test(s)) {
   370→    const q = s
   371→      .replace(/^.*?(?:سعر|بكام)\s*/, '')
   372→      .replace(/[؟?].*$/, '')
   373→      .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   374→      .replace(/^\s*(?:في|من)\s*/, '')
   375→      .trim()
   376→    if (q.length >= 2) return { intent: 'read', tool: 'search_catalog', fields: { query: q.slice(0, 40) } }
   377→  }
   378→  // Attendance (gym): noun «حضور» + Egyptian verb forms («كريم حضر»,
   379→  // «لعب حصة PT», «خلص تدريب») — a member WORKED OUT today.
   380→  if (/(حضور|(^| )حضر( |$)|لعب حصه?|خلص (تدريب|حصه?)|اتمرن|اتدرب)/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   381→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   382→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   383→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   384→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   385→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   386→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   387→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   388→    if (name && (isQuestion || /عضويه/.test(s))) {
   389→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   390→    }
   391→  }
   392→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   393→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   394→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   395→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   396→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   397→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   398→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   399→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   400→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   401→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   402→  // fire a collection (caught by the expiring rule above instead).
   403→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   404→    return { intent: 'write', opType: 'collection', fields: {} }
   405→  }
   406→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   407→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   408→  return { intent: 'chat', fields: {} }
   409→}

   439→/** Parse the LLM JSON contract: single intent OR the "intents" array (multi). */
   440→function parseIntentJsonMulti(raw: string): { intents: RawIntentJson[]; single: ExtractedIntent | null } {
   441→  const cleaned = raw.replace(/```json|```/g, '').trim()
   442→  const start = cleaned.indexOf('{')
   443→  const end = cleaned.lastIndexOf('}')
   444→  if (start < 0 || end <= start) return { intents: [], single: null }
   445→  try {
   446→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent> & { intents?: RawIntentJson[] }
   447→    if (Array.isArray(json.intents)) {
   448→      const valid = json.intents.filter((r) => r && typeof r.intent === 'string')
   449→      if (valid.length >= 2) return { intents: valid, single: null }
   450→      if (valid.length === 1) return { intents: [], single: rawToSingle(valid[0]) }
   451→      return { intents: [], single: null }
   452→    }
   453→    if (!json.intent) return { intents: [], single: null }
   454→    return { intents: [], single: rawToSingle(json as RawIntentJson) }
   455→  } catch {
   456→    return { intents: [], single: null }
   457→  }
   458→}

   460→function rawToSingle(json: RawIntentJson): ExtractedIntent | null {
   461→  const intent = json.intent === 'read' || json.intent === 'write' || json.intent === 'chat' ? json.intent : null
   462→  if (!intent) return null
   463→  return {
   464→    intent,
   465→    tool: json.tool,
   466→    opType: json.opType as OpType | undefined,
   467→    fields: (json.fields as Record<string, unknown>) ?? {},
   468→    question: json.question,
   469→    replyHint: json.replyHint,
   470→  }
   471→}

   473→/** Read-request trigger keywords → tool (used by the deterministic splitter). */
   474→const READ_SPLITTERS: [RegExp, string][] = [
   475→  [/ملخص|حال اليوم|كويس اليوم|عملت ايه النهارده|اقفل الورديه/, 'get_daily_summary'],
   476→  [/وضع الخزنه|رصيد الخزنه|الخزنه|الكاش/, 'get_cash_position'],
   477→  [/النواقص|المخزون الناقص|الناقص|المخزون|استوك/, 'list_low_stock'],
   478→  [/صلاحيه?|منتهي|قربت تخلص|تنتهي/, 'list_expiring'],
   479→  [/مين عليها|عليها فلوس|عليهم فلوس|مديون|تحصيلات|اللي ليا/, 'list_receivables'],
   480→  [/اللي علينا|للموردين|الموردين|مستحقات/, 'list_payables'],
   481→  [/آخر الحركات|اخر الحركات|سجل العمليات|آخر العمليات|اخر العمليات|العمليات المستنيه/, 'get_activity_history'],
   482→  [/سعر|بكام/, 'search_catalog'],
   483→]
   484→
   485→/** Sequence verbs that START a new request mid-message (never paid-clauses). */
   486→const WRITE_CUT_RE = /بعت|اشتريت|اشتري|سجل|حصلت|قبضت|استلمت|سدد|صرفت|مصروف|تسويه|عكسها|اعكس/
   487→
   488→/**
   489→ * Deterministic compound splitter — «ملخص اليوم ومين عليها فلوس وايه الناقص
   490→ * في المخزون» → 3 read sub-intents. Cuts at read-trigger keywords, sequence
   491→ * verbs (بعت/سجل/…) and connectors (و بعدين/كمان) — never inside a multi-item
   492→ * basket («بعت ٢ لبن و ١٠ بيض» stays ONE write). Zero LLM.
   493→ */
   494→function splitSubIntents(text: string, businessType: string): SubIntent[] {
   495→  const norm = toWesternDigits(normalizeAr(text))
   496→  if (norm.length < 8) return []
   497→  type Hit = { at: number; len: number; tool: string | null }
   498→  const hits: Hit[] = []
   499→  for (const [re, tool] of READ_SPLITTERS) {
   500→    const g = new RegExp(re.source, 'g')
   501→    let m: RegExpExecArray | null
   502→    while ((m = g.exec(norm)) != null) {
   503→      hits.push({ at: m.index, len: m[0].length, tool })
   504→      if (m.index === g.lastIndex) g.lastIndex++
   505→    }
   506→  }
   507→  // sequence verbs — only when a و/، actually precedes them (mid-message cut)
   508→  {
   509→    const g = new RegExp(WRITE_CUT_RE.source, 'g')
   510→    let m: RegExpExecArray | null
   511→    while ((m = g.exec(norm)) != null) {
   512→      const at = m.index
   513→      const prev = at > 0 ? norm[at - 1] : ''
   514→      const prev2 = at > 1 ? norm[at - 2] : ''
   515→      const prev3 = at > 2 ? norm[at - 3] : ''
   516→      const isBoundary =
   517→        prev === 'و' || prev === '،' || prev === ',' || prev === '؛' || (prev === ' ' && (prev2 === 'و' || prev2 === '،')) || (prev === ' ' && prev2 === ' ' && prev3 === 'و')
   518→      if (isBoundary) hits.push({ at, len: m[0].length, tool: null })
   519→      if (m.index === g.lastIndex) g.lastIndex++
   520→    }
   521→  }
   522→  // connectors «و بعدين» — cut at the connector itself
   523→  {
   524→    const g = /(بعدين|بعد كده|كمان)/g
   525→    let m: RegExpExecArray | null
   526→    while ((m = g.exec(norm)) != null) {
   527→      hits.push({ at: m.index, len: m[0].length, tool: null })
   528→      if (m.index === g.lastIndex) g.lastIndex++
   529→    }
   530→  }
   531→  hits.sort((a, b) => a.at - b.at)
   532→  if (hits.length === 0) return []
   533→
   534→  // dedupe overlaps + same-tool continuations, build cut points
   535→  const cuts: { at: number; tool: string | null }[] = []
   536→  let coveredEnd = -1
   537→  let currentTool: string | null = null
   538→  for (const h of hits) {
   539→    if (h.at < coveredEnd) continue // overlapping keyword (الخزنه inside وضع الخزنه)
   540→    if (h.tool != null && h.tool === currentTool) {
   541→      // same tool still talking — «سعر العنبر بكام؟» is ONE search, and
   542→      // «سعر العود وبكام المسك» is one question about two items (the catalog
   543→      // matcher is bidirectional-includes, so both still show in one card)
   544→      coveredEnd = Math.max(coveredEnd, h.at + h.len)
   545→      continue
   546→    }
   547→    // cut position: include a glued و in the PREVIOUS segment
   548→    const cutAt = h.at > 0 && norm[h.at - 1] === 'و' ? h.at - 1 : h.at
   549→    cuts.push({ at: cutAt, tool: h.tool })
   550→    coveredEnd = h.at + h.len
   551→    currentTool = h.tool
   552→  }
   553→  if (cuts.length === 0) return []
   554→
   555→  // build segments
   556→  const segments: { text: string; tool: string | null }[] = []
   557→  const seg0 = norm.slice(0, cuts[0].at).replace(/[\s،,؛]+$/, '').trim()
   558→  if (seg0.length > 0) segments.push({ text: seg0, tool: null })
   559→  for (let i = 0; i < cuts.length; i++) {
   560→    const end = i + 1 < cuts.length ? cuts[i + 1].at : norm.length
   561→    const seg = norm.slice(cuts[i].at, end).replace(/^[\s،,؛و]+/, '').replace(/[\s،,؛]+$/, '').trim()
   562→    if (seg.length > 0) segments.push({ text: seg, tool: cuts[i].tool })
   563→  }
   564→
   565→  // map segments → sub-intents
   566→  const subs: SubIntent[] = []
   567→  for (let i = 0; i < segments.length; i++) {
   568→    const seg = segments[i]
   569→    // politeness prefixes («قوللي», «اط») — skip chat-y tiny segments
   570→    if (seg.tool == null) {
   571→      if (i === 0 && seg.text.length <= 6) continue
   572→      const mapped = ruleBasedIntent(seg.text, businessType)
   573→      if (mapped.intent === 'read' && mapped.tool) subs.push({ ...mapped, intent: 'read', tool: mapped.tool, sourceText: seg.text })
   574→      else if (mapped.intent === 'write' && mapped.opType) subs.push({ ...mapped, intent: 'write', sourceText: seg.text })
   575→      continue
   576→    }
   577→    // tool-owned segment
   578→    if (seg.tool === 'search_catalog') {
   579→      const q = seg.text
   580→        .replace(/^(?:ايه|ايه هو|هو)?\s*(?:سعر|بكام)\s*/, '')
   581→        .replace(/[؟?].*$/, '')
   582→        .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   583→        .replace(/^\s*(?:في|من)\s*/, '')
   584→        .trim()
   585→      subs.push({ intent: 'read', tool: 'search_catalog', fields: q.length >= 2 ? { query: q.slice(0, 40) } : {}, sourceText: seg.text })
   586→    } else {
   587→      subs.push({ intent: 'read', tool: seg.tool, fields: {}, sourceText: seg.text })
   588→    }
   589→  }
   590→
   591→  // merge CONSECUTIVE same-opType writes into one basket (multi-item scan)
   592→  const merged: SubIntent[] = []
   593→  for (const s of subs) {
   594→    const last = merged[merged.length - 1]
   595→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   596→      last.sourceText = `${last.sourceText} و ${s.sourceText}`
   597→      last.fields = { ...last.fields, ...s.fields }
   598→      continue
   599→    }
   600→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   601→      continue // duplicate read
   602→    }
   603→    merged.push({ ...s })
   604→  }
   605→  return merged
   606→}

   608→/** Normalize LLM multi-intents: validate, merge consecutive same-opType writes. */
   609→function normalizeMultiIntents(raws: RawIntentJson[], fullText: string): SubIntent[] {
   610→  const subs: SubIntent[] = []
   611→  for (const r of raws) {
   612→    if (r.intent === 'read' && r.tool && READ_TOOLS.has(r.tool)) {
   613→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   614→      subs.push({ intent: 'read', tool: r.tool as ReadTool, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   615→    } else if (r.intent === 'write' && r.opType && VALID_OP_TYPES.has(r.opType)) {
   616→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   617→      subs.push({ intent: 'write', opType: r.opType as OpType, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   618→    }
   619→  }
   620→  const merged: SubIntent[] = []
   621→  for (const s of subs) {
   622→    const last = merged[merged.length - 1]
   623→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   624→      // consecutive same-type writes = ONE basket; full-text scan finds the lines
   625→      last.sourceText = fullText
   626→      last.fields = { ...last.fields, ...s.fields }
   627→      continue
   628→    }
   629→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   630→      continue // duplicate read
   631→    }
   632→    merged.push(s)
   633→  }
   634→  return merged
   635→}

   637→// ---------------------------------------------------------------------------
   638→// Proactive hints — when a read tool surfaced real counts worth attention,
   639→// the reply closes with one short heads-up (numbers only from real data).
   640→// ---------------------------------------------------------------------------
   641→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   642→  for (const c of cards) {
   643→    if (c.kind !== 'table' || !c.footer) continue
   644→    for (const f of c.footer) {
   645→      const m = String(f.value).match(/[\d٠-٩]+/)
   646→      if (m) {
   647→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   648→        const n = Number(west)
   649→        if (Number.isFinite(n)) return n
   650→      }
   651→    }
   652→  }
   653→  return null
   654→}

   656→function proactiveHintFor(tool: string, cards: ResultCardDTO[]): string {
   657→  if (tool === 'list_low_stock') {
   658→    const n = tableFooterCount(cards)
   659→    if (n != null && n > 0) return `قفل جملة قصيرة تنبّه على ${n} صنف ناقص من غير ما تكرر الأرقام كلها.`
   660→  }
   661→  if (tool === 'list_expiring') {
   662→    const n = tableFooterCount(cards)
   663→    if (n != null && n > 0) return `قفل جملة قصيرة تنبّه على ${n} صنف صلاحيته قربت تخلص من غير ما تكرر الأرقام كلها.`
   664→  }
   665→  if (tool === 'get_activity_history') {
   666→    const pending = cards
   667→      .flatMap((c) => (c.kind === 'table' ? c.rows : []))
   668→      .filter((r) => String(r.status ?? '').includes('مستنية')).length
   669→    if (pending > 0) return `قفل جملة قصيرة تنبّه على ${pending} عملية مستنية اعتماد من غير ما تكرر الأرقام كلها.`
   670→  }
   671→  return ''
   672→}

   674→/**
   675→ * Deterministic reply for read turns when the LLM reply stream fails
   676→ * (rate-limit / network). NEVER generic — quotes the REAL numbers the tools
   677→ * just fetched, so the owner still gets a useful answer with the data below.
   678→ */
   679→function deterministicReadReply(cards: ResultCardDTO[]): string {
   680→  for (const c of cards) {
   681→    if (c.kind === 'kv' && c.rows.length > 0) {
   682→      const parts = c.rows.slice(0, 4).map((r) => `${r.label} ${r.value}`).join(' · ')
   683→      return `دي أرقام «${c.title}» من سجلك: ${parts}.`
   684→    }
   685→  }
   686→  const table = cards.find((c) => c.kind === 'table') as Extract<ResultCardDTO, { kind: 'table' }> | undefined
   687→  if (table) {
   688→    const n = table.rows.length
   689→    return `دي «${table.title}» من سجلك — ${toArabicNum(n)} ${n === 1 ? 'صف' : n === 2 ? 'صفين' : 'صفوف'}، التفاصيل في الكارت تحت.`
   690→  }
   691→  const text = cards.find((c) => c.kind === 'text') as Extract<ResultCardDTO, { kind: 'text' }> | undefined
   692→  if (text && text.body.trim()) return text.body.trim().slice(0, 220)
   693→  return 'دي النتايج من سجل الشغل، التفاصيل في الكارت تحت.'
   694→}

   696→/** Deterministic numbered multi-intent reply — answers EVERY sub-request, no LLM. */
   697→function deterministicMultiReply(parts: string[], overflow: number): string {
   698→  const sections = parts.map((p, i) => `${AR_ORDINALS[i] ?? i + 1}. ${p}`)
   699→  const overflowNote = overflow > 0 ? `\n\n${toArabicNum(overflow)} ${overflow === 1 ? 'طلب' : 'طلبات'} تانية بعد دول — قولي نكمل فيهم؟` : ''
   700→  const closer = sections.some((s) => s.includes('مسودة') || s.includes('مستنية'))
   701→    ? '\n\nالمسودات تحت مستنية اعتمادك.'
   702→    : '\n\nتحب تفاصيل أكتر على أي جزء؟'
   703→  return sections.join('\n\n') + overflowNote + closer
   704→}

   706→/** Multi-intent LLM reply must answer every part with numbered sections — verify. */
   707→function replyAnswersAllParts(reply: string, expected: number): boolean {
   708→  const numbered = (reply.match(/^[٠-٩][.،)]/gm) ?? []).length + (reply.match(/^[1-4][.،)]/gm) ?? []).length
   709→  if (expected >= 2 && numbered < 2) return false
   710→  return true
   711→}

   713→/** Advisory fallback when the LLM stream dies — warm, NOT a chip wall. */
   714→function advisoryFallback(v: VerticalDef): string {
   715→  const two = v.chips.slice(0, 2)
   716→  return `معلش النت شكله تقيل دلوقتي 😅 — سؤالك مهم ومش هضيعه. جرّب تبعتله تاني بعد شوية، أو اسألني حاجة من الشغل زي: «${two.join('» · «')}».`
   717→}

   734→/**
   735→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   736→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   737→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   738→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   739→ * customer — the ل/من marker is what marks the ROLE.
   740→ */
   741→function findPartyMention<T extends { name: string }>(
   742→  text: string,
   743→  parties: T[],
   744→  marker: 'ل' | 'من' = 'ل',
   745→): T | null {
   746→  const norm = normalizeAr(text)
   747→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   748→  for (const p of parties) {
   749→    const words = normalizeAr(p.name)
   750→      .split(' ')
   751→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   752→      .filter((w) => w.length >= 2)
   753→    for (const w of words) {
   754→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   755→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   756→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   757→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   758→    }
   759→  }
   760→  return null
   761→}

   763→function fieldsToCard(ref: string, type: OpType, risk: Risk, summary: string, fields: { label: string; value: string; hint?: string }[], effects: { label: string; value: string; tone?: 'good' | 'bad' | 'neutral' }[], dangerNote?: string): ActionCardDTO {
   764→  return {

   766→    ref,
   767→    type,
   768→    risk,
   769→    title: TYPE_AR[type] ?? type,
   770→    summary,
   771→    fields,
   772→    effects,
   773→    status: 'AWAITING_APPROVAL',
   774→    dangerNote,
   775→  }
   776→}

   778→/** عطارة «البيع بالمبلغ»: «بـ20 جنيه فلفل أسود» → (amount, in EGP) or null.
   779→ * Runs on WESTERN-digit normalized text (normalizeAr already ate the ـ tatweel
   780→ * in «بـ20»). Requires a real money word (جنيه/ج) after the number so a bare
   781→ * quantity («بعت 20 فلفل») can never fire it. */
   782→function matchReverseSellAmount(normWestern: string): number | null {
   783→  const m = normWestern.match(/(?:^|\s)(?:بال|بمبلغ|ب)([0-9]+(?:[.,][0-9]+)?)\s*(?:جنيه|جنيهات|ج)(?=\s|$)/)
   784→  if (!m) return null
   785→  const amount = parseFloat(m[1].replace(',', '.'))
   786→  return Number.isFinite(amount) && amount > 0 ? amount : null
   787→}
   788→
   789→/** Remove the «بـN جنيه» clause so the money number can never masquerade as a
   790→ * weight in ANY quantity parser downstream (segment scan / parseQuantity). */
   791→function stripReverseSellAmount(normWestern: string): string {
   792→  return normWestern.replace(/(?:^|\s)(?:بال|بمبلغ|ب)[0-9]+(?:[.,][0-9]+)?\s*(?:جنيه|جنيهات|ج)(?=\s|$)/g, ' ')
   793→}
   794→
   795→/** Paid amount: LLM field → explicit «دفع 20» clause → آجل = 0 → default cash (total). */
   796→function resolvePaid(norm: string, f: Record<string, unknown>, total: number): number {
   797→  if (typeof f.paid === 'number' && f.paid > 0) return Math.min(Math.round(f.paid * 100), total)
   798→  const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   799→  if (paidWord?.[1]) {
   800→    const n = parseQuantity(paidWord[1], null).qty
   801→    if (n != null && n > 0) return Math.min(Math.round(n * 100), total)
   802→  }
   803→  if (/(?:بال)?اجل|علي الحساب|نسيون/.test(norm)) return 0
   804→  return total
   805→}

   807→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit, memCtx: MemoryContext): Promise<ResolvedWrite> {
   808→  const { business, user } = ctx
   809→  const type = (intent.opType ?? 'sale') as OpType
   810→  const f = intent.fields ?? {}
   811→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   812→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   813→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   814→
   815→  const text = ctx.text
   816→  const norm = normalizeAr(text)
   817→  const normW = toWesternDigits(norm)
   818→
   819→  switch (type) {
   820→    case 'sale':
   821→    case 'purchase': {
   822→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   823→      // «دفع ٢٠ بس» can never be misread as the sold quantity.
   824→      const textNoPaid = norm.replace(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?[٠-٩0-9][٠-٩0-9.,]*\s*(?:بس|فقط|تمام)?/g, ' ')
   825→      // ---- عطارة reverse-selling («بعت بـ20 جنيه فلفل») ----
   826→      // The money clause MUST be detected and stripped BEFORE any quantity
   827→      // parsing: a bare «20» in «بـ20 جنيه» is the AMOUNT, not a weight. The
   828→      // deterministic scanners and the LLM both tend to read it as qty=20 —
   829→      // the pattern match on «بـN جنيه» is the ground truth.
   830→      const reverseAmountEgp = business.type === 'attar' && type === 'sale' ? matchReverseSellAmount(normW) : null
   831→      const scanText = reverseAmountEgp != null ? stripReverseSellAmount(toWesternDigits(textNoPaid)) : textNoPaid
   832→      // ---- Multi-item scan (deterministic — works with or without the LLM) ----
   833→      // «سجّل بيع ٢ لبن و ١٠ بيض» → segments → per-item quantity. When TWO or
   834→      // more distinct catalog items are mentioned we draft ONE multi-line
   835→      // operation (a single basket the cashier approves once), like the real POS.
   836→      // Longest-alias-first + overlap rejection: «بانادول» alone can never fan
   837→      // out into «بانادول أكتيف» + «بانادول شرائط» — the more specific name wins.
   838→      const segments = scanText.split(/\s*(?:،|,|\+|؛|;)\s*|\s+و\s+|\s+و(?=[٠-٩0-9])/).filter(Boolean)
   839→      const byItem = new Map<string, { item: Item; qty: number | null; unit: string | null }>()
   840→      for (const seg of segments) {
   841→        const segNorm = normalizeAr(seg)
   842→        const hits: { item: Item; token: string; at: number }[] = []
   843→        for (const it of items) {
   844→          // canonical names + column aliases + LEXICON dialect aliases (boost)
   845→          const lexAl = memCtx.lexicon.filter((l) => l.entityRef === it.id).map((l) => l.alias)
   846→          const names = [it.name, ...(it.aliases ? it.aliases.split(',').map((a) => a.trim()) : []), ...lexAl]
   847→          let best: { token: string; at: number } | null = null
   848→          for (const n of names) {
   849→            const nn = normalizeAr(n)
   850→            if (nn.length >= 3) {
   851→              const at = segNorm.indexOf(nn)
   852→              if (at >= 0 && (!best || nn.length > best.token.length)) best = { token: nn, at }
   853→            }
   854→          }
   855→          if (best) hits.push({ item: it, ...best })
   856→        }
   857→        hits.sort((a, b) => b.token.length - a.token.length)
   858→        const taken: Array<[number, number]> = []
   859→        for (const h of hits) {
   860→          if (taken.some(([s, e]) => h.at < e && h.at + h.token.length > s)) continue
   861→          taken.push([h.at, h.at + h.token.length])
   862→          const q = parseQuantity(seg, null)
   863→          const prev = byItem.get(h.item.id)
   864→          byItem.set(h.item.id, {
   865→            item: h.item,
   866→            qty: q.qty != null && q.qty > 0 ? (prev?.qty ?? 0) + q.qty : prev?.qty ?? null,
   867→            unit: prev?.unit ?? unitToCode(q.unit) ?? null,
   868→          })
   869→        }
   870→      }
   871→      const multi = byItem.size >= 2
   872→
   873→      if (multi) {
   874→        // ---- Multi-line draft: one basket, real prices per line ----
   875→        const lines: { item: Item; qty: number; unitPrice: number; total: number }[] = []
   876→        const issues: string[] = []
   877→        for (const { item, qty, unit } of byItem.values()) {
   878→          const rawUnit = unit ?? item.unit // default = the item's own canonical unit
   879→          const baseQty = convertToBaseUnit(qty ?? 1, rawUnit, item.unit)
   880→          if (baseQty == null || !(baseQty > 0)) {
   881→            issues.push(`الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})`)
   882→            continue
   883→          }
   884→          const qtyN = Math.round(baseQty * 1000) / 1000
   885→          if (type === 'sale' && qtyN > item.stockQty) {
   886→            issues.push(`المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}`)
   887→            continue
   888→          }
   889→          const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   890→          lines.push({ item, qty: qtyN, unitPrice, total: Math.round(qtyN * unitPrice) })
   891→        }
   892→        if (lines.length === 0) {
   893→          return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: issues.join('؛ ') || 'الكمية قد إيه لكل صنف؟', issues }
   894→        }
   895→        const total = lines.reduce((s, l) => s + l.total, 0)
   896→        // Paid (clamped to the basket total; آجل = 0)
   897→        const paid = resolvePaid(norm, f, total)
   898→        // Party (lexicon first, then LLM field, then deterministic phrasing)
   899→        const lexCustomer = type === 'sale' ? lexiconResolve(norm, memCtx.lexicon, customers)?.entity : null
   900→        if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
   901→        const customer = lexCustomer ?? (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   902→        const supplier = (f.supplierName ? matchEntity(String(f.supplierName), suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   903→
   904→        const payload: Record<string, unknown> = {
   905→          lines: lines.map((l) => ({ itemId: l.item.id, name: l.item.name, qty: l.qty, unit: l.item.unit, unitPrice: l.unitPrice, total: l.total })),
   906→          total,
   907→          paid,
   908→          ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   909→          ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   910→        }
   911→        const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   912→        const risk = computeRisk(type, total)
   913→        const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${lines.map((l) => `${fmtQty(l.qty, arUnit(l.item.unit))} ${l.item.name}`).join(' + ')} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   914→        // BUSINESS_DNA credit-limit enforcement (آجل on a capped party)
   915→        let dangerNote: string | undefined
   916→        let creditField: { label: string; value: string; hint?: string } | undefined
   917→        if (type === 'sale' && customer && total - paid > 0 && memCtx.rules.length > 0) {
   918→          const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
   919→          if (check) {
   920→            dangerNote = check.warning
   921→            creditField = { label: '⚠️ قانون المحل', value: `هيوصل رصيد ${customer.name} لـ${egp(check.projectedPiastres)} — القانون: ${egp(check.limitPiastres)}` }
   922→            memCtx.referencedIds.push(check.rule.id)
   923→          }
   924→        }
   925→        const card = fieldsToCard(ref, type, risk, summary, [
   926→          ...lines.map((l) => ({ label: l.item.name, value: `${fmtQty(l.qty, arUnit(l.item.unit))} × ${egp(l.unitPrice)} = ${egp(l.total)}` })),
   927→          { label: 'الإجمالي', value: egp(total) },
   928→          { label: 'مدفوع', value: egp(paid) },
   929→          ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   930→          ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   931→          ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   932→          ...(creditField ? [creditField] : []),
   933→          ...issues.map((i) => ({ label: '⚠ مش ضمن المسودة', value: i })),
   934→        ], [
   935→          ...lines.map((l) => ({ label: `مخزون ${l.item.name}`, value: `${fmtQty(l.item.stockQty, arUnit(l.item.unit))} ← ${fmtQty(l.item.stockQty + (type === 'sale' ? -l.qty : l.qty), arUnit(l.item.unit))}`, tone: 'neutral' as const })),
   936→          { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' as const : 'bad' as const },
   937→        ], dangerNote ?? RISK_META[risk].hint)
   938→        return { ok: true, opType: type, payload, summary, card, issues: issues.length > 0 ? issues : undefined }
   939→      }
   940→
   941→      // ---- Single-item draft (original path) ----
   942→      // Item: lexicon (dialect alias) → LLM field → text mention → fuzzy whole-text match
   943→      const rawItemName = String(f.itemName ?? '')
   944→      const mentioned = byItem.size === 1 ? [...byItem.values()][0] : null
   945→      const lexHit = lexiconResolve(norm, memCtx.lexicon, items)
   946→      const item = lexHit?.entity ?? (rawItemName ? matchEntity(rawItemName, items)?.entity : null) ?? mentioned?.item ?? matchEntity(text, items)?.entity ?? null
   947→      if (lexHit) {
   948→        memCtx.referencedIds.push(lexHit.entry.id)
   949→      } else if (item) {
   950→        // LEXICON learning: a dialect token that overlaps the canonical name
   951→        const alias = learnableAlias(text, item)
   952→        if (alias) await saveLexiconMemory({ businessId: business.id, userId: user.id }, alias, item)
   953→      }
   954→      if (!item) {
   955→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: `مش لاقي الصنف — ممكن تكتبه بالاسم اللي في الكتالوج؟` }
   956→      }
   957→
   958→      // ---- عطارة reverse selling: «بعت بـ20 جنيه فلفل أسود» → وزن بالجرام ----
   959→      if (reverseAmountEgp != null) {
   960→        // scanText already has the money clause stripped, so ANY number still
   961→        // left in it is a real weight («بعت 100 جرام كركديه بـ20 جنيه» → qty wins)
   962→        const qtyAfterStrip = parseQuantity(scanText, null).qty
   963→        const digitsAfterStrip = /[0-9]/.test(scanText)
   964→        // The LLM tends to echo the money as qty — only a qty that ALSO has a
   965→        // real number left in the stripped text (or a word-number like نص) counts
   966→        const llmQtyIsReal =
   967→          typeof f.qty === 'number' && f.qty > 0 && f.qty !== reverseAmountEgp && (digitsAfterStrip || (f.unit && unitToCode(String(f.unit)) != null))
   968→        const noExplicitQty = qtyAfterStrip == null && mentioned?.qty == null && !llmQtyIsReal
   969→        if (noExplicitQty) {
   970→          if (item.unit !== 'gram' && item.unit !== 'kg') {
   971→            return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `«${item.name}» بيتاع بـ${arUnit(item.unit)} مش بالوزن — قولي الكمية بالوحدة بتاعته وأنا أحسبها لك` }
   972→          }
   973→          const amountEgp = reverseAmountEgp
   974→          const perGram = item.unit === 'gram' ? item.sellPrice : Math.round(item.sellPrice / 1000) // kg → per gram
   975→          const amountPiastres = Math.round(amountEgp * 100)
   976→          if (perGram > 0 && amountPiastres > 0) {
   977→            const grams = Math.round(amountPiastres / perGram)
   978→            if (grams < 1) {
   979→              return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `${egp(amountPiastres)} أقل من سعر الجرام (${egp(perGram)}) — الزبون ده عايز ياخد حاجة تانية؟` }
   980→            }
   981→            if (grams > item.stockQty) {
   982→              return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — ${egp(amountPiastres)} يعني ${fmtQty(grams, 'جرام')}` }
   983→            }
   984→            // revenue = the money the customer asked to buy with (integer piastres)
   985→            const total = amountPiastres
   986→            const paid = resolvePaid(norm, f, total)
   987→            const customer = (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? findPartyMention(text, customers)
   988→            const payload: Record<string, unknown> = {
   989→              itemId: item.id,
   990→              itemName: item.name,
   991→              qty: grams,
   992→              unit: 'gram',
   993→              unitPrice: perGram,
   994→              total,
   995→              paid,
   996→              reverseSell: true,
   997→              requestedAmount: amountPiastres,
   998→              ...(customer ? { customerId: customer.id, customerName: customer.name } : {}),
   999→            }
  1000→            const risk = computeRisk(type, total)
  1001→            const summary = `بيع بالوزن: ${fmtQty(grams, 'جرام')} ${item.name} بمبلغ ${egp(total)}${customer ? ` — ${customer.name}` : ''}`
  1002→            let dangerNote: string | undefined
  1003→            let creditField: { label: string; value: string; hint?: string } | undefined
  1004→            if (customer && total - paid > 0 && memCtx.rules.length > 0) {
  1005→              const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
  1006→              if (check) {
  1007→                dangerNote = check.warning
  1008→                creditField = { label: '⚠️ قانون المحل', value: `هيوصل رصيد ${customer.name} لـ${egp(check.projectedPiastres)} — القانون: ${egp(check.limitPiastres)}` }
  1009→                memCtx.referencedIds.push(check.rule.id)
  1010→              }
  1011→            }
  1012→            const card = fieldsToCard('', type, risk, summary, [
  1013→              { label: 'الصنف', value: item.name },
  1014→              { label: 'الوزن المحسوب', value: fmtQty(grams, 'جرام'), hint: `⚖️ اوزن ${fmtQty(grams, 'جرام')} للزبون` },
  1015→              { label: 'سعر الجرام', value: egp(perGram) },
  1016→              { label: 'المبلغ (بيع بـ)', value: egp(total) },
  1017→              { label: 'مدفوع', value: egp(paid) },
  1018→              ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
  1019→              ...(customer ? [{ label: 'العميل', value: customer.name }] : []),
  1020→              ...(creditField ? [creditField] : []),
  1021→            ], [
  1022→              { label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty - grams, arUnit(item.unit))}`, tone: 'neutral' },
  1023→              { label: 'الخزنة', value: egp(paid), tone: 'good' },
  1024→            ], dangerNote ?? RISK_META[risk].hint)
  1025→            return { ok: true, opType: type, payload, summary, card }
  1026→          }
  1027→        }
  1028→      }
  1029→
  1030→      // Quantity: LLM → mention → text (paid clause already stripped) → default 1
  1031→      // («بع لبن» = one unit of the item's own canonical kind — the cashier's shorthand)
  1032→      // (when a reverse-sell amount was detected but a real qty survived the
  1033→      // strip, the normal path runs — the explicit weight is the ground truth)
  1034→      const textQ = parseQuantity(reverseAmountEgp != null ? scanText : textNoPaid, null)
  1035→      const llmQty = typeof f.qty === 'number' && f.qty > 0 ? f.qty : null
  1036→      // reverse-sell candidate: an LLM qty equal to the money is the echo — drop it
  1037→      const qtySource = reverseAmountEgp != null && llmQty === reverseAmountEgp ? null : llmQty
  1038→      const qty = qtySource ?? mentioned?.qty ?? textQ.qty ?? 1
  1039→      if (qty <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: 'الكمية قد إيه بالظبط؟' }
  1040→      // Unit: LLM → mention → text word → the item's OWN canonical unit (الافتراضي)
  1041→      const rawUnit = unitToCode(f.unit as string) ?? mentioned?.unit ?? unitToCode(textQ.unit) ?? item.unit
  1042→      const baseQty = convertToBaseUnit(qty, rawUnit, item.unit)

  1299→// R17 «العقل الحقيقي» — THE BRAIN LOOP (model-driven tools + free chat)
  1300→// ---------------------------------------------------------------------------
  1301→/** One model-issued tool call: {"tool":"record_sale","args":{…}}. */
  1302→interface BrainToolCall {
  1303→  tool: string
  1304→  args: Record<string, unknown>
  1305→}
  1306→
  1307→/**
  1308→ * Parse a brain reply: {"tool_calls":[…]} (direct OR embedded in prose) →
  1309→ * the calls; null → plain conversational text (free-chat / final reply).
  1310→ */
  1311→function parseBrainReply(raw: string): BrainToolCall[] | null {
  1312→  const cleaned = raw.trim()
  1313→  if (!cleaned) return null
  1314→  const tryParse = (s: string): BrainToolCall[] | null => {
  1315→    try {
  1316→      const obj = JSON.parse(s) as { tool_calls?: unknown }
  1317→      if (obj && Array.isArray(obj.tool_calls)) {
  1318→        const calls = obj.tool_calls
  1319→          .filter(
  1320→            (c): c is Record<string, unknown> =>
  1321→              !!c && typeof c === 'object' && typeof (c as Record<string, unknown>).tool === 'string',
  1322→          )
  1323→          .map((c) => ({
  1324→            tool: String(c.tool),
  1325→            args: (c.args && typeof c.args === 'object' && !Array.isArray(c.args) ? (c.args as Record<string, unknown>) : {}) as Record<string, unknown>,
  1326→          }))
  1327→        if (calls.length > 0) return calls
  1328→      }
  1329→    } catch {
  1330→      /* not JSON → text */
  1331→    }
  1332→    return null
  1333→  }
  1334→  const direct = tryParse(cleaned.replace(/^```(?:json)?\s*/i, '').replace(/```\s*$/i, '').trim())
  1335→  if (direct) return direct
  1336→  // the model wrapped the JSON in prose — extract the {…} block
  1337→  const m = cleaned.match(/\{\s*"tool_calls"/)
  1338→  if (m && m.index !== undefined) {
  1339→    const end = cleaned.lastIndexOf('}')
  1340→    if (end > m.index) {
  1341→      const embedded = tryParse(cleaned.slice(m.index, end + 1))
  1342→      if (embedded) return embedded
  1343→    }
  1344→  }
  1345→  return null
  1346→}

  1349→ * Arg names are TOLERANT: models spell the same arg differently across calls

  1351→ * when the first spelling already lands. */

  1457→/**
  1458→ * The public turn runner — a telemetry shell around the real engine:
  1459→ * measures total latency + TTFT (first delta/step on the wire), resets the
  1460→ * LLM health flags (retry/fallback ladder), writes exactly ONE UsageEvent per
  1461→ * turn {kind, label = أول ٤٠ حرف من رسالة المستخدم, latencyMs, ttftMs,
  1462→ * tokens ≈ len/3, fallback}, plus a 'verify' event when verification did work.
  1463→ */
  1464→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
  1465→  ensurePrismaFresh() // dev-server prisma hot-reload seam (no-op on fresh processes)
  1466→  const t0 = Date.now()
  1467→  let ttftMs: number | null = null
  1468→  let sawFirstToken = false
  1469→  const meta: TurnMeta = { kind: 'chat', fallback: false, notesJotted: 0 }
  1470→  llmHealth.retried = false
  1471→  llmHealth.fellBack = false
  1472→  const trackedEmit: TurnEmit = (ev) => {
  1473→    if (!sawFirstToken && (ev.event === 'delta' || ev.event === 'step')) {
  1474→      sawFirstToken = true
  1475→      ttftMs = Date.now() - t0
  1476→    }
  1477→    emit(ev)
  1478→  }
  1479→  let result: TurnResult
  1480→  try {
  1481→    result = await runAgentTurnInner(ctx, trackedEmit, meta)
  1482→  } catch (err) {
  1483→    // the engine itself exploded — record the failure, then let the route's
  1484→    // catch send the SSE error (the user never sees a half-turn silently die)
  1485→    void trackTurnUsage(ctx.business.id, ctx.user.id, meta.kind, ctx.text, {
  1486→      latencyMs: Date.now() - t0,
  1487→      ttftMs,
  1488→      tokens: 0,
  1489→      fallback: true,
  1490→    })
  1491→    throw err
  1492→  }
  1493→  void trackTurnUsage(ctx.business.id, ctx.user.id, meta.kind, ctx.text, {
  1494→    latencyMs: Date.now() - t0,
  1495→    ttftMs,
  1496→    tokens: Math.ceil(result.replyText.length / 3),
  1497→    fallback: meta.fallback || llmHealth.retried || llmHealth.fellBack,
  1498→  })
  1499→  if (result.verified && (result.verified.status === 'corrected' || result.verified.checks.length > 0)) {
  1500→    void trackKindEvent(ctx.business.id, ctx.user.id, 'verify', result.verified.status)
  1501→  }
  1502→  result.notesJotted = meta.notesJotted
  1503→  return result
  1504→}
