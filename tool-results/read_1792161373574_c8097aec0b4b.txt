    50→export async function createDraft(params: {
    51→  businessId: string
    52→  userId: string
    53→  type: OpType
    54→  summary: string
    55→  payload: DraftFields
    56→  risk?: string
    57→  conversationId?: string | null
    58→  idempotencyKey: string
    59→}): Promise<Operation> {
    60→  const existing = await db.operation.findUnique({ where: { idempotencyKey: params.idempotencyKey } })
    61→  if (existing) return existing
    62→
    63→  const risk = params.risk ?? computeRisk(params.type, (params.payload.total as number) ?? undefined)
    64→  const ref = await nextRef(params.businessId)
    65→  const op = await db.operation.create({
    66→    data: {
    67→      ref,
    68→      businessId: params.businessId,
    69→      type: params.type,
    70→      status: 'AWAITING_APPROVAL',
    71→      source: 'chat',
    72→      userId: params.userId,
    73→      risk,
    74→      summary: params.summary,
    75→      payload: JSON.stringify(params.payload),
    76→      idempotencyKey: params.idempotencyKey,
    77→      conversationId: params.conversationId ?? null,
    78→    },
    79→  })
    80→  await audit(params.businessId, params.userId, 'operation.drafted', ref, { summary: params.summary, risk })
    81→  return op
    82→}

    87→export async function executeOperation(opId: string, decidedBy: string): Promise<{ op: Operation; outcome: ExecOutcome }> {
    88→  const op = await db.operation.findUnique({ where: { id: opId } })
    89→  if (!op) throw new Error('العملية مش موجودة')
    90→  if (op.status !== 'AWAITING_APPROVAL') throw new Error(`العملية مش في حالة انتظار اعتماد (حالتها: ${op.status})`)
    91→
    92→  await db.operation.update({
    93→    where: { id: opId },
    94→    data: { status: 'EXECUTING', decidedAt: new Date() },
    95→  })
    96→
    97→  const payload = JSON.parse(op.payload) as Record<string, unknown>
    98→  let outcome: ExecOutcome
    99→
   100→  try {
   101→    outcome = await EXECUTORS[op.type as OpType](op.businessId, op.ref, payload)
   102→  } catch (err) {
   103→    const reason = err instanceof Error ? err.message : 'فشل غير معروف'
   104→    await db.operation.update({
   105→      where: { id: opId },
   106→      data: { status: 'FAILED_SAFE', failReason: reason, executedAt: new Date() },
   107→    })
   108→    await audit(op.businessId, decidedBy, 'operation.failed_safe', op.ref, { reason })
   109→    return { op: await db.operation.findUniqueOrThrow({ where: { id: opId } }), outcome: { ok: false, checks: [], effects: [], failReason: reason } }
   110→  }
   111→
   112→  if (!outcome.ok) {
   113→    await db.operation.update({
   114→      where: { id: opId },
   115→      data: { status: 'FAILED_SAFE', failReason: outcome.failReason ?? 'فشل التحقق', executedAt: new Date() },
   116→    })
   117→    await audit(op.businessId, decidedBy, 'operation.failed_safe', op.ref, { failReason: outcome.failReason })
   118→  } else {
   119→    await db.operation.update({
   120→      where: { id: opId },
   121→      data: { status: 'SUCCEEDED', result: JSON.stringify(outcome), executedAt: new Date() },
   122→    })
   123→    await audit(op.businessId, decidedBy, 'operation.succeeded', op.ref, { checks: outcome.checks.length })
   124→  }
   125→  return { op: await db.operation.findUniqueOrThrow({ where: { id: opId } }), outcome }
   126→}

   128→export async function rejectOperation(opId: string, decidedBy: string, reason?: string): Promise<Operation> {
   129→  const op = await db.operation.findUniqueOrThrow({ where: { id: opId } })
   130→  if (op.status !== 'AWAITING_APPROVAL') throw new Error('العملية مش في حالة انتظار اعتماد')
   131→  const updated = await db.operation.update({
   132→    where: { id: opId },
   133→    data: { status: 'CANCELLED', reason: reason ?? 'رفضها المستخدم', decidedAt: new Date() },
   134→  })
   135→  await audit(op.businessId, decidedBy, 'operation.cancelled', op.ref, { reason: reason ?? 'رفض' })
   136→  return updated
   137→}

   144→async function stockBefore(itemId: string): Promise<number> {
   145→  const it = await db.item.findUniqueOrThrow({ where: { id: itemId }, select: { stockQty: true } })
   146→  return it.stockQty

   148→async function cashNet(businessId: string): Promise<number> {
   149→  const agg = await db.cashEntry.aggregate({ where: { businessId }, _sum: { amount: true } })
   150→  const entries = await db.cashEntry.findMany({ where: { businessId }, select: { direction: true, amount: true } })
   151→  const inn = entries.filter((e) => e.direction === 'in').reduce((s, e) => s + e.amount, 0)
   152→  const out = entries.filter((e) => e.direction === 'out').reduce((s, e) => s + e.amount, 0)
   153→  void agg
   154→  const biz = await db.business.findUniqueOrThrow({ where: { id: businessId }, select: { openingCash: true } })
   155→  return biz.openingCash + inn - out
   156→}

   390→    if (!item) return { ok: false, checks: [], effects: [], failReason: 'الصنف مش موجود' }
   391→    const delta = Number(p.delta ?? 0)
   392→    if (delta === 0) return { ok: false, checks: [], effects: [], failReason: 'التسوية صفر' }
   393→    const before = await stockBefore(itemId)
   394→    const after = before + Math.round(delta)
   395→    if (after < 0) return { ok: false, checks: [], effects: [], failReason: `التسوية هتخلي المخزون بالسالب (${fmtQty(after, arUnit(item.unit))})` }
   396→    const reason = String(p.reason ?? '')
   397→    if (!reason) return { ok: false, checks: [], effects: [], failReason: 'التسوية محتاجة سبب إلزامي' }
   398→    await db.item.update({ where: { id: itemId }, data: { stockQty: after } })
   399→    const checks = [{ label: 'المخزون اتعدل بالمقدار المطلوب', expected: fmtQty(after, arUnit(item.unit)), actual: fmtQty(await stockBefore(itemId), arUnit(item.unit)), pass: true }]
   400→    return {
   401→      ok: checks.every((c) => c.pass),
   402→      checks,
   403→      effects: [{ label: `مخزون ${item.name}`, value: `${fmtQty(before, arUnit(item.unit))} ← ${fmtQty(after, arUnit(item.unit))}`, tone: delta > 0 ? 'good' : 'bad' }],
   404→    }

   407→  async membership(businessId, ref, p) {
   408→    const customerId = p.customerId ? String(p.customerId) : null
   409→    const name = String(p.customerName ?? '')
   410→    const plan = String(p.plan ?? 'شهرية')
   411→    const total = Number(p.total ?? 0)
   412→    if (total <= 0) return { ok: false, checks: [], effects: [], failReason: 'قيمة الاشتراك لازم تكون أكبر من صفر' }
   413→    const paid = Number(p.paid ?? total)
   414→    const remaining = total - paid
   415→    const months = Number(p.months ?? 1)
   416→
   417→    let customer
   418→    if (customerId) {
   419→      customer = await db.customer.findUniqueOrThrow({ where: { id: customerId } })
   420→    } else {
   421→      const existing = await db.customer.findFirst({ where: { businessId, name } })
   422→      customer =
   423→        existing ??
   424→        (await db.customer.create({ data: { businessId, name, plan, balance: 0 } }))
   425→    }
   426→    const base = customer.expiresAt && customer.expiresAt > new Date() ? customer.expiresAt : new Date()
   427→    const newExpiry = new Date(base)
   428→    newExpiry.setMonth(newExpiry.getMonth() + months)
   429→
   430→    await db.customer.update({
   431→      where: { id: customer.id },
   432→      data: {
   433→        plan,
   434→        expiresAt: newExpiry,
   435→        ...(remaining !== 0 ? { balance: { increment: remaining } } : {}),
   436→      },
   437→    })
   438→    await db.cashEntry.create({ data: { businessId, direction: 'in', amount: paid, reason: `اشتراك — ${customer.name} (${plan})`, ref } })
   439→
   440→    const after = await db.customer.findUniqueOrThrow({ where: { id: customer.id } })
   441→    return {

   445→      ],
   446→      effects: [
   447→        { label: `اشتراك ${customer.name}`, value: `${plan} — لحد ${arDate(newExpiry)}`, tone: 'good' },
   448→        { label: 'خزنة (قبض)', value: egp(paid), tone: 'good' },
   449→        ...(remaining > 0 ? [{ label: 'باقي على العضو', value: egp(remaining), tone: 'bad' as const }] : []),
   450→      ],

   466→  async reversal(businessId, ref, p) {
   467→    const originalRef = String(p.originalRef ?? '')
   468→    const original = await db.operation.findFirst({
   469→      where: { businessId, ref: originalRef, status: 'SUCCEEDED' },
   470→    })
   471→    if (!original) return { ok: false, checks: [], effects: [], failReason: `العملية الأصلية ${originalRef} مش موجودة أو مش منفذة` }
   472→    if (original.reversedByRef) return { ok: false, checks: [], effects: [], failReason: `العملية دي معكوسة قبل كده بـ${original.reversedByRef}` }
   473→    const op = JSON.parse(original.payload) as Record<string, unknown>
   474→
   475→    // Inverse effects
   476→    switch (original.type as OpType) {
   477→      case 'sale': {
   478→        // Multi-line basket: restore each line's stock
   479→        const lines = Array.isArray(op.lines) ? (op.lines as Record<string, unknown>[]).filter((l) => l && typeof l === 'object') : []
   480→        if (lines.length > 0) {
   481→          for (const l of lines) {
   482→            await db.item.update({ where: { id: String(l.itemId) }, data: { stockQty: { increment: Math.round(Number(l.qty ?? 0)) } } })
   483→          }
   484→        } else if (op.itemId) await db.item.update({ where: { id: String(op.itemId) }, data: { stockQty: { increment: Math.round(Number(op.qty ?? 0)) } } })
   485→        if (op.paid) await db.cashEntry.create({ data: { businessId, direction: 'out', amount: Number(op.paid), reason: `عكس بيع ${originalRef}`, ref } })
   486→        if (op.customerId && Number(op.total ?? 0) - Number(op.paid ?? 0) !== 0) {
   487→          await db.customer.update({ where: { id: String(op.customerId) }, data: { balance: { decrement: Number(op.total) - Number(op.paid) } } })
   488→        }
   489→        break
   490→      }
   491→      case 'purchase': {
   492→        const lines = Array.isArray(op.lines) ? (op.lines as Record<string, unknown>[]).filter((l) => l && typeof l === 'object') : []
   493→        if (lines.length > 0) {
   494→          for (const l of lines) {
   495→            await db.item.update({ where: { id: String(l.itemId) }, data: { stockQty: { decrement: Math.round(Number(l.qty ?? 0)) } } })
   496→          }
   497→        } else if (op.itemId) await db.item.update({ where: { id: String(op.itemId) }, data: { stockQty: { decrement: Math.round(Number(op.qty ?? 0)) } } })
   498→        if (op.paid) await db.cashEntry.create({ data: { businessId, direction: 'in', amount: Number(op.paid), reason: `عكس شرا ${originalRef}`, ref } })
   499→        if (op.supplierId && Number(op.total ?? 0) - Number(op.paid ?? 0) !== 0) {
   500→          await db.supplier.update({ where: { id: String(op.supplierId) }, data: { balance: { decrement: Number(op.total) - Number(op.paid) } } })
   501→        }
   502→        break
   503→      }
   504→      case 'expense': {
   505→        await db.cashEntry.create({ data: { businessId, direction: 'in', amount: Number(op.total ?? 0), reason: `عكس مصروف ${originalRef}`, ref } })
   506→        break
   507→      }
   508→      case 'collection': {
   509→        await db.customer.update({ where: { id: String(op.customerId) }, data: { balance: { increment: Number(op.amount ?? 0) } } })
   510→        await db.cashEntry.create({ data: { businessId, direction: 'out', amount: Number(op.amount ?? 0), reason: `عكس تحصيل ${originalRef}`, ref } })
   511→        break
   512→      }
   513→      case 'payment': {
   514→        await db.supplier.update({ where: { id: String(op.supplierId) }, data: { balance: { increment: Number(op.amount ?? 0) } } })
   515→        await db.cashEntry.create({ data: { businessId, direction: 'in', amount: Number(op.amount ?? 0), reason: `عكس سداد ${originalRef}`, ref } })
   516→        break
   517→      }
   518→      default:
   519→        return { ok: false, checks: [], effects: [], failReason: `نوع العملية ${original.type} مش بيتعكس تلقائيًا` }
   520→    }
   521→
   522→    await db.operation.update({ where: { id: original.id }, data: { reversedByRef: ref } })
   523→    await db.operation.update({ where: { ref }, data: { reversalOfRef: originalRef } })
   524→    return {
   525→      ok: true,
   526→      checks: [{ label: `الأثر اتلغى لـ${originalRef}`, expected: 'معكوسة', actual: (await db.operation.findUniqueOrThrow({ where: { id: original.id } })).reversedByRef ?? '—', pass: true }],
   527→      effects: [{ label: `عكس ${original.summary}`, value: originalRef, tone: 'neutral' }],
   528→    }
   529→  },
   530→}
