    51→// ---------------------------------------------------------------------------
    52→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
    53→// and a hard floor so a slow provider never gets delay stacked on top of it.
    54→// ---------------------------------------------------------------------------
    55→const REPLY_PACING_BUDGET_MS = 1500
    56→const PIECE_DELAY_MIN_MS = 3
    57→const PIECE_DELAY_MAX_MS = 26
    58→
    59→interface PacerState {
    60→  t0: number | null
    61→  emitted: number
    62→}
    63→
    64→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
    65→ * The moment elapsed time eats the whole budget (the provider itself is slow),
    66→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
    67→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
    68→  const now = Date.now()
    69→  if (p.t0 == null) p.t0 = now
    70→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
    71→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
    72→  // Estimate the pieces still coming: at least the rest of this chunk, and at
    73→  // least as many as we've already emitted (replies rarely stop abruptly).
    74→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
    75→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
    76→}

    78→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
    79→ * (identity/memory replies: whole text known upfront, exact remaining count). */
    80→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
    81→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
    82→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
    83→  // project targets ES2017, where the `s` flag is a tsc error).
    84→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
    85→  const t0 = Date.now()
    86→  for (let i = 0; i < pieces.length; i++) {
    87→    emit({ event: 'delta', text: pieces[i] })
    88→    const remainingPieces = pieces.length - i - 1
    89→    if (remainingPieces <= 0) break
    90→    const remainingBudget = budgetMs - (Date.now() - t0)
    91→    if (remainingBudget <= 0) {
    92→      await sleep(PIECE_DELAY_MIN_MS)
    93→      continue
    94→    }
    95→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
    96→  }
    97→}

   129→/** «انت مين» / «اسمك ايه» / «بتعمل ايه»… (run on normalizeAr'd text).

   131→const IDENTITY_QUESTION_RES: RegExp[] = [
   132→  /(^|\s)(انت|انتا|انتي)\s*(مين|من)\s*(انت|انتي|يعني)?(\s|$|؟|\?)/,
   133→  /(^|\s)مين\s*(انت|انتا|انتي)(\s|$|؟|\?)/,

   137→  /بتعمل\s*(ايه|اد\s*ايه)/,
   138→  /(^|\s)انت\s*(ايه|مين)(\s|$|؟|\?)/,

   141→/** Strong business keywords mean the message is NOT about the agent's identity. */
   142→const IDENTITY_BLOCK_RE =
   143→  /(بعت|بيع|شرا|اشتري|خزن|كاش|مخزون|ناقص|ملخص|صلاحيه|مديون|مورد|تحصيل|مصروف|سدد|اشتراك|حضور|تسويه|عكس|احفظ|تسجل|سجل بيع)/

   145→function isIdentityQuestion(norm: string): boolean {
   146→  if (norm.length > 80 || IDENTITY_BLOCK_RE.test(norm)) return false
   147→  return IDENTITY_QUESTION_RES.some((re) => re.test(norm))

   150→const MEMORY_SAVE_RE = /(احفظ|خليك فاكر|افتكر اني|انا بفضل|بستحب)/
   151→const DIACRITICS = '[\\u064B-\\u0652\\u0670\\u0640]*'
   152→const CHAR_ALTS: Record<string, string> = { ا: '[اأإآٱ]', ي: '[يى]', ه: '[هة]' }
   153→
   154→/** Keyword → regex that tolerates diacritics/hamza variants in the ORIGINAL text. */
   155→function flexiblePattern(kw: string): RegExp {
   156→  const parts = Array.from(kw).map((c) => CHAR_ALTS[c] ?? c.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
   157→  return new RegExp(parts.join(DIACRITICS))
   158→}

   160→/** Extract the to-remember text from «احفظ اني…» style requests, keeping the
   161→ * user's own words. Returns null when nothing memorable follows the keyword. */
   162→function extractMemorySave(text: string, norm: string): string | null {
   163→  if (!MEMORY_SAVE_RE.test(norm) || norm.length > 260) return null
   164→  for (const kw of ['خليك فاكر اني', 'خليك فاكر', 'افتكر اني', 'احفظ', 'انا بفضل', 'بستحب']) {
   165→    const m = flexiblePattern(kw).exec(text)
   166→    if (!m) continue
   167→    const raw = text.slice(m.index + m[0].length)
   168→    const cleaned = raw
   169→      .replace(/^[،,.\s]+/, '')
   170→      .replace(/^(?:لي\s+)?(?:اني|إني|أني|ان|إن)\s+/, '')
   171→      .replace(/[،,.!؟?\s]+$/, '')
   172→      .trim()
   173→      .slice(0, 200)
   174→    return cleaned.length >= 3 ? cleaned : null
   175→  }
   176→  return null
   177→}

   222→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   223→  const s = normalizeAr(text)
   224→  const isGym = businessType === 'gym'
   225→
   226→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   227→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   228→  }
   229→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   230→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   231→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   232→  }
   233→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   234→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   235→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   236→  if (/(مديون|اللي ليا|تحصيلات|عليهم|العملاء|الاجل)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   237→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   238→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   239→  if (/حضور/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   240→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   241→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   242→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   243→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   244→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   245→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   246→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   247→    if (name && (isQuestion || /عضويه/.test(s))) {
   248→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   249→    }
   250→  }
   251→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   252→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   253→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   254→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   255→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   256→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   257→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   258→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   259→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   260→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   261→  // fire a collection (caught by the expiring rule above instead).
   262→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   263→    return { intent: 'write', opType: 'collection', fields: {} }
   264→  }
   265→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   266→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   267→  return { intent: 'chat', fields: {} }
   268→}

   270→function parseIntentJson(raw: string): ExtractedIntent | null {
   271→  const cleaned = raw.replace(/```json|```/g, '').trim()
   272→  const start = cleaned.indexOf('{')
   273→  const end = cleaned.lastIndexOf('}')
   274→  if (start < 0 || end <= start) return null
   275→  try {
   276→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent>
   277→    if (!json.intent) return null
   278→    return {
   279→      intent: json.intent === 'read' || json.intent === 'write' || json.intent === 'chat' ? json.intent : 'chat',
   280→      tool: json.tool,
   281→      opType: json.opType as OpType | undefined,
   282→      fields: (json.fields as Record<string, unknown>) ?? {},
   283→      question: json.question,
   284→      replyHint: json.replyHint,
   285→    }
   286→  } catch {
   287→    return null
   288→  }
   289→}

   295→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   296→  for (const c of cards) {
   297→    if (c.kind !== 'table' || !c.footer) continue
   298→    for (const f of c.footer) {
   299→      const m = String(f.value).match(/[\d٠-٩]+/)
   300→      if (m) {
   301→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   302→        const n = Number(west)
   303→        if (Number.isFinite(n)) return n
   304→      }
   305→    }
   306→  }
   307→  return null
   308→}

   318→  }
   319→  if (tool === 'get_activity_history') {
   320→    const pending = cards
   321→      .flatMap((c) => (c.kind === 'table' ? c.rows : []))
   322→      .filter((r) => String(r.status ?? '').includes('مستنية')).length
   323→    if (pending > 0) return `قفل جملة قصيرة تنبّه على ${pending} عملية مستنية اعتماد من غير ما تكرر الأرقام كلها.`
   324→  }
   325→  return ''
   326→}

   328→/**
   329→ * Deterministic reply for read turns when the LLM reply stream fails
   330→ * (rate-limit / network). NEVER generic — quotes the REAL numbers the tools
   331→ * just fetched, so the owner still gets a useful answer with the data below.
   332→ */
   333→function deterministicReadReply(cards: ResultCardDTO[]): string {
   334→  for (const c of cards) {
   335→    if (c.kind === 'kv' && c.rows.length > 0) {
   336→      const parts = c.rows.slice(0, 4).map((r) => `${r.label} ${r.value}`).join(' · ')
   337→      return `دي أرقام «${c.title}» من سجلك: ${parts}.`
   338→    }
   339→  }
   340→  const table = cards.find((c) => c.kind === 'table') as Extract<ResultCardDTO, { kind: 'table' }> | undefined
   341→  if (table) {
   342→    const n = table.rows.length
   343→    return `دي «${table.title}» من سجلك — ${n} ${n === 1 ? 'صف' : n === 2 ? 'صفين' : 'صفوف'}، التفاصيل في الكارت تحت.`
   344→  }
   345→  const text = cards.find((c) => c.kind === 'text') as Extract<ResultCardDTO, { kind: 'text' }> | undefined
   346→  if (text && text.body.trim()) return text.body.trim().slice(0, 220)
   347→  return 'دي النتايج من سجل الشغل، التفاصيل في الكارت تحت.'
   348→}

   364→interface ResolvedWrite {
   365→  ok: boolean

   367→  payload: Record<string, unknown>
   368→  summary: string
   369→  card: ActionCardDTO
   370→  missing?: string

   372→  /** Multi-item drafts: lines that could NOT be drafted (stock/unit), shown to the user. */
   373→  issues?: string[]

   376→/**
   377→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   378→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   379→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   380→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   381→ * customer — the ل/من marker is what marks the ROLE.
   382→ */
   383→function findPartyMention<T extends { name: string }>(
   384→  text: string,
   385→  parties: T[],
   386→  marker: 'ل' | 'من' = 'ل',
   387→): T | null {
   388→  const norm = normalizeAr(text)
   389→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   390→  for (const p of parties) {
   391→    const words = normalizeAr(p.name)
   392→      .split(' ')
   393→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   394→      .filter((w) => w.length >= 2)
   395→    for (const w of words) {
   396→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   397→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   398→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   399→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   400→    }
   401→  }
   402→  return null
   403→}

   406→  return {
   407→    kind: 'draft',
   408→    ref,
   409→    type,
   410→    risk,
   411→    title: TYPE_AR[type] ?? type,
   412→    summary,
   413→    fields,
   414→    effects,
   415→    status: 'AWAITING_APPROVAL',
   416→    dangerNote,
   417→  }
   418→}

   420→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit): Promise<ResolvedWrite> {
   421→  const { business, user } = ctx
   422→  const type = (intent.opType ?? 'sale') as OpType
   423→  const f = intent.fields ?? {}
   424→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   425→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   426→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   427→
   428→  const text = ctx.text
   429→  const norm = normalizeAr(text)
   430→
   431→  switch (type) {
   432→    case 'sale':
   433→    case 'purchase': {
   434→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   435→      // «دفع ٢٠ بس» can never be misread as the sold quantity.
   436→      const textNoPaid = norm.replace(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?[٠-٩0-9][٠-٩0-9.,]*\s*(?:بس|فقط|تمام)?/g, ' ')
   437→      // ---- Multi-item scan (deterministic — works with or without the LLM) ----
   438→      // «سجّل بيع ٢ لبن و ١٠ بيض» → segments → per-item quantity. When TWO or
   439→      // more distinct catalog items are mentioned we draft ONE multi-line
   440→      // operation (a single basket the cashier approves once), like the real POS.
   441→      // Longest-alias-first + overlap rejection: «بانادول» alone can never fan
   442→      // out into «بانادول أكتيف» + «بانادول شرائط» — the more specific name wins.
   443→      const segments = textNoPaid.split(/\s*(?:،|,|\+|؛|;)\s*|\s+و\s+|\s+و(?=[٠-٩0-9])/).filter(Boolean)
   444→      const byItem = new Map<string, { item: Item; qty: number | null; unit: string | null }>()
   445→      for (const seg of segments) {
   446→        const segNorm = normalizeAr(seg)
   447→        const hits: { item: Item; token: string; at: number }[] = []
   448→        for (const it of items) {
   449→          const names = [it.name, ...(it.aliases ? it.aliases.split(',').map((a) => a.trim()) : [])]
   450→          let best: { token: string; at: number } | null = null
   451→          for (const n of names) {
   452→            const nn = normalizeAr(n)
   453→            if (nn.length >= 3) {
   454→              const at = segNorm.indexOf(nn)
   455→              if (at >= 0 && (!best || nn.length > best.token.length)) best = { token: nn, at }
   456→            }
   457→          }
   458→          if (best) hits.push({ item: it, ...best })
   459→        }
   460→        hits.sort((a, b) => b.token.length - a.token.length)
   461→        const taken: Array<[number, number]> = []
   462→        for (const h of hits) {
   463→          if (taken.some(([s, e]) => h.at < e && h.at + h.token.length > s)) continue
   464→          taken.push([h.at, h.at + h.token.length])
   465→          const q = parseQuantity(seg, null)
   466→          const prev = byItem.get(h.item.id)
   467→          byItem.set(h.item.id, {
   468→            item: h.item,
   469→            qty: q.qty != null && q.qty > 0 ? (prev?.qty ?? 0) + q.qty : prev?.qty ?? null,
   470→            unit: prev?.unit ?? unitToCode(q.unit) ?? null,
   471→          })
   472→        }
   473→      }
   474→      const multi = byItem.size >= 2
   475→
   476→      if (multi) {
   477→        // ---- Multi-line draft: one basket, real prices per line ----
   478→        const lines: { item: Item; qty: number; unitPrice: number; total: number }[] = []
   479→        const issues: string[] = []
   480→        for (const { item, qty, unit } of byItem.values()) {
   481→          const rawUnit = unit ?? item.unit // default = the item's own canonical unit
   482→          const baseQty = convertToBaseUnit(qty ?? 1, rawUnit, item.unit)
   483→          if (baseQty == null || !(baseQty > 0)) {
   484→            issues.push(`الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})`)
   485→            continue
   486→          }
   487→          const qtyN = Math.round(baseQty * 1000) / 1000
   488→          if (type === 'sale' && qtyN > item.stockQty) {
   489→            issues.push(`المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}`)
   490→            continue
   491→          }
   492→          const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   493→          lines.push({ item, qty: qtyN, unitPrice, total: Math.round(qtyN * unitPrice) })
   494→        }
   495→        if (lines.length === 0) {
   496→          return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: issues.join('؛ ') || 'الكمية قد إيه لكل صنف؟', issues }
   497→        }
   498→        const total = lines.reduce((s, l) => s + l.total, 0)
   499→        // Paid (clamped to the basket total)
   500→        let paid = total
   501→        const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   502→        if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
   503→        else if (paidWord?.[1]) {
   504→          const n = parseQuantity(paidWord[1], null).qty
   505→          if (n != null && n > 0) paid = Math.round(n * 100)
   506→        }
   507→        paid = Math.min(paid, total)
   508→        // Party
   509→        const customer = (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   510→        const supplier = (f.supplierName ? matchEntity(String(f.supplierName), suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   511→
   512→        const payload: Record<string, unknown> = {
   513→          lines: lines.map((l) => ({ itemId: l.item.id, name: l.item.name, qty: l.qty, unit: l.item.unit, unitPrice: l.unitPrice, total: l.total })),
   514→          total,
   515→          paid,
   516→          ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   517→          ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   518→        }
   519→        const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   520→        const risk = computeRisk(type, total)
   521→        const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${lines.map((l) => `${fmtQty(l.qty, arUnit(l.item.unit))} ${l.item.name}`).join(' + ')} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   522→        const card = fieldsToCard(ref, type, risk, summary, [
   523→          ...lines.map((l) => ({ label: l.item.name, value: `${fmtQty(l.qty, arUnit(l.item.unit))} × ${egp(l.unitPrice)} = ${egp(l.total)}` })),
   524→          { label: 'الإجمالي', value: egp(total) },
   525→          { label: 'مدفوع', value: egp(paid) },
   526→          ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   527→          ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   528→          ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   529→          ...issues.map((i) => ({ label: '⚠ مش ضمن المسودة', value: i })),
   530→        ], [
   531→          ...lines.map((l) => ({ label: `مخزون ${l.item.name}`, value: `${fmtQty(l.item.stockQty, arUnit(l.item.unit))} ← ${fmtQty(l.item.stockQty + (type === 'sale' ? -l.qty : l.qty), arUnit(l.item.unit))}`, tone: 'neutral' as const })),
   532→          { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' as const : 'bad' as const },
   533→        ], RISK_META[risk].hint)
   534→        return { ok: true, opType: type, payload, summary, card, issues: issues.length > 0 ? issues : undefined }
   535→      }
   536→
   537→      // ---- Single-item draft (original path) ----
   538→      // Item: LLM field → text mention → fuzzy whole-text match
   539→      const rawItemName = String(f.itemName ?? '')
   540→      const mentioned = byItem.size === 1 ? [...byItem.values()][0] : null
   541→      const item = (rawItemName ? matchEntity(rawItemName, items)?.entity : null) ?? mentioned?.item ?? matchEntity(text, items)?.entity ?? null
   542→      if (!item) {
   543→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: `مش لاقي الصنف — ممكن تكتبه بالاسم اللي في الكتالوج؟` }
   544→      }
   545→      // Quantity: LLM → mention → text (paid clause already stripped) → default 1
   546→      // («بع لبن» = one unit of the item's own canonical kind — the cashier's shorthand)
   547→      const textQ = parseQuantity(textNoPaid, null)
   548→      const qty = (typeof f.qty === 'number' && f.qty > 0 ? f.qty : null) ?? mentioned?.qty ?? textQ.qty ?? 1
   549→      if (qty <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: 'الكمية قد إيه بالظبط؟' }
   550→      // Unit: LLM → mention → text word → the item's OWN canonical unit (الافتراضي)
   551→      const rawUnit = unitToCode(f.unit as string) ?? mentioned?.unit ?? unitToCode(textQ.unit) ?? item.unit
   552→      const baseQty = convertToBaseUnit(qty, rawUnit, item.unit)
   553→      if (baseQty == null || !(baseQty > 0)) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})` }
   554→      const qtyN = Math.round(baseQty * 1000) / 1000
   555→      if (type === 'sale' && qtyN > item.stockQty) {
   556→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}` }
   557→      }
   558→      // Total from REAL price
   559→      const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   560→      const total = Math.round(qtyN * unitPrice)
   561→      // Paid
   562→      let paid = total
   563→      const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   564→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
   565→      else if (paidWord?.[1]) {
   566→        const n = parseQuantity(paidWord[1], null).qty
   567→        if (n != null && n > 0) paid = Math.round(n * 100)
   568→      }
   569→      paid = Math.min(paid, total)
   570→      // Party: LLM field → deterministic Egyptian-phrasing scan («لأحمد», «العميل أحمد», «من المورد»)
   571→      const customerName = String(f.customerName ?? '')
   572→      const customer = (customerName ? matchEntity(customerName, customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   573→      const supplierName = String(f.supplierName ?? '')
   574→      const supplier = (supplierName ? matchEntity(supplierName, suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   575→
   576→      const payload: Record<string, unknown> = {
   577→        itemId: item.id,
   578→        itemName: item.name,
   579→        qty: qtyN,
   580→        unit: item.unit,
   581→        unitPrice,
   582→        total,
   583→        paid,
   584→        ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   585→        ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   586→      }
   587→      const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   588→      const risk = computeRisk(type, total)
   589→      const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${fmtQty(qtyN, arUnit(item.unit))} ${item.name} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   590→      const card = fieldsToCard(ref, type, risk, summary, [
   591→        { label: 'الصنف', value: item.name },
   592→        { label: 'الكمية', value: fmtQty(qtyN, arUnit(item.unit)) },
   593→        { label: `سعر ${type === 'sale' ? 'البيع' : 'الشرا'} (للوحدة)`, value: egp(unitPrice) },
   594→        { label: 'الإجمالي', value: egp(total) },
   595→        { label: 'مدفوع', value: egp(paid) },
   596→        ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   597→        ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   598→        ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   599→      ], [
   600→        { label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + (type === 'sale' ? -qtyN : qtyN), arUnit(item.unit))}`, tone: 'neutral' },
   601→        { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' : 'bad' },
   602→      ], RISK_META[risk].hint)
   603→      return { ok: true, opType: type, payload, summary, card }
   604→    }
   605→
   606→    case 'expense': {
   607→      const amountRaw = f.amount as number | undefined
   608→      let amount = amountRaw && amountRaw > 0 ? Math.round(amountRaw * 100) : null
   609→      if (amount == null) {
   610→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   611→        if (m) {
   612→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d))).replace(',', '.')
   613→          const parsed = parseFloat(western)
   614→          if (!Number.isNaN(parsed)) amount = Math.round(parsed * 100)
   615→        }
   616→      }
   617→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: 'المبلغ قد إيه؟' }
   618→      const label = String(f.label ?? '') || (norm.includes('كهربا') ? 'كهربا' : norm.includes('ايجار') ? 'إيجار' : norm.includes('تليفون') ? 'تليفون' : 'مصروف')
   619→      const payload = { label, total: amount }
   620→      const risk = computeRisk(type, amount)
   621→      const summary = `مصروف ${label} بـ${egp(amount)}`
   622→      const card = fieldsToCard('', type, risk, summary, [
   623→        { label: 'البند', value: label },
   624→        { label: 'المبلغ', value: egp(amount) },
   625→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
   626→      return { ok: true, opType: type, payload, summary, card }
   627→    }
   628→
   629→    case 'collection': {
   630→      const customersAll = customers.length > 0 ? customers : []
   631→      const customerName = String(f.customerName ?? '')
   632→      const customer = matchEntity(customerName || text, customersAll)?.entity ?? null
   633→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العميل', failMessage: 'تحصيل من مين بالظبط؟' }
   634→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
   635→      if (amount == null) {
   636→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   637→        if (m) {
   638→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   639→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
   640→        }
   641→      }
   642→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هتحصّل قد إيه من ${customer.name}؟` }
   643→      if (amount > customer.balance) {
   644→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `${customer.name} عليه ${egp(customer.balance)} بس — مش ممكن تحصّل ${egp(amount)}` }
   645→      }
   646→      const payload = { customerId: customer.id, customerName: customer.name, amount }
   647→      const risk = computeRisk(type, amount)
   648→      const summary = `تحصيل ${egp(amount)} من ${customer.name}`
   649→      const card = fieldsToCard('', type, risk, summary, [
   650→        { label: 'العميل', value: customer.name },
   651→        { label: 'المبلغ', value: egp(amount) },
   652→        { label: 'المديونية الحالية', value: egp(customer.balance) },
   653→        { label: 'بعد التحصيل', value: egp(customer.balance - amount) },
   654→      ], [{ label: 'الخزنة', value: egp(amount), tone: 'good' }], RISK_META[risk].hint)
   655→      return { ok: true, opType: type, payload, summary, card }
   656→    }
   657→
   658→    case 'payment': {
   659→      const supplierName = String(f.supplierName ?? '')
   660→      const supplier = matchEntity(supplierName || text, suppliers)?.entity ?? null
   661→      if (!supplier) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المورد', failMessage: 'هسدد لمين؟' }
   662→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
   663→      if (amount == null) {
   664→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
   665→        if (m) {
   666→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   667→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
   668→        }
   669→      }
   670→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هسدد قد إيه لـ${supplier.name}؟` }
   671→      if (amount > supplier.balance) {
   672→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المستحق لـ${supplier.name} ${egp(supplier.balance)} بس` }
   673→      }
   674→      const payload = { supplierId: supplier.id, supplierName: supplier.name, amount }
   675→      const risk = computeRisk(type, amount)
   676→      const summary = `سداد ${egp(amount)} لـ${supplier.name}`
   677→      const card = fieldsToCard('', type, risk, summary, [
   678→        { label: 'المورد', value: supplier.name },
   679→        { label: 'المبلغ', value: egp(amount) },
   680→        { label: 'المستفق الحالي', value: egp(supplier.balance) },
   681→        { label: 'بعد السداد', value: egp(supplier.balance - amount) },
   682→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
   683→      return { ok: true, opType: type, payload, summary, card }
   684→    }
   685→
   686→    case 'membership': {
   687→      const isRenewal = norm.includes('تجديد') || norm.includes('جدد')
   688→      const explicitName = String(f.customerName ?? '').trim()
   689→      const customer = explicitName ? matchEntity(explicitName, customers)?.entity ?? null : matchEntity(text, customers)?.entity ?? null
   690→      if (!customer && isRenewal) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'تجديد لمين؟' }
   691→      // A NEW subscription with NO name at all (rule path, question-like text)
   692→      // must never draft «عضو جديد» — ask for the name first.
   693→      if (!customer && !explicitName) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'اشتراك جديد لمين؟ اكتبلي اسم العضو' }
   694→      const plan = String(f.plan ?? 'شهرية')
   695→      const months = Math.max(1, Math.min(12, Math.round(Number(f.months ?? 1))))
   696→      // plan price from items (gym plans are items) or fallback 35000
   697→      const planItem = items.find((i) => normalizeAr(i.name).includes(normalizeAr(plan)))
   698→      const total = Math.round((planItem?.sellPrice ?? 35000) * months)
   699→      let paid = total
   700→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
   701→      paid = Math.min(paid, total)
   702→      const name = customer?.name ?? explicitName
   703→      const payload = {
   704→        ...(customer ? { customerId: customer.id } : { customerName: name }),
   705→        plan,
   706→        months,
   707→        total,
   708→        paid,
   709→      }
   710→      const risk = computeRisk(type, total)
   711→      const summary = `${isRenewal ? 'تجديد' : 'اشتراك جديد'} ${plan} لـ${name} — ${egp(total)}`
   712→      const card = fieldsToCard('', type, risk, summary, [
   713→        { label: 'العضو', value: name },
   714→        { label: 'الخطة', value: plan },
   715→        { label: 'المدة', value: fmtQty(months, 'شهر') },
   716→        { label: 'الإجمالي', value: egp(total) },
   717→        { label: 'مدفوع', value: egp(paid) },
   718→      ], [
   719→        { label: 'الخزنة', value: egp(paid), tone: 'good' },
   720→        ...(total - paid > 0 ? [{ label: 'باقي على العضو', value: egp(total - paid), tone: 'bad' as const }] : []),
   721→      ], RISK_META[risk].hint)
   722→      return { ok: true, opType: type, payload, summary, card }
   723→    }
   724→
   725→    case 'attendance': {
   726→      const customerName = String(f.customerName ?? '')
   727→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
   728→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'حضور مين؟' }
   729→      const payload = { customerId: customer.id, customerName: customer.name }
   730→      const risk = computeRisk(type)
   731→      const summary = `حضور ${customer.name}`
   732→      const card = fieldsToCard('', type, risk, summary, [
   733→        { label: 'العضو', value: customer.name },
   734→        { label: 'اليوم', value: 'النهاردة' },
   735→        ...(customer.expiresAt ? [{ label: 'الاشتراك لحد', value: arDate(customer.expiresAt) }] : []),
   736→      ], [], RISK_META[risk].hint)
   737→      return { ok: true, opType: type, payload, summary, card }
   738→    }
   739→
   740→    case 'stock_adjustment': {
   741→      const rawItemName = String(f.itemName ?? '')
   742→      const item = matchEntity(rawItemName || text, items)?.entity ?? null
   743→      if (!item) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: 'تسوية مخزون لأنهي صنف؟' }
   744→      let delta = (f.delta as number | undefined) ?? null
   745→      if (delta == null && typeof f.qty === 'number') delta = f.qty - item.stockQty
   746→      if (delta == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية الجديدة/الفرق', failMessage: `المخزون الحالي ${fmtQty(item.stockQty, arUnit(item.unit))} — عايزه يبقى قد إيه؟` }
   747→      const reason = String(f.reason ?? '') || 'تسوية جرد'
   748→      const payload = { itemId: item.id, itemName: item.name, delta, reason, unit: item.unit }
   749→      const risk = computeRisk(type)
   750→      const summary = `تسوية مخزون ${item.name}: ${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`
   751→      const card = fieldsToCard('', type, risk, summary, [
   752→        { label: 'الصنف', value: item.name },
   753→        { label: 'الحالي', value: fmtQty(item.stockQty, arUnit(item.unit)) },
   754→        { label: 'الجديد', value: fmtQty(item.stockQty + delta, arUnit(item.unit)) },
   755→        { label: 'السبب', value: reason },
   756→      ], [{ label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`, tone: delta >= 0 ? 'good' : 'bad' }], 'تسوية المخزون عملية حساسة (R3) — محتاجة سبب واعتماد')
   757→      return { ok: true, opType: type, payload, summary, card }
   758→    }
   759→
   760→    case 'reversal': {
   761→      const ref = String(f.ref ?? '') || (norm.match(/WKL-\d+/)?.[0] ?? '')
   762→      if (!ref) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'رقم العملية', failMessage: 'عايز تعكس أنهي عملية؟ اكتب رقمها (زي WKL-1006)' }
   763→      const original = await db.operation.findFirst({ where: { businessId: business.id, ref } })
   764→      if (!original) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `ملقيتش عملية رقم ${ref}` }
   765→      if (original.reversedByRef) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `العملية ${ref} معكوسة قبل كده بـ${original.reversedByRef}` }
   766→      const reason = String(f.reason ?? '') || 'تصحيح خطأ'
   767→      const payload = { originalRef: ref, reason }
   768→      const risk = computeRisk(type)
   769→      const summary = `عكس العملية ${ref} (${original.summary})`
   770→      const card = fieldsToCard('', type, risk, summary, [
   771→        { label: 'العملية الأصلية', value: `${ref} — ${TYPE_AR[original.type] ?? original.type}` },
   772→        { label: 'ملخصها', value: original.summary },
   773→        { label: 'السبب', value: reason },
   774→      ], [{ label: 'الأثر', value: 'هيتلغى بالكامل بعملية عكسية موثقة', tone: 'neutral' }], 'عكس عملية مالية حساس (R3) — التاريخ مش بيتمسح، بيتصحح')
   775→      return { ok: true, opType: type, payload, summary, card }
   776→    }
   777→
   778→    default:
   779→      return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: 'نوع العملية مش مدعوم' }
   780→  }
   781→}

   786→export interface TurnResult {
   787→  replyText: string
   788→  card?: ActionCardDTO | null
   789→  cards?: ResultCardDTO[] | null
   790→  thinking: ThinkingStep[]
   791→  modelLabel: string
   792→  opId?: string
   793→  actions?: ChatActionDTO[] | null
   794→}
   795→
   796→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
   797→  const { business, user, text } = ctx
   798→  const v = getVertical(business.type)
   799→  const model = await resolveModel(ctx.modelKey)
   800→  const modelLabel = modelLabelFor(model)
   801→
   802→  // Load context once (memories raised to 8; pending ops count for the persona)
   803→  const [items, customers, suppliers, memories, pendingCount] = await Promise.all([
   804→    db.item.findMany({ where: { businessId: business.id } }),
   805→    db.customer.findMany({ where: { businessId: business.id } }),
   806→    db.supplier.findMany({ where: { businessId: business.id } }),
   807→    db.agentMemory.findMany({ where: { businessId: business.id, userId: user.id }, orderBy: { createdAt: 'desc' }, take: 8 }),
   808→    db.operation.count({ where: { businessId: business.id, status: 'AWAITING_APPROVAL' } }),
   809→  ])
   810→
   811→  // Thinking steps — with a «فتكرت من ذاكرتي» recall step injected right after
   812→  // «بفهم طلبك» whenever the agent actually remembers something about this user.
   813→  const memStep: ThinkingStep[] =
   814→    memories.length > 0
   815→      ? [
   816→          {
   817→            label: 'فتكرت من ذاكرتي',
   818→            detail: memories[0].text.slice(0, 60) + (memories[0].text.length > 60 ? '…' : ''),
   819→            state: 'done',
   820→            icon: 'verify',
   821→          },
   822→        ]
   823→      : []
   824→  const steps: ThinkingStep[] = [
   825→    { label: 'بفهم طلبك', detail: 'بحلل الجملة وأستخرج القصد والحقول', state: 'active', icon: 'understand' },
   826→    ...memStep,
   827→    { label: 'بدوّر في سجلك', detail: 'بدوّر على الأصناف والأسماء الحقيقية', state: 'pending', icon: 'search' },
   828→    { label: 'بتحقق من الأرقام', detail: 'بحسب من الأسعار الفعلية مش من دماغي', state: 'pending', icon: 'verify' },
   829→    { label: 'بجهّز الرد', detail: '', state: 'pending', icon: 'plan' },
   830→  ]
   831→  const IDX = { understand: 0, search: 1 + memStep.length, verify: 2 + memStep.length, reply: 3 + memStep.length }
   832→  const pushThinking = () => emit({ event: 'thinking', steps: steps.map((s) => ({ ...s })) })
   833→  const done = (i: number, detail?: string) => {
   834→    if (steps[i]) {
   835→      steps[i].state = 'done'
   836→      if (detail) steps[i].detail = detail
   837→    }
   838→    pushThinking()
   839→  }
   840→  pushThinking()
   841→
   842→  const norm = normalizeAr(text)
   843→
   844→  // ---- PASS 0a: Identity question → instant, deterministic, zero LLM ----
   845→  if (isIdentityQuestion(norm)) {
   846→    const lowStockCount = items.filter((i) => i.stockQty <= i.lowStockThreshold).length
   847→    const receivablesTotal = customers.reduce((s, c) => s + Math.max(0, c.balance), 0)
   848→    done(IDX.understand, 'سؤال عن هويتي — بجاوب من سجلك من غير موديل')
   849→    done(IDX.search, 'جبت بياناتي من سجلك الحقيقي')
   850→    done(IDX.verify, 'عدّيت النواقص والعمليات المستنية من قاعدة البيانات')
   851→    emit({ event: 'step', label: 'بتعرّف بنفسي', state: 'done' })
   852→    done(IDX.reply)
   853→    const replyText = buildIdentityReply({
   854→      business,
   855→      user,
   856→      verticalLabel: v.label,
   857→      memoryText: memories[0]?.text ?? null,
   858→      lowStockCount,
   859→      pendingCount,
   860→      receivablesTotal,
   861→    })
   862→    const actions: ChatActionDTO[] = [
   863→      { id: 'act-today', label: 'لوحة اليوم', view: 'today', icon: 'today', tone: 'primary' },
   864→      ...(lowStockCount > 0
   865→        ? [{ id: 'act-inventory', label: 'شوف المخزون', view: 'inventory' as const, icon: 'boxes' as const, tone: 'outline' as const }]
   866→        : []),
   867→    ]
   868→    await streamDeterministic(replyText, emit, 1200)
   869→    emit({ event: 'actions', actions })
   870→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions }
   871→  }
   872→
   873→  // ---- PASS 0b: Memory save → instant, persisted, zero LLM ----
   874→  const memoryText = extractMemorySave(text, norm)
   875→  if (memoryText) {
   876→    // Dedupe: re-saving the exact same preference must NOT clutter the memory
   877→    // list — the owner repeating himself deserves a calm «already saved».
   878→    const isDuplicate = memories.some((m) => normalizeAr(m.text) === normalizeAr(memoryText))
   879→    if (!isDuplicate) {
   880→      await db.agentMemory.create({
   881→        data: { businessId: business.id, userId: user.id, kind: 'preference', text: memoryText },
   882→      })
   883→    }
   884→    done(IDX.understand, 'طلب حفظ في الذاكرة — مش محتاج موديل')
   885→    done(IDX.search, 'بتخزنها في ذاكرتي عنك')
   886→    done(IDX.verify, isDuplicate ? 'موجودة أصلًا — من غير تكرار' : 'اتحفظت فعليًا في قاعدة البيانات')
   887→    emit({ event: 'step', label: 'بحفظ في ذاكرتي', state: 'done' })
   888→    done(IDX.reply)
   889→    const replyText = isDuplicate
   890→      ? `دي محفوظة عندي أصلًا ✅ — «${memoryText}». مطمن، مش هنسيها.`
   891→      : `اتحفظت في ذاكرتي ✅ — هفتكرها: «${memoryText}». لما الموقف ده يتكرر هرفعها على طول.`
   892→    await streamDeterministic(replyText, emit, 1000)
   893→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions: null }
   894→  }
   895→
   896→  // ---- PASS 1: Understand (LLM with rule-based fallback) ----
   897→  let intent: ExtractedIntent
   898→  try {
   899→    const system = buildSystemPrompt({ business, user, items, customers, suppliers, memories, history: ctx.history })
   900→    const raw = await completeChat(model, [
   901→      { role: 'system', content: system },
   902→      { role: 'user', content: text },
   903→    ])
   904→    intent = parseIntentJson(raw) ?? ruleBasedIntent(text, business.type)
   905→    done(IDX.understand, intent.replyHint ? `فهمت: ${intent.replyHint}` : 'فهمت الطلب')
   906→  } catch (err) {
   907→    console.error('[harness] intent pass failed → rule-based fallback:', err)
   908→    intent = ruleBasedIntent(text, business.type)
   909→    done(IDX.understand, 'فهمت الطلب بالقواعد المحلية (الموديل مش متاح)')
   910→  }
   911→
   912→  // ---- PASS 2+3: Act ----
   913→  let situation = ''
   914→  let card: ActionCardDTO | null = null
   915→  let cards: ResultCardDTO[] | null = null
   916→  let opId: string | undefined
   917→  let replyInstruction = ''
   918→  let questionMode = false
   919→  let actions: ChatActionDTO[] | null = null
   920→  let permissionDenied = false
   921→  let permissionMsg = ''
   922→  let draftIssues: string[] | null = null
   923→
   924→  if (intent.intent === 'read' && intent.tool) {
   925→    emit({ event: 'step', label: `بشغّل أداة ${intent.tool}`, state: 'active' })
   926→    done(IDX.search, 'لقيت البيانات في السجل')
   927→    cards = await runReadTool(intent.tool as ReadTool, business.id, intent.fields ?? {})
   928→    emit({ event: 'cards', cards })
   929→    emit({ event: 'step', label: `بشغّل أداة ${intent.tool}`, state: 'done' })
   930→    done(IDX.verify, 'الأرقام جاية من قاعدة البيانات مباشرة')
   931→    situation = `شغّلت أداة القراءة ${intent.tool} ودي النتايج:\n${JSON.stringify(cards, null, 1).slice(0, 1600)}`
   932→    replyInstruction = 'لخّص النتيجة للمستخدم بالمصري بأرقامها الصحيحة من الجدول فوق. لو مفيش بيانات قول كده بصراحة.'
   933→    const hint = proactiveHintFor(intent.tool, cards)
   934→    if (hint) replyInstruction += ` ${hint}`
   935→    actions = TOOL_ACTIONS[intent.tool] ?? null
   936→  } else if (intent.intent === 'write' && intent.opType) {
   937→    const opType = intent.opType as OpType
   938→    if (!canDraft(user.role, opType)) {
   939→      permissionDenied = true
   940→      permissionMsg = `عملية «${TYPE_AR[opType] ?? opType}» محتاجة صلاحية أعلى — صلاحية ${user.name} (${roleTitle(user)}) مش بتسمح بتسجيلها. دي من اختصاص صاحب المنشأة أو المشرف بس، ولا حاجة اتنفذت.`
   941→      situation = `المستخدم (${user.role} — ${roleTitle(user)}) مش مسموحله يسجّل ${TYPE_AR[opType] ?? opType}. العملية دي بتتسجل من المالك أو المشرف بس.`
   942→      replyInstruction = 'وضّح بذوق إن العملية دي محتاجة صلاحية أعلى (المالك أو المشرف) ومتشغلش أي أداة. أكّد إن مفيش حاجة اتنفذت.'
   943→      done(IDX.search, 'راجعت الصلاحيات — العملية دي مش من نصيبك')
   944→      done(IDX.verify, 'ولا حاجة اتنفذت — الخزنة والمخزون زي ما هما')
   945→    } else {
   946→      done(IDX.search, 'بمشي على الأصناف والأسماء الحقيقية')
   947→      const resolved = await resolveWrite(ctx, intent, emit)
   948→      if (!resolved.ok) {
   949→        done(IDX.verify, resolved.failMessage ?? '')
   950→        situation = `معلومات ناقصة: ${resolved.missing ?? ''} — ${resolved.failMessage ?? ''}`
   951→        replyInstruction = 'اسأل سؤال واحد قصير يطلب أهم حقل ناقص. متاخدش أي قرار.'
   952→        questionMode = true
   953→      } else {
   954→        done(IDX.verify, `حسبت الإجمالي من السعر الحقيقي: ${resolved.summary}`)
   955→        emit({ event: 'step', label: `بجهّز مسودة ${TYPE_AR[resolved.opType]}`, state: 'active' })
   956→        const idem = `${business.id}:${user.id}:${Date.now()}:${Math.random().toString(36).slice(2, 8)}`
   957→        const draft = await createDraft({
   958→          businessId: business.id,
   959→          userId: user.id,
   960→          type: resolved.opType,
   961→          summary: resolved.summary,
   962→          payload: resolved.payload,
   963→          conversationId: ctx.conversationId,
   964→          idempotencyKey: idem,
   965→        })
   966→        opId = draft.id
   967→        card = { ...resolved.card, ref: draft.ref, opId: draft.id }
   968→        emit({ event: 'card', card })
   969→        emit({ event: 'step', label: `بجهّز مسودة ${TYPE_AR[resolved.opType]}`, state: 'done' })
   970→        situation = `اتعملت مسودة عملة رقم ${draft.ref} (${resolved.summary}) مستنية اعتماد المستخدم. درجة الخطورة ${draft.risk}. مفيش حاجة اتنفذت في الخزنة أو المخزون لسه.`
   971→        replyInstruction = `اكتب تأكيد قصير إن المسودة جاهزة بالأرقام الصح وإنها مستنية اعتماده${draft.risk === 'R3' ? '، وذكّره إنها عملية حساسة محتاجة سبب' : ''}. خليك مختصر.`
   972→        // Multi-item drafts with lines that could NOT be drafted — NEVER silent:
   973→        // the reply must explicitly name them and why.
   974→        if (resolved.issues && resolved.issues.length > 0) {
   975→          situation += ` بس فيه أصناف اتمسك معاها: ${resolved.issues.join('؛ ')}.`
   976→          replyInstruction += ` وبعدين نبّه صراحة على الأصناف اللي مش ضمن المسودة وسببها: ${resolved.issues.join('؛ ')} — لو عايز يسجلها يبعتها في رسالة لوحدها.`
   977→          draftIssues = resolved.issues
   978→        }
   979→        actions = DRAFT_ACTIONS
   980→      }
   981→    }
   982→  } else {
   983→    // chat
   984→    done(IDX.search, 'كلام عام')
   985→    done(IDX.verify, '')
   986→    situation = `رسالة عامة (تحية/سؤال خارج الأدوات). سياق الشغل: ${v.label} — ${business.name}.`
   987→    replyInstruction = `ردّ بالمصري بلطف وقصر. لو مناسب اعرض 2-3 من الأوامر اللي بتنفع ${v.shortLabel}: ${v.chips.slice(0, 3).join(' · ')}`
   988→  }
   989→
   990→  done(IDX.reply)
   991→
   992→  // ---- PASS 4: Explain (letter-by-letter, ADAPTIVE pacing) ----
   993→  let replyText = ''
   994→  const pacer: PacerState = { t0: null, emitted: 0 }
   995→  // NOTE: the reply pass MUST carry a trailing user turn — providers reject a
   996→  // conversation that is only a system/assistant prompt (z-ai: code 1214).
   997→  const messages: LLMMessage[] = [
   998→    { role: 'system', content: buildReplyPrompt({ business: { name: business.name, type: business.type }, userName: user.name, situation, instruction: replyInstruction }) },
   999→    { role: 'user', content: 'اكتب ردك النهائي دلوقتي بناءً على الوضع الحالي والمطلوب فوق.' },
  1000→  ]
  1001→  try {
  1002→    for await (const chunk of streamChat(model, messages)) {
  1003→      if (chunk.type !== 'content') continue
  1004→      replyText += chunk.text
  1005→      // Re-pace into tiny letter-by-letter deltas — adaptively: total pacing
  1006→      // is capped by the budget; a slow provider gets the 3ms floor instead.
  1007→      // [\s\S] keeps the model's own line breaks alive in the stream
  1008→      // (plain `.` drops \n; ES2017 target forbids the dotAll `s` flag).
  1009→      const pieces = chunk.text.match(/[\s\S]{1,2}/g) ?? []
  1010→      for (let i = 0; i < pieces.length; i++) {
  1011→        emit({ event: 'delta', text: pieces[i] })
  1012→        pacer.emitted++
  1013→        await sleep(adaptivePieceDelay(pacer, pieces.length - i - 1))
  1014→      }
  1015→    }
  1016→  } catch (err) {
  1017→    console.error('[harness] reply stream failed:', err)
  1018→    if (!replyText) {
  1019→      const fallback = card
  1020→        ? `جهزتلك مسودة «${card.title}» — ${card.summary}. راجعها واعتمدها لو تمام.${draftIssues ? ` بس خد بالك: ${draftIssues.join('؛ ')}.` : ''}`
  1021→        : cards
  1022→          ? deterministicReadReply(cards)
  1023→          : questionMode
  1024→            ? (situation.match(/— (.+)$/)?.[1] ?? 'محتاج تفاصيل أكتر لو سمحت.')
  1025→            : permissionDenied
  1026→              // The cashier must NEVER read «تمام» on a denied operation —
  1027→              // that reads like «it's done». Say the permission truth instead.
  1028→              ? permissionMsg
  1029→              : chatFallbackWithChips(v.chips)
  1030→      await streamDeterministic(fallback, emit, 900)
  1031→      replyText = fallback
  1032→    }
  1033→  }
  1034→
  1035→  if (!replyText.trim()) {
  1036→    const fallback = card ? `جهزت المسودة — ${card.summary}` : permissionDenied ? permissionMsg : chatFallbackWithChips(v.chips)
  1037→    await streamDeterministic(fallback, emit, 900)
  1038→    replyText = fallback
  1039→  }
  1040→
  1041→  // ---- PASS 5: Deep-link action buttons (after deltas, before done) ----
  1042→  if (actions && actions.length > 0) {
  1043→    emit({ event: 'actions', actions })
  1044→  }
  1045→
  1046→  return { replyText, card, cards, thinking: steps, modelLabel, opId, actions: actions && actions.length > 0 ? actions : null }
  1047→}
