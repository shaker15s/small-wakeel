    85→// ---------------------------------------------------------------------------
    86→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
    87→// and a hard floor so a slow provider never gets delay stacked on top of it.
    88→// ---------------------------------------------------------------------------
    89→const REPLY_PACING_BUDGET_MS = 1500
    90→const PIECE_DELAY_MIN_MS = 3
    91→const PIECE_DELAY_MAX_MS = 26
    92→
    93→interface PacerState {
    94→  t0: number | null
    95→  emitted: number
    96→}
    97→
    98→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
    99→ * The moment elapsed time eats the whole budget (the provider itself is slow),
   100→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
   101→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
   102→  const now = Date.now()
   103→  if (p.t0 == null) p.t0 = now
   104→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
   105→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
   106→  // Estimate the pieces still coming: at least the rest of this chunk, and at
   107→  // least as many as we've already emitted (replies rarely stop abruptly).
   108→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
   109→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
   110→}

   112→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
   113→ * (identity/memory replies: whole text known upfront, exact remaining count). */
   114→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
   115→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
   116→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
   117→  // project targets ES2017, where the `s` flag is a tsc error).
   118→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
   119→  const t0 = Date.now()
   120→  for (let i = 0; i < pieces.length; i++) {
   121→    emit({ event: 'delta', text: pieces[i] })
   122→    const remainingPieces = pieces.length - i - 1
   123→    if (remainingPieces <= 0) break
   124→    const remainingBudget = budgetMs - (Date.now() - t0)
   125→    if (remainingBudget <= 0) {
   126→      await sleep(PIECE_DELAY_MIN_MS)
   127→      continue
   128→    }
   129→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
   130→  }
   131→}

   270→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   271→  const s = normalizeAr(text)
   272→  const isGym = businessType === 'gym'
   273→
   274→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   275→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   276→  }
   277→  // Opinion/decision-seeking → ADVISORY free-chat (the real-employee mode).
   278→  // Must beat the price-query rule below so «رأيك أزود سعر الكمون؟» is a
   279→  // conversation about the decision, never a bare catalog scan.
   280→  if (/(رايك|رأيك|ايه رايك|تنصحني|نصيحه|اعمل ايه لو|تشوف ايه|رايك ايه|استشاره)/.test(s) && !/(بعت|بيع|سجل بيع|تحصيل|اشتري)/.test(s)) {
   281→    return { intent: 'chat', fields: {}, replyHint: 'استشارة شغل' }
   282→  }
   283→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   284→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   285→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   286→  }
   287→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   288→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   289→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   290→  if (/(مديون|اللي ليا|تحصيلات|مين عليها|عليها فلوس|عليهم فلوس|العملاء)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   291→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   292→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   293→  // Price query («ايه سعر عود كمبودي؟» / «بكام المسك؟») → catalog search (R0 read)
   294→  if (/(سعر|بكام)/.test(s) && !/(بعت|بيع|بع |خرج|سلف|دفع|سجل)/.test(s)) {
   295→    const q = s
   296→      .replace(/^.*?(?:سعر|بكام)\s*/, '')
   297→      .replace(/[؟?].*$/, '')
   298→      .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   299→      .replace(/^\s*(?:في|من)\s*/, '')
   300→      .trim()
   301→    if (q.length >= 2) return { intent: 'read', tool: 'search_catalog', fields: { query: q.slice(0, 40) } }
   302→  }
   303→  if (/حضور/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   304→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   305→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   306→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   307→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   308→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   309→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   310→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   311→    if (name && (isQuestion || /عضويه/.test(s))) {
   312→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   313→    }
   314→  }
   315→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   316→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   317→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   318→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   319→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   320→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   321→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   322→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   323→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   324→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   325→  // fire a collection (caught by the expiring rule above instead).
   326→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   327→    return { intent: 'write', opType: 'collection', fields: {} }
   328→  }
   329→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   330→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   331→  return { intent: 'chat', fields: {} }
   332→}

   362→/** Parse the LLM JSON contract: single intent OR the "intents" array (multi). */
   363→function parseIntentJsonMulti(raw: string): { intents: RawIntentJson[]; single: ExtractedIntent | null } {
   364→  const cleaned = raw.replace(/```json|```/g, '').trim()
   365→  const start = cleaned.indexOf('{')
   366→  const end = cleaned.lastIndexOf('}')
   367→  if (start < 0 || end <= start) return { intents: [], single: null }
   368→  try {
   369→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent> & { intents?: RawIntentJson[] }
   370→    if (Array.isArray(json.intents)) {
   371→      const valid = json.intents.filter((r) => r && typeof r.intent === 'string')
   372→      if (valid.length >= 2) return { intents: valid, single: null }
   373→      if (valid.length === 1) return { intents: [], single: rawToSingle(valid[0]) }
   374→      return { intents: [], single: null }
   375→    }
   376→    if (!json.intent) return { intents: [], single: null }
   377→    return { intents: [], single: rawToSingle(json as RawIntentJson) }
   378→  } catch {
   379→    return { intents: [], single: null }
   380→  }
   381→}

   396→/** Read-request trigger keywords → tool (used by the deterministic splitter). */
   397→const READ_SPLITTERS: [RegExp, string][] = [
   398→  [/ملخص|حال اليوم|كويس اليوم|عملت ايه النهارده|اقفل الورديه/, 'get_daily_summary'],
   399→  [/وضع الخزنه|رصيد الخزنه|الخزنه|الكاش/, 'get_cash_position'],
   400→  [/النواقص|المخزون الناقص|الناقص|المخزون|استوك/, 'list_low_stock'],
   401→  [/صلاحيه?|منتهي|قربت تخلص|تنتهي/, 'list_expiring'],
   402→  [/مين عليها|عليها فلوس|عليهم فلوس|مديون|تحصيلات|اللي ليا/, 'list_receivables'],
   403→  [/اللي علينا|للموردين|الموردين|مستحقات/, 'list_payables'],
   404→  [/آخر الحركات|اخر الحركات|سجل العمليات|آخر العمليات|اخر العمليات|العمليات المستنيه/, 'get_activity_history'],
   405→  [/سعر|بكام/, 'search_catalog'],
   406→]

   408→/** Sequence verbs that START a new request mid-message (never paid-clauses). */
   409→const WRITE_CUT_RE = /بعت|اشتريت|اشتري|سجل|حصلت|قبضت|استلمت|سدد|صرفت|مصروف|تسويه|عكسها|اعكس/
   410→
   411→/**
   412→ * Deterministic compound splitter — «ملخص اليوم ومين عليها فلوس وايه الناقص
   413→ * في المخزون» → 3 read sub-intents. Cuts at read-trigger keywords, sequence
   414→ * verbs (بعت/سجل/…) and connectors (و بعدين/كمان) — never inside a multi-item
   415→ * basket («بعت ٢ لبن و ١٠ بيض» stays ONE write). Zero LLM.
   416→ */
   417→function splitSubIntents(text: string, businessType: string): SubIntent[] {
   418→  const norm = toWesternDigits(normalizeAr(text))
   419→  if (norm.length < 8) return []
   420→  type Hit = { at: number; len: number; tool: string | null }
   421→  const hits: Hit[] = []
   422→  for (const [re, tool] of READ_SPLITTERS) {
   423→    const g = new RegExp(re.source, 'g')
   424→    let m: RegExpExecArray | null
   425→    while ((m = g.exec(norm)) != null) {
   426→      hits.push({ at: m.index, len: m[0].length, tool })
   427→      if (m.index === g.lastIndex) g.lastIndex++
   428→    }
   429→  }
   430→  // sequence verbs — only when a و/، actually precedes them (mid-message cut)
   431→  {
   432→    const g = new RegExp(WRITE_CUT_RE.source, 'g')
   433→    let m: RegExpExecArray | null
   434→    while ((m = g.exec(norm)) != null) {
   435→      const at = m.index
   436→      const prev = at > 0 ? norm[at - 1] : ''
   437→      const prev2 = at > 1 ? norm[at - 2] : ''
   438→      const prev3 = at > 2 ? norm[at - 3] : ''
   439→      const isBoundary =
   440→        prev === 'و' || prev === '،' || prev === ',' || prev === '؛' || (prev === ' ' && (prev2 === 'و' || prev2 === '،')) || (prev === ' ' && prev2 === ' ' && prev3 === 'و')
   441→      if (isBoundary) hits.push({ at, len: m[0].length, tool: null })
   442→      if (m.index === g.lastIndex) g.lastIndex++
   443→    }
   444→  }
   445→  // connectors «و بعدين» — cut at the connector itself
   446→  {
   447→    const g = /(بعدين|بعد كده|كمان)/g
   448→    let m: RegExpExecArray | null
   449→    while ((m = g.exec(norm)) != null) {
   450→      hits.push({ at: m.index, len: m[0].length, tool: null })
   451→      if (m.index === g.lastIndex) g.lastIndex++
   452→    }
   453→  }
   454→  hits.sort((a, b) => a.at - b.at)
   455→  if (hits.length === 0) return []
   456→
   457→  // dedupe overlaps + same-tool continuations, build cut points
   458→  const cuts: { at: number; tool: string | null }[] = []
   459→  let coveredEnd = -1
   460→  let currentTool: string | null = null
   461→  for (const h of hits) {
   462→    if (h.at < coveredEnd) continue // overlapping keyword (الخزنه inside وضع الخزنه)
   463→    if (h.tool != null && h.tool === currentTool) {
   464→      // same tool still talking — «سعر العنبر بكام؟» is ONE search, and
   465→      // «سعر العود وبكام المسك» is one question about two items (the catalog
   466→      // matcher is bidirectional-includes, so both still show in one card)
   467→      coveredEnd = Math.max(coveredEnd, h.at + h.len)
   468→      continue
   469→    }
   470→    // cut position: include a glued و in the PREVIOUS segment
   471→    const cutAt = h.at > 0 && norm[h.at - 1] === 'و' ? h.at - 1 : h.at
   472→    cuts.push({ at: cutAt, tool: h.tool })
   473→    coveredEnd = h.at + h.len
   474→    currentTool = h.tool
   475→  }
   476→  if (cuts.length === 0) return []
   477→
   478→  // build segments
   479→  const segments: { text: string; tool: string | null }[] = []
   480→  const seg0 = norm.slice(0, cuts[0].at).replace(/[\s،,؛]+$/, '').trim()
   481→  if (seg0.length > 0) segments.push({ text: seg0, tool: null })
   482→  for (let i = 0; i < cuts.length; i++) {
   483→    const end = i + 1 < cuts.length ? cuts[i + 1].at : norm.length
   484→    const seg = norm.slice(cuts[i].at, end).replace(/^[\s،,؛و]+/, '').replace(/[\s،,؛]+$/, '').trim()
   485→    if (seg.length > 0) segments.push({ text: seg, tool: cuts[i].tool })
   486→  }
   487→
   488→  // map segments → sub-intents
   489→  const subs: SubIntent[] = []
   490→  for (let i = 0; i < segments.length; i++) {
   491→    const seg = segments[i]
   492→    // politeness prefixes («قوللي», «اط») — skip chat-y tiny segments
   493→    if (seg.tool == null) {
   494→      if (i === 0 && seg.text.length <= 6) continue
   495→      const mapped = ruleBasedIntent(seg.text, businessType)
   496→      if (mapped.intent === 'read' && mapped.tool) subs.push({ ...mapped, intent: 'read', tool: mapped.tool, sourceText: seg.text })
   497→      else if (mapped.intent === 'write' && mapped.opType) subs.push({ ...mapped, intent: 'write', sourceText: seg.text })
   498→      continue
   499→    }
   500→    // tool-owned segment
   501→    if (seg.tool === 'search_catalog') {
   502→      const q = seg.text
   503→        .replace(/^(?:ايه|ايه هو|هو)?\s*(?:سعر|بكام)\s*/, '')
   504→        .replace(/[؟?].*$/, '')
   505→        .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   506→        .replace(/^\s*(?:في|من)\s*/, '')
   507→        .trim()
   508→      subs.push({ intent: 'read', tool: 'search_catalog', fields: q.length >= 2 ? { query: q.slice(0, 40) } : {}, sourceText: seg.text })
   509→    } else {
   510→      subs.push({ intent: 'read', tool: seg.tool, fields: {}, sourceText: seg.text })
   511→    }
   512→  }
   513→
   514→  // merge CONSECUTIVE same-opType writes into one basket (multi-item scan)
   515→  const merged: SubIntent[] = []
   516→  for (const s of subs) {
   517→    const last = merged[merged.length - 1]
   518→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   519→      last.sourceText = `${last.sourceText} و ${s.sourceText}`
   520→      last.fields = { ...last.fields, ...s.fields }
   521→      continue
   522→    }
   523→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   524→      continue // duplicate read
   525→    }
   526→    merged.push({ ...s })
   527→  }
   528→  return merged
   529→}

   531→/** Normalize LLM multi-intents: validate, merge consecutive same-opType writes. */
   532→function normalizeMultiIntents(raws: RawIntentJson[], fullText: string): SubIntent[] {
   533→  const subs: SubIntent[] = []
   534→  for (const r of raws) {
   535→    if (r.intent === 'read' && r.tool && READ_TOOLS.has(r.tool)) {
   536→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   537→      subs.push({ intent: 'read', tool: r.tool as ReadTool, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   538→    } else if (r.intent === 'write' && r.opType && VALID_OP_TYPES.has(r.opType)) {
   539→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   540→      subs.push({ intent: 'write', opType: r.opType as OpType, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   541→    }
   542→  }
   543→  const merged: SubIntent[] = []
   544→  for (const s of subs) {
   545→    const last = merged[merged.length - 1]
   546→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   547→      // consecutive same-type writes = ONE basket; full-text scan finds the lines
   548→      last.sourceText = fullText
   549→      last.fields = { ...last.fields, ...s.fields }
   550→      continue
   551→    }
   552→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   553→      continue // duplicate read
   554→    }
   555→    merged.push(s)
   556→  }
   557→  return merged
   558→}

   564→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   565→  for (const c of cards) {
   566→    if (c.kind !== 'table' || !c.footer) continue
   567→    for (const f of c.footer) {
   568→      const m = String(f.value).match(/[\d٠-٩]+/)
   569→      if (m) {
   570→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   571→        const n = Number(west)
   572→        if (Number.isFinite(n)) return n
   573→      }
   574→    }
   575→  }
   576→  return null
   577→}

   597→/**
   598→ * Deterministic reply for read turns when the LLM reply stream fails
   599→ * (rate-limit / network). NEVER generic — quotes the REAL numbers the tools
   600→ * just fetched, so the owner still gets a useful answer with the data below.

   602→function deterministicReadReply(cards: ResultCardDTO[]): string {
   603→  for (const c of cards) {
   604→    if (c.kind === 'kv' && c.rows.length > 0) {
   605→      const parts = c.rows.slice(0, 4).map((r) => `${r.label} ${r.value}`).join(' · ')

   607→    }
   608→  }
   609→  const table = cards.find((c) => c.kind === 'table') as Extract<ResultCardDTO, { kind: 'table' }> | undefined

   614→  const text = cards.find((c) => c.kind === 'text') as Extract<ResultCardDTO, { kind: 'text' }> | undefined

   619→/** Deterministic numbered multi-intent reply — answers EVERY sub-request, no LLM. */
   620→function deterministicMultiReply(parts: string[], overflow: number): string {

   657→/**
   658→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   659→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   660→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   661→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   662→ * customer — the ل/من marker is what marks the ROLE.
   663→ */
   664→function findPartyMention<T extends { name: string }>(
   665→  text: string,
   666→  parties: T[],
   667→  marker: 'ل' | 'من' = 'ل',
   668→): T | null {
   669→  const norm = normalizeAr(text)
   670→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   671→  for (const p of parties) {
   672→    const words = normalizeAr(p.name)
   673→      .split(' ')
   674→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   675→      .filter((w) => w.length >= 2)
   676→    for (const w of words) {
   677→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   678→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   679→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   680→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   681→    }
   682→  }
   683→  return null
   684→}

   701→/** عطارة «البيع بالمبلغ»: «بـ20 جنيه فلفل أسود» → (amount, in EGP) or null.
   702→ * Runs on WESTERN-digit normalized text (normalizeAr already ate the ـ tatweel
   703→ * in «بـ20»). Requires a real money word (جنيه/ج) after the number so a bare
   704→ * quantity («بعت 20 فلفل») can never fire it. */
   705→function matchReverseSellAmount(normWestern: string): number | null {
   706→  const m = normWestern.match(/(?:^|\s)(?:بال|بمبلغ|ب)([0-9]+(?:[.,][0-9]+)?)\s*(?:جنيه|جنيهات|ج)(?=\s|$)/)
   707→  if (!m) return null
   708→  const amount = parseFloat(m[1].replace(',', '.'))
   709→  return Number.isFinite(amount) && amount > 0 ? amount : null
   710→}

   712→/** Remove the «بـN جنيه» clause so the money number can never masquerade as a
   713→ * weight in ANY quantity parser downstream (segment scan / parseQuantity). */
   714→function stripReverseSellAmount(normWestern: string): string {
   715→  return normWestern.replace(/(?:^|\s)(?:بال|بمبلغ|ب)[0-9]+(?:[.,][0-9]+)?\s*(?:جنيه|جنيهات|ج)(?=\s|$)/g, ' ')
   716→}

   718→/** Paid amount: LLM field → explicit «دفع 20» clause → آجل = 0 → default cash (total). */
   719→function resolvePaid(norm: string, f: Record<string, unknown>, total: number): number {
   720→  if (typeof f.paid === 'number' && f.paid > 0) return Math.min(Math.round(f.paid * 100), total)
   721→  const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   722→  if (paidWord?.[1]) {
   723→    const n = parseQuantity(paidWord[1], null).qty
   724→    if (n != null && n > 0) return Math.min(Math.round(n * 100), total)
   725→  }
   726→  if (/(?:بال)?اجل|علي الحساب|نسيون/.test(norm)) return 0
   727→  return total
   728→}

   730→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit, memCtx: MemoryContext): Promise<ResolvedWrite> {
   731→  const { business, user } = ctx
   732→  const type = (intent.opType ?? 'sale') as OpType
   733→  const f = intent.fields ?? {}
   734→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   735→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   736→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   737→
   738→  const text = ctx.text
   739→  const norm = normalizeAr(text)
   740→  const normW = toWesternDigits(norm)
   741→
   742→  switch (type) {
   743→    case 'sale':
   744→    case 'purchase': {
   745→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   746→      // «دفع ٢٠ بس» can never be misread as the sold quantity.
   747→      const textNoPaid = norm.replace(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?[٠-٩0-9][٠-٩0-9.,]*\s*(?:بس|فقط|تمام)?/g, ' ')
   748→      // ---- عطارة reverse-selling («بعت بـ20 جنيه فلفل») ----
   749→      // The money clause MUST be detected and stripped BEFORE any quantity
   750→      // parsing: a bare «20» in «بـ20 جنيه» is the AMOUNT, not a weight. The
   751→      // deterministic scanners and the LLM both tend to read it as qty=20 —
   752→      // the pattern match on «بـN جنيه» is the ground truth.
   753→      const reverseAmountEgp = business.type === 'attar' && type === 'sale' ? matchReverseSellAmount(normW) : null
   754→      const scanText = reverseAmountEgp != null ? stripReverseSellAmount(toWesternDigits(textNoPaid)) : textNoPaid
   755→      // ---- Multi-item scan (deterministic — works with or without the LLM) ----
   756→      // «سجّل بيع ٢ لبن و ١٠ بيض» → segments → per-item quantity. When TWO or
   757→      // more distinct catalog items are mentioned we draft ONE multi-line
   758→      // operation (a single basket the cashier approves once), like the real POS.
   759→      // Longest-alias-first + overlap rejection: «بانادول» alone can never fan
   760→      // out into «بانادول أكتيف» + «بانادول شرائط» — the more specific name wins.
   761→      const segments = scanText.split(/\s*(?:،|,|\+|؛|;)\s*|\s+و\s+|\s+و(?=[٠-٩0-9])/).filter(Boolean)
   762→      const byItem = new Map<string, { item: Item; qty: number | null; unit: string | null }>()
   763→      for (const seg of segments) {
   764→        const segNorm = normalizeAr(seg)
   765→        const hits: { item: Item; token: string; at: number }[] = []
   766→        for (const it of items) {
   767→          // canonical names + column aliases + LEXICON dialect aliases (boost)
   768→          const lexAl = memCtx.lexicon.filter((l) => l.entityRef === it.id).map((l) => l.alias)
   769→          const names = [it.name, ...(it.aliases ? it.aliases.split(',').map((a) => a.trim()) : []), ...lexAl]
   770→          let best: { token: string; at: number } | null = null
   771→          for (const n of names) {
   772→            const nn = normalizeAr(n)
   773→            if (nn.length >= 3) {
   774→              const at = segNorm.indexOf(nn)
   775→              if (at >= 0 && (!best || nn.length > best.token.length)) best = { token: nn, at }
   776→            }
   777→          }
   778→          if (best) hits.push({ item: it, ...best })
   779→        }
   780→        hits.sort((a, b) => b.token.length - a.token.length)
   781→        const taken: Array<[number, number]> = []
   782→        for (const h of hits) {
   783→          if (taken.some(([s, e]) => h.at < e && h.at + h.token.length > s)) continue
   784→          taken.push([h.at, h.at + h.token.length])
   785→          const q = parseQuantity(seg, null)
   786→          const prev = byItem.get(h.item.id)
   787→          byItem.set(h.item.id, {
   788→            item: h.item,
   789→            qty: q.qty != null && q.qty > 0 ? (prev?.qty ?? 0) + q.qty : prev?.qty ?? null,
   790→            unit: prev?.unit ?? unitToCode(q.unit) ?? null,
   791→          })
   792→        }
   793→      }
   794→      const multi = byItem.size >= 2
   795→
   796→      if (multi) {
   797→        // ---- Multi-line draft: one basket, real prices per line ----
   798→        const lines: { item: Item; qty: number; unitPrice: number; total: number }[] = []
   799→        const issues: string[] = []
   800→        for (const { item, qty, unit } of byItem.values()) {
   801→          const rawUnit = unit ?? item.unit // default = the item's own canonical unit
   802→          const baseQty = convertToBaseUnit(qty ?? 1, rawUnit, item.unit)
   803→          if (baseQty == null || !(baseQty > 0)) {
   804→            issues.push(`الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})`)
   805→            continue
   806→          }
   807→          const qtyN = Math.round(baseQty * 1000) / 1000
   808→          if (type === 'sale' && qtyN > item.stockQty) {
   809→            issues.push(`المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}`)
   810→            continue
   811→          }
   812→          const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   813→          lines.push({ item, qty: qtyN, unitPrice, total: Math.round(qtyN * unitPrice) })
   814→        }
   815→        if (lines.length === 0) {
   816→          return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: issues.join('؛ ') || 'الكمية قد إيه لكل صنف؟', issues }
   817→        }
   818→        const total = lines.reduce((s, l) => s + l.total, 0)
   819→        // Paid (clamped to the basket total; آجل = 0)
   820→        const paid = resolvePaid(norm, f, total)
   821→        // Party (lexicon first, then LLM field, then deterministic phrasing)
   822→        const lexCustomer = type === 'sale' ? lexiconResolve(norm, memCtx.lexicon, customers)?.entity : null
   823→        if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
   824→        const customer = lexCustomer ?? (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   825→        const supplier = (f.supplierName ? matchEntity(String(f.supplierName), suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   826→
   827→        const payload: Record<string, unknown> = {
   828→          lines: lines.map((l) => ({ itemId: l.item.id, name: l.item.name, qty: l.qty, unit: l.item.unit, unitPrice: l.unitPrice, total: l.total })),
   829→          total,
   830→          paid,
   831→          ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   832→          ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   833→        }
   834→        const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   835→        const risk = computeRisk(type, total)
   836→        const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${lines.map((l) => `${fmtQty(l.qty, arUnit(l.item.unit))} ${l.item.name}`).join(' + ')} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   837→        // BUSINESS_DNA credit-limit enforcement (آجل on a capped party)
   838→        let dangerNote: string | undefined
   839→        let creditField: { label: string; value: string; hint?: string } | undefined
   840→        if (type === 'sale' && customer && total - paid > 0 && memCtx.rules.length > 0) {
   841→          const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
   842→          if (check) {
   843→            dangerNote = check.warning
   844→            creditField = { label: '⚠️ قانون المحل', value: `هيوصل رصيد ${customer.name} لـ${egp(check.projectedPiastres)} — القانون: ${egp(check.limitPiastres)}` }
   845→            memCtx.referencedIds.push(check.rule.id)
   846→          }
   847→        }
   848→        const card = fieldsToCard(ref, type, risk, summary, [
   849→          ...lines.map((l) => ({ label: l.item.name, value: `${fmtQty(l.qty, arUnit(l.item.unit))} × ${egp(l.unitPrice)} = ${egp(l.total)}` })),
   850→          { label: 'الإجمالي', value: egp(total) },
   851→          { label: 'مدفوع', value: egp(paid) },
   852→          ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   853→          ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   854→          ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   855→          ...(creditField ? [creditField] : []),
   856→          ...issues.map((i) => ({ label: '⚠ مش ضمن المسودة', value: i })),
   857→        ], [
   858→          ...lines.map((l) => ({ label: `مخزون ${l.item.name}`, value: `${fmtQty(l.item.stockQty, arUnit(l.item.unit))} ← ${fmtQty(l.item.stockQty + (type === 'sale' ? -l.qty : l.qty), arUnit(l.item.unit))}`, tone: 'neutral' as const })),
   859→          { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' as const : 'bad' as const },
   860→        ], dangerNote ?? RISK_META[risk].hint)
   861→        return { ok: true, opType: type, payload, summary, card, issues: issues.length > 0 ? issues : undefined }
   862→      }
   863→
   864→      // ---- Single-item draft (original path) ----

   866→      const rawItemName = String(f.itemName ?? '')

   868→      const lexHit = lexiconResolve(norm, memCtx.lexicon, items)
   869→      const item = lexHit?.entity ?? (rawItemName ? matchEntity(rawItemName, items)?.entity : null) ?? mentioned?.item ?? matchEntity(text, items)?.entity ?? null
   870→      if (lexHit) {
   871→        memCtx.referencedIds.push(lexHit.entry.id)

   874→        const alias = learnableAlias(text, item)
   875→        if (alias) await saveLexiconMemory({ businessId: business.id, userId: user.id }, alias, item)
   876→      }

   879→      }
   880→
   881→      // ---- عطارة reverse selling: «بعت بـ20 جنيه فلفل أسود» → وزن بالجرام ----
   882→      if (reverseAmountEgp != null) {
   883→        // scanText already has the money clause stripped, so ANY number still
   884→        // left in it is a real weight («بعت 100 جرام كركديه بـ20 جنيه» → qty wins)

   886→        const digitsAfterStrip = /[0-9]/.test(scanText)

   890→          typeof f.qty === 'number' && f.qty > 0 && f.qty !== reverseAmountEgp && (digitsAfterStrip || (f.unit && unitToCode(String(f.unit)) != null))

   898→          const amountPiastres = Math.round(amountEgp * 100)
   899→          if (perGram > 0 && amountPiastres > 0) {

   902→              return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `${egp(amountPiastres)} أقل من سعر الجرام (${egp(perGram)}) — الزبون ده عايز ياخد حاجة تانية؟` }

   908→            const total = amountPiastres
   909→            const paid = resolvePaid(norm, f, total)

   912→              itemId: item.id,
   913→              itemName: item.name,
   914→              qty: grams,
   915→              unit: 'gram',
   916→              unitPrice: perGram,

   919→              reverseSell: true,
   920→              requestedAmount: amountPiastres,

   922→            }
   923→            const risk = computeRisk(type, total)
   924→            const summary = `بيع بالوزن: ${fmtQty(grams, 'جرام')} ${item.name} بمبلغ ${egp(total)}${customer ? ` — ${customer.name}` : ''}`
   925→            let dangerNote: string | undefined
   926→            let creditField: { label: string; value: string; hint?: string } | undefined
   927→            if (customer && total - paid > 0 && memCtx.rules.length > 0) {

  1018→        ...(creditField ? [creditField] : []),
  1019→      ], [
  1020→        { label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + (type === 'sale' ? -qtyN : qtyN), arUnit(item.unit))}`, tone: 'neutral' },
  1021→        { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' : 'bad' },
  1022→      ], dangerNote ?? RISK_META[risk].hint)
  1023→      return { ok: true, opType: type, payload, summary, card }
  1024→    }
  1025→
  1026→    case 'expense': {
  1027→      const amountRaw = f.amount as number | undefined
  1028→      let amount = amountRaw && amountRaw > 0 ? Math.round(amountRaw * 100) : null
  1029→      if (amount == null) {
  1030→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1031→        if (m) {
  1032→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d))).replace(',', '.')
  1033→          const parsed = parseFloat(western)
  1034→          if (!Number.isNaN(parsed)) amount = Math.round(parsed * 100)
  1035→        }
  1036→      }
  1037→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: 'المبلغ قد إيه؟' }
  1038→      const label = String(f.label ?? '') || (norm.includes('كهربا') ? 'كهربا' : norm.includes('ايجار') ? 'إيجار' : norm.includes('تليفون') ? 'تليفون' : 'مصروف')
  1039→      const payload = { label, total: amount }
  1040→      const risk = computeRisk(type, amount)
  1041→      const summary = `مصروف ${label} بـ${egp(amount)}`
  1042→      const card = fieldsToCard('', type, risk, summary, [
  1043→        { label: 'البند', value: label },
  1044→        { label: 'المبلغ', value: egp(amount) },
  1045→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
  1046→      return { ok: true, opType: type, payload, summary, card }
  1047→    }
  1048→
  1049→    case 'collection': {
  1050→      const customersAll = customers.length > 0 ? customers : []
  1051→      const customerName = String(f.customerName ?? '')
  1052→      const lexCustomer = lexiconResolve(norm, memCtx.lexicon, customersAll)?.entity ?? null
  1053→      if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
  1054→      const customer = lexCustomer ?? matchEntity(customerName || text, customersAll)?.entity ?? null
  1055→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العميل', failMessage: 'تحصيل من مين بالظبط؟' }
  1056→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
  1057→      if (amount == null) {
  1058→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1059→        if (m) {
  1060→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
  1061→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
  1062→        }
  1063→      }
  1064→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هتحصّل قد إيه من ${customer.name}؟` }
  1065→      if (amount > customer.balance) {
  1066→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `${customer.name} عليه ${egp(customer.balance)} بس — مش ممكن تحصّل ${egp(amount)}` }
  1067→      }
  1068→      const payload = { customerId: customer.id, customerName: customer.name, amount }
  1069→      const risk = computeRisk(type, amount)
  1070→      const summary = `تحصيل ${egp(amount)} من ${customer.name}`
  1071→      const card = fieldsToCard('', type, risk, summary, [
  1072→        { label: 'العميل', value: customer.name },
  1073→        { label: 'المبلغ', value: egp(amount) },
  1074→        { label: 'المديونية الحالية', value: egp(customer.balance) },
  1075→        { label: 'بعد التحصيل', value: egp(customer.balance - amount) },
  1076→      ], [{ label: 'الخزنة', value: egp(amount), tone: 'good' }], RISK_META[risk].hint)
  1077→      return { ok: true, opType: type, payload, summary, card }
  1078→    }
  1079→
  1080→    case 'payment': {
  1081→      const supplierName = String(f.supplierName ?? '')
  1082→      const supplier = matchEntity(supplierName || text, suppliers)?.entity ?? null
  1083→      if (!supplier) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المورد', failMessage: 'هسدد لمين؟' }
  1084→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
  1085→      if (amount == null) {
  1086→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1087→        if (m) {
  1088→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
  1089→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
  1090→        }
  1091→      }
  1092→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هسدد قد إيه لـ${supplier.name}؟` }
  1093→      if (amount > supplier.balance) {
  1094→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المستحق لـ${supplier.name} ${egp(supplier.balance)} بس` }
  1095→      }
  1096→      const payload = { supplierId: supplier.id, supplierName: supplier.name, amount }
  1097→      const risk = computeRisk(type, amount)
  1098→      const summary = `سداد ${egp(amount)} لـ${supplier.name}`
  1099→      const card = fieldsToCard('', type, risk, summary, [
  1100→        { label: 'المورد', value: supplier.name },
  1101→        { label: 'المبلغ', value: egp(amount) },
  1102→        { label: 'المستفق الحالي', value: egp(supplier.balance) },
  1103→        { label: 'بعد السداد', value: egp(supplier.balance - amount) },
  1104→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
  1105→      return { ok: true, opType: type, payload, summary, card }
  1106→    }
  1107→
  1108→    case 'membership': {
  1109→      const isRenewal = norm.includes('تجديد') || norm.includes('جدد')
  1110→      const explicitName = String(f.customerName ?? '').trim()
  1111→      const customer = explicitName ? matchEntity(explicitName, customers)?.entity ?? null : matchEntity(text, customers)?.entity ?? null
  1112→      if (!customer && isRenewal) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'تجديد لمين؟' }
  1113→      // A NEW subscription with NO name at all (rule path, question-like text)
  1114→      // must never draft «عضو جديد» — ask for the name first.
  1115→      if (!customer && !explicitName) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'اشتراك جديد لمين؟ اكتبلي اسم العضو' }
  1116→      const plan = String(f.plan ?? 'شهرية')
  1117→      const months = Math.max(1, Math.min(12, Math.round(Number(f.months ?? 1))))
  1118→      // plan price from items (gym plans are items) or fallback 35000
  1119→      const planItem = items.find((i) => normalizeAr(i.name).includes(normalizeAr(plan)))
  1120→      const total = Math.round((planItem?.sellPrice ?? 35000) * months)
  1121→      let paid = total
  1122→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
  1123→      paid = Math.min(paid, total)
  1124→      const name = customer?.name ?? explicitName
  1125→      const payload = {
  1126→        ...(customer ? { customerId: customer.id } : { customerName: name }),
  1127→        plan,
  1128→        months,
  1129→        total,
  1130→        paid,
  1131→      }
  1132→      const risk = computeRisk(type, total)
  1133→      const summary = `${isRenewal ? 'تجديد' : 'اشتراك جديد'} ${plan} لـ${name} — ${egp(total)}`
  1134→      const card = fieldsToCard('', type, risk, summary, [
  1135→        { label: 'العضو', value: name },
  1136→        { label: 'الخطة', value: plan },
  1137→        { label: 'المدة', value: fmtQty(months, 'شهر') },
  1138→        { label: 'الإجمالي', value: egp(total) },
  1139→        { label: 'مدفوع', value: egp(paid) },
  1140→      ], [
  1141→        { label: 'الخزنة', value: egp(paid), tone: 'good' },
  1142→        ...(total - paid > 0 ? [{ label: 'باقي على العضو', value: egp(total - paid), tone: 'bad' as const }] : []),
  1143→      ], RISK_META[risk].hint)
  1144→      return { ok: true, opType: type, payload, summary, card }
  1145→    }
  1146→
  1147→    case 'attendance': {
  1148→      const customerName = String(f.customerName ?? '')
  1149→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
  1150→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'حضور مين؟' }
  1151→      const payload = { customerId: customer.id, customerName: customer.name }
  1152→      const risk = computeRisk(type)
  1153→      const summary = `حضور ${customer.name}`
  1154→      const card = fieldsToCard('', type, risk, summary, [
  1155→        { label: 'العضو', value: customer.name },
  1156→        { label: 'اليوم', value: 'النهاردة' },
  1157→        ...(customer.expiresAt ? [{ label: 'الاشتراك لحد', value: arDate(customer.expiresAt) }] : []),
  1158→      ], [], RISK_META[risk].hint)
  1159→      return { ok: true, opType: type, payload, summary, card }
  1160→    }
  1161→
  1162→    case 'stock_adjustment': {
  1163→      const rawItemName = String(f.itemName ?? '')
  1164→      const item = matchEntity(rawItemName || text, items)?.entity ?? null
  1165→      if (!item) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: 'تسوية مخزون لأنهي صنف؟' }
  1166→      let delta = (f.delta as number | undefined) ?? null
  1167→      if (delta == null && typeof f.qty === 'number') delta = f.qty - item.stockQty
  1168→      if (delta == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية الجديدة/الفرق', failMessage: `المخزون الحالي ${fmtQty(item.stockQty, arUnit(item.unit))} — عايزه يبقى قد إيه؟` }
  1169→      const reason = String(f.reason ?? '') || 'تسوية جرد'
  1170→      const payload = { itemId: item.id, itemName: item.name, delta, reason, unit: item.unit }
  1171→      const risk = computeRisk(type)
  1172→      const summary = `تسوية مخزون ${item.name}: ${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`
  1173→      const card = fieldsToCard('', type, risk, summary, [
  1174→        { label: 'الصنف', value: item.name },
  1175→        { label: 'الحالي', value: fmtQty(item.stockQty, arUnit(item.unit)) },
  1176→        { label: 'الجديد', value: fmtQty(item.stockQty + delta, arUnit(item.unit)) },
  1177→        { label: 'السبب', value: reason },
  1178→      ], [{ label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`, tone: delta >= 0 ? 'good' : 'bad' }], 'تسوية المخزون عملية حساسة (R3) — محتاجة سبب واعتماد')
  1179→      return { ok: true, opType: type, payload, summary, card }
  1180→    }
  1181→
  1182→    case 'reversal': {
  1183→      const ref = String(f.ref ?? '') || (norm.match(/WKL-\d+/)?.[0] ?? '')
  1184→      if (!ref) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'رقم العملية', failMessage: 'عايز تعكس أنهي عملية؟ اكتب رقمها (زي WKL-1006)' }
  1185→      const original = await db.operation.findFirst({ where: { businessId: business.id, ref } })
  1186→      if (!original) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `ملقيتش عملية رقم ${ref}` }
  1187→      if (original.reversedByRef) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `العملية ${ref} معكوسة قبل كده بـ${original.reversedByRef}` }
  1188→      const reason = String(f.reason ?? '') || 'تصحيح خطأ'
  1189→      const payload = { originalRef: ref, reason }
  1190→      const risk = computeRisk(type)
  1191→      const summary = `عكس العملية ${ref} (${original.summary})`
  1192→      const card = fieldsToCard('', type, risk, summary, [
  1193→        { label: 'العملية الأصلية', value: `${ref} — ${TYPE_AR[original.type] ?? original.type}` },
  1194→        { label: 'ملخصها', value: original.summary },
  1195→        { label: 'السبب', value: reason },
  1196→      ], [{ label: 'الأثر', value: 'هيتلغى بالكامل بعملية عكسية موثقة', tone: 'neutral' }], 'عكس عملية مالية حساس (R3) — التاريخ مش بيتمسح، بيتصحح')
  1197→      return { ok: true, opType: type, payload, summary, card }
  1198→    }
  1199→
  1200→    default:
  1201→      return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: 'نوع العملية مش مدعوم' }
  1202→  }
  1203→}

  1208→export interface TurnResult {
  1209→  replyText: string

  1217→
  1218→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
  1219→  const { business, user, text } = ctx
  1220→  const v = getVertical(business.type)
  1221→  const model = await resolveModel(ctx.modelKey)
  1222→  const modelLabel = modelLabelFor(model)
  1223→
  1224→  // Load context once (memories raised to 8; pending ops count for the persona)
  1225→  const [items, customers, suppliers, memories, pendingCount] = await Promise.all([
  1226→    db.item.findMany({ where: { businessId: business.id } }),
  1227→    db.customer.findMany({ where: { businessId: business.id } }),
  1228→    db.supplier.findMany({ where: { businessId: business.id } }),
  1229→    db.agentMemory.findMany({ where: { businessId: business.id, userId: user.id, active: true }, orderBy: { createdAt: 'desc' }, take: 12 }),
  1230→    db.operation.count({ where: { businessId: business.id, status: 'AWAITING_APPROVAL' } }),
  1231→  ])
  1232→  const rules = memories.filter((m) => m.kind === 'rule')
  1233→  const memCtx: MemoryContext = {
  1234→    lexicon: parseLexicon(memories),
  1235→    rules,
  1236→    referencedIds: [],
  1237→  }
  1238→  // Prompt memories: preferences + facts (rules are injected in their own block)
  1239→  const promptMemories = memories.filter((m) => m.kind === 'preference' || m.kind === 'fact')
  1240→
  1241→  // Thinking steps — with a «فتكرت من ذاكرتي» recall step injected right after
  1242→  // «بفهم طلبك» whenever the agent actually remembers something about this user.
  1243→  const memStep: ThinkingStep[] =
  1244→    memories.length > 0
  1245→      ? [
  1246→          {
  1247→            label: 'فتكرت من ذاكرتي',
  1248→            detail: memories[0].text.slice(0, 60) + (memories[0].text.length > 60 ? '…' : ''),
  1249→            state: 'done',
  1250→            icon: 'verify',
  1251→          },
  1252→        ]
  1253→      : []
  1254→  const steps: ThinkingStep[] = [
  1255→    { label: 'بفهم طلبك', detail: 'بحلل الجملة وأستخرج القصد والحقول', state: 'active', icon: 'understand' },
  1256→    ...memStep,
  1257→    { label: 'بدوّر في سجلك', detail: 'بدوّر على الأصناف والأسماء الحقيقية', state: 'pending', icon: 'search' },
  1258→    { label: 'بتحقق من الأرقام', detail: 'بحسب من الأسعار الفعلية مش من دماغي', state: 'pending', icon: 'verify' },
  1259→    { label: 'بجهّز الرد', detail: '', state: 'pending', icon: 'plan' },
  1260→  ]
  1261→  const IDX = { understand: 0, search: 1 + memStep.length, verify: 2 + memStep.length, reply: 3 + memStep.length }
  1262→  const pushThinking = () => emit({ event: 'thinking', steps: steps.map((s) => ({ ...s })) })
  1263→  const done = (i: number, detail?: string) => {
  1264→    if (steps[i]) {
  1265→      steps[i].state = 'done'
  1266→      if (detail) steps[i].detail = detail
  1267→    }
  1268→    pushThinking()
  1269→  }
  1270→  pushThinking()
  1271→
  1272→  const norm = normalizeAr(text)
  1273→
  1274→  // ---- PASS 0a: Identity question → instant, deterministic, zero LLM ----
  1275→  if (isIdentityQuestion(norm)) {
  1276→    const lowStockCount = items.filter((i) => i.stockQty <= i.lowStockThreshold).length
  1277→    const receivablesTotal = customers.reduce((s, c) => s + Math.max(0, c.balance), 0)
  1278→    done(IDX.understand, 'سؤال عن هويتي — بجاوب من سجلك من غير موديل')
  1279→    done(IDX.search, 'جبت بياناتي من سجلك الحقيقي')
  1280→    done(IDX.verify, 'عدّيت النواقص والعمليات المستنية من قاعدة البيانات')
  1281→    emit({ event: 'step', label: 'بتعرّف بنفسي', state: 'done' })
  1282→    done(IDX.reply)
  1283→    const identityMemory = memories.find((m) => m.kind === 'preference' || m.kind === 'rule' || m.kind === 'fact') ?? null
  1284→    if (identityMemory) memCtx.referencedIds.push(identityMemory.id)
  1285→    const replyText = buildIdentityReply({
  1286→      business,
  1287→      user,
  1288→      verticalLabel: v.label,
  1289→      memoryText: identityMemory?.text ?? null,
  1290→      lowStockCount,
  1291→      pendingCount,
  1292→      receivablesTotal,
  1293→    })
  1294→    const actions: ChatActionDTO[] = [
  1295→      { id: 'act-today', label: 'لوحة اليوم', view: 'today', icon: 'today', tone: 'primary' },
  1296→      ...(lowStockCount > 0
  1297→        ? [{ id: 'act-inventory', label: 'شوف المخزون', view: 'inventory' as const, icon: 'boxes' as const, tone: 'outline' as const }]
  1298→        : []),
  1299→    ]
  1300→    await streamDeterministic(replyText, emit, 1200)
  1301→    emit({ event: 'actions', actions })
  1302→    await trackUsage(business.id, user.id, 'chat', 'chat')
  1303→    await bumpMemoryUsage(memCtx.referencedIds)
  1304→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions }
  1305→  }
  1306→
  1307→  // ---- PASS 0b: Memory save (preference OR BUSINESS_DNA rule) → instant, persisted, zero LLM ----
  1308→  const saveRequest = extractSaveRequest(text, norm) ?? extractRuleRequest(text, norm)
  1309→  if (saveRequest) {
  1310→    // Dedupe: re-saving the exact same memory must NOT clutter the memory
  1311→    // list — the owner repeating himself deserves a calm «already saved».
  1312→    const isDuplicate = memories.some((m) => normalizeAr(m.text) === normalizeAr(saveRequest.text))
  1313→    if (!isDuplicate) {
  1314→      await db.agentMemory.create({
  1315→        data: {
  1316→          businessId: business.id,
  1317→          userId: user.id,
  1318→          kind: saveRequest.kind,
  1319→          category: saveRequest.category,
  1320→          text: saveRequest.text,
  1321→          confidence: saveRequest.confidence,
  1322→        },
  1323→      })
  1324→    }
  1325→    const isRule = saveRequest.kind === 'rule'
  1326→    done(IDX.understand, isRule ? 'قانون من صاحب الشغل — مش محتاج موديل' : 'طلب حفظ في الذاكرة — مش محتاج موديل')
  1327→    done(IDX.search, 'بتخزنها في ذاكرتي عنك')
  1328→    done(IDX.verify, isDuplicate ? 'موجودة أصلًا — من غير تكرار' : 'اتحفظت فعليًا في قاعدة البيانات')
  1329→    emit({ event: 'step', label: 'بحفظ في ذاكرتي', state: 'done' })
  1330→    done(IDX.reply)
  1331→    const replyText = isDuplicate
  1332→      ? `دي محفوظة عندي أصلًا ✅ — «${saveRequest.text}». مطمن، مش هنسيها.`
  1333→      : isRule
  1334→        ? `اتحفظت قانون من قوانين المحل ✅ — «${saveRequest.text}».\nمن دلوقتي أي عملية تخص الموضوع ده هتتنبه عليه قبل الاعتماد.`
  1335→        : `اتحفظت في ذاكرتي ✅ — هفتكرها: «${saveRequest.text}». لما الموقف ده يتكرر هرفعها على طول.`

  1644→    if (subs.length >= 2 && multiReplyParts && !replyAnswersAllParts(replyText, Math.min(subs.length, 4))) {
  1645→      replyText = deterministicMultiReply(multiReplyParts, multiOverflow)
  1646→    }
  1647→  } catch (err) {
  1648→    console.error('[harness] reply stream failed:', err)
  1649→    if (!replyText) {
  1650→      const fallback = advisoryMode
  1651→        ? advisoryFallback(v)
  1652→        : multiReplyParts
  1653→          ? deterministicMultiReply(multiReplyParts, multiOverflow)
  1654→          : card
  1655→            ? `جهزتلك مسودة «${card.title}» — ${card.summary}. راجعها واعتمدها لو تمام.${draftIssues ? ` بس خد بالك: ${draftIssues.join('؛ ')}.` : ''}`
  1656→            : cards
  1657→              ? deterministicReadReply(cards)
  1658→              : questionMode
  1659→                ? (situation.match(/— (.+)$/)?.[1] ?? 'محتاج تفاصيل أكتر لو سمحت.')
  1660→                : permissionDenied
  1661→                  // The cashier must NEVER read «تمام» on a denied operation —
  1662→                  // that reads like «it's done». Say the permission truth instead.
  1663→                  ? permissionMsg
  1664→                  : chatFallbackWithChips(v)
  1665→      await streamDeterministic(fallback, emit, 900)
  1666→      replyText = fallback
  1667→    }
  1668→  }
  1669→
  1670→  if (!replyText.trim()) {
  1671→    const fallback = advisoryMode
  1672→      ? advisoryFallback(v)
  1673→      : multiReplyParts
  1674→        ? deterministicMultiReply(multiReplyParts, multiOverflow)
  1675→        : card ? `جهزت المسودة — ${card.summary}` : permissionDenied ? permissionMsg : chatFallbackWithChips(v)
  1676→    await streamDeterministic(fallback, emit, 900)
  1677→    replyText = fallback
  1678→  }
  1679→
  1680→  // ---- PASS 5: Deep-link action buttons (after deltas, before done) ----
  1681→  if (actions && actions.length > 0) {
  1682→    emit({ event: 'actions', actions })
  1683→  }
  1684→
  1685→  await bumpMemoryUsage(memCtx.referencedIds)
  1686→  return { replyText, card, cards, thinking: steps, modelLabel, opId, actions: actions && actions.length > 0 ? actions : null }
  1687→}
