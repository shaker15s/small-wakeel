    85→// ---------------------------------------------------------------------------
    86→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
    87→// and a hard floor so a slow provider never gets delay stacked on top of it.
    88→// ---------------------------------------------------------------------------
    89→const REPLY_PACING_BUDGET_MS = 1500
    90→const PIECE_DELAY_MIN_MS = 3
    91→const PIECE_DELAY_MAX_MS = 26
    92→
    93→interface PacerState {
    94→  t0: number | null
    95→  emitted: number
    96→}
    97→
    98→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
    99→ * The moment elapsed time eats the whole budget (the provider itself is slow),
   100→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
   101→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
   102→  const now = Date.now()
   103→  if (p.t0 == null) p.t0 = now
   104→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
   105→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
   106→  // Estimate the pieces still coming: at least the rest of this chunk, and at
   107→  // least as many as we've already emitted (replies rarely stop abruptly).
   108→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
   109→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
   110→}

   112→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
   113→ * (identity/memory replies: whole text known upfront, exact remaining count). */
   114→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
   115→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
   116→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
   117→  // project targets ES2017, where the `s` flag is a tsc error).
   118→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
   119→  const t0 = Date.now()
   120→  for (let i = 0; i < pieces.length; i++) {
   121→    emit({ event: 'delta', text: pieces[i] })
   122→    const remainingPieces = pieces.length - i - 1
   123→    if (remainingPieces <= 0) break
   124→    const remainingBudget = budgetMs - (Date.now() - t0)
   125→    if (remainingBudget <= 0) {
   126→      await sleep(PIECE_DELAY_MIN_MS)
   127→      continue
   128→    }
   129→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
   130→  }
   131→}

   265→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   266→  const s = normalizeAr(text)
   267→  const isGym = businessType === 'gym'
   268→
   269→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   270→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   271→  }
   272→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   273→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   274→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   275→  }
   276→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   277→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   278→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   279→  if (/(مديون|اللي ليا|تحصيلات|مين عليها|عليها فلوس|عليهم فلوس|العملاء)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   280→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   281→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   282→  // Price query («ايه سعر عود كمبودي؟» / «بكام المسك؟») → catalog search (R0 read)
   283→  if (/(سعر|بكام)/.test(s) && !/(بعت|بيع|بع |خرج|سلف|دفع|سجل)/.test(s)) {
   284→    const q = s.replace(/^.*?(?:سعر|بكام)\s*/, '').replace(/[؟?].*$/, '').trim()
   285→    if (q.length >= 2) return { intent: 'read', tool: 'search_catalog', fields: { query: q.slice(0, 40) } }
   286→  }
   287→  if (/حضور/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   288→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   289→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   290→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   291→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   292→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   293→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   294→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   295→    if (name && (isQuestion || /عضويه/.test(s))) {
   296→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   297→    }
   298→  }
   299→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   300→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   301→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   302→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   303→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   304→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   305→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   306→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   307→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   308→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   309→  // fire a collection (caught by the expiring rule above instead).
   310→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   311→    return { intent: 'write', opType: 'collection', fields: {} }
   312→  }
   313→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   314→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }

   346→/** Parse the LLM JSON contract: single intent OR the "intents" array (multi). */
   347→function parseIntentJsonMulti(raw: string): { intents: RawIntentJson[]; single: ExtractedIntent | null } {
   348→  const cleaned = raw.replace(/```json|```/g, '').trim()
   349→  const start = cleaned.indexOf('{')
   350→  const end = cleaned.lastIndexOf('}')
   351→  if (start < 0 || end <= start) return { intents: [], single: null }
   352→  try {
   353→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent> & { intents?: RawIntentJson[] }
   354→    if (Array.isArray(json.intents)) {
   355→      const valid = json.intents.filter((r) => r && typeof r.intent === 'string')
   356→      if (valid.length >= 2) return { intents: valid, single: null }
   357→      if (valid.length === 1) return { intents: [], single: rawToSingle(valid[0]) }
   358→      return { intents: [], single: null }
   359→    }
   360→    if (!json.intent) return { intents: [], single: null }
   361→    return { intents: [], single: rawToSingle(json as RawIntentJson) }
   362→  } catch {
   363→    return { intents: [], single: null }
   364→  }

   380→/** Read-request trigger keywords → tool (used by the deterministic splitter). */
   381→const READ_SPLITTERS: [RegExp, string][] = [
   382→  [/ملخص|حال اليوم|كويس اليوم|عملت ايه النهارده|اقفل الورديه/, 'get_daily_summary'],
   383→  [/وضع الخزنه|رصيد الخزنه|الخزنه|الكاش/, 'get_cash_position'],
   384→  [/النواقص|المخزون الناقص|الناقص|المخزون|استوك/, 'list_low_stock'],
   385→  [/صلاحيه?|منتهي|قربت تخلص|تنتهي/, 'list_expiring'],
   386→  [/مين عليها|عليها فلوس|عليهم فلوس|مديون|تحصيلات|اللي ليا/, 'list_receivables'],
   387→  [/اللي علينا|للموردين|الموردين|مستحقات/, 'list_payables'],
   388→  [/آخر الحركات|اخر الحركات|سجل العمليات|آخر العمليات|اخر العمليات|العمليات المستنيه/, 'get_activity_history'],
   389→  [/سعر|بكام/, 'search_catalog'],
   390→]
   391→
   392→/** Sequence verbs that START a new request mid-message (never paid-clauses). */
   393→const WRITE_CUT_RE = /بعت|اشتريت|اشتري|سجل|حصلت|قبضت|استلمت|سدد|صرفت|مصروف|تسويه|عكسها|اعكس/
   394→
   395→/**
   396→ * Deterministic compound splitter — «ملخص اليوم ومين عليها فلوس وايه الناقص
   397→ * في المخزون» → 3 read sub-intents. Cuts at read-trigger keywords, sequence
   398→ * verbs (بعت/سجل/…) and connectors (و بعدين/كمان) — never inside a multi-item
   399→ * basket («بعت ٢ لبن و ١٠ بيض» stays ONE write). Zero LLM.
   400→ */
   401→function splitSubIntents(text: string, businessType: string): SubIntent[] {
   402→  const norm = toWesternDigits(normalizeAr(text))
   403→  if (norm.length < 8) return []
   404→  type Hit = { at: number; len: number; tool: string | null }
   405→  const hits: Hit[] = []
   406→  for (const [re, tool] of READ_SPLITTERS) {
   407→    const g = new RegExp(re.source, 'g')
   408→    let m: RegExpExecArray | null
   409→    while ((m = g.exec(norm)) != null) {
   410→      hits.push({ at: m.index, len: m[0].length, tool })
   411→      if (m.index === g.lastIndex) g.lastIndex++
   412→    }
   413→  }
   414→  // sequence verbs — only when a و/، actually precedes them (mid-message cut)
   415→  {
   416→    const g = new RegExp(WRITE_CUT_RE.source, 'g')
   417→    let m: RegExpExecArray | null
   418→    while ((m = g.exec(norm)) != null) {
   419→      const at = m.index
   420→      const prev = at > 0 ? norm[at - 1] : ''
   421→      const prev2 = at > 1 ? norm[at - 2] : ''
   422→      const prev3 = at > 2 ? norm[at - 3] : ''
   423→      const isBoundary =
   424→        prev === 'و' || prev === '،' || prev === ',' || prev === '؛' || (prev === ' ' && (prev2 === 'و' || prev2 === '،')) || (prev === ' ' && prev2 === ' ' && prev3 === 'و')
   425→      if (isBoundary) hits.push({ at, len: m[0].length, tool: null })
   426→      if (m.index === g.lastIndex) g.lastIndex++
   427→    }
   428→  }
   429→  // connectors «و بعدين» — cut at the connector itself
   430→  {
   431→    const g = /(بعدين|بعد كده|كمان)/g
   432→    let m: RegExpExecArray | null
   433→    while ((m = g.exec(norm)) != null) {
   434→      hits.push({ at: m.index, len: m[0].length, tool: null })
   435→      if (m.index === g.lastIndex) g.lastIndex++
   436→    }
   437→  }
   438→  hits.sort((a, b) => a.at - b.at)
   439→  if (hits.length === 0) return []
   440→
   441→  // dedupe overlaps + same-tool continuations, build cut points
   442→  const cuts: { at: number; tool: string | null }[] = []
   443→  let coveredEnd = -1
   444→  let currentTool: string | null = null
   445→  for (const h of hits) {
   446→    if (h.at < coveredEnd) continue // overlapping keyword (الخزنه inside وضع الخزنه)
   447→    if (h.tool != null && h.tool === currentTool && h.tool !== 'search_catalog') {
   448→      coveredEnd = Math.max(coveredEnd, h.at + h.len) // same tool still talking
   449→      continue
   450→    }
   451→    // cut position: include a glued و in the PREVIOUS segment
   452→    const cutAt = h.at > 0 && norm[h.at - 1] === 'و' ? h.at - 1 : h.at
   453→    cuts.push({ at: cutAt, tool: h.tool })
   454→    coveredEnd = h.at + h.len
   455→    currentTool = h.tool
   456→  }
   457→  if (cuts.length === 0) return []
   458→
   459→  // build segments
   460→  const segments: { text: string; tool: string | null }[] = []
   461→  const seg0 = norm.slice(0, cuts[0].at).replace(/[\s،,؛]+$/, '').trim()
   462→  if (seg0.length > 0) segments.push({ text: seg0, tool: null })
   463→  for (let i = 0; i < cuts.length; i++) {
   464→    const end = i + 1 < cuts.length ? cuts[i + 1].at : norm.length
   465→    const seg = norm.slice(cuts[i].at, end).replace(/^[\s،,؛و]+/, '').replace(/[\s،,؛]+$/, '').trim()
   466→    if (seg.length > 0) segments.push({ text: seg, tool: cuts[i].tool })
   467→  }
   468→
   469→  // map segments → sub-intents
   470→  const subs: SubIntent[] = []
   471→  for (let i = 0; i < segments.length; i++) {
   472→    const seg = segments[i]
   473→    // politeness prefixes («قوللي», «اط») — skip chat-y tiny segments
   474→    if (seg.tool == null) {
   475→      if (i === 0 && seg.text.length <= 6) continue
   476→      const mapped = ruleBasedIntent(seg.text, businessType)
   477→      if (mapped.intent === 'read' && mapped.tool) subs.push({ ...mapped, intent: 'read', tool: mapped.tool, sourceText: seg.text })
   478→      else if (mapped.intent === 'write' && mapped.opType) subs.push({ ...mapped, intent: 'write', sourceText: seg.text })
   479→      continue
   480→    }
   481→    // tool-owned segment
   482→    if (seg.tool === 'search_catalog') {
   483→      const q = seg.text.replace(/^(?:ايه|ايه هو|هو)?\s*(?:سعر|بكام)\s*/, '').replace(/[؟?].*$/, '').trim()
   484→      subs.push({ intent: 'read', tool: 'search_catalog', fields: q.length >= 2 ? { query: q.slice(0, 40) } : {}, sourceText: seg.text })
   485→    } else {
   486→      subs.push({ intent: 'read', tool: seg.tool, fields: {}, sourceText: seg.text })
   487→    }
   488→  }
   489→
   490→  // merge CONSECUTIVE same-opType writes into one basket (multi-item scan)
   491→  const merged: SubIntent[] = []
   492→  for (const s of subs) {
   493→    const last = merged[merged.length - 1]
   494→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   495→      last.sourceText = `${last.sourceText} و ${s.sourceText}`
   496→      last.fields = { ...last.fields, ...s.fields }
   497→      continue
   498→    }
   499→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   500→      continue // duplicate read
   501→    }
   502→    merged.push({ ...s })
   503→  }
   504→  return merged
   505→}

   507→/** Normalize LLM multi-intents: validate, merge consecutive same-opType writes. */
   508→function normalizeMultiIntents(raws: RawIntentJson[], fullText: string): SubIntent[] {
   509→  const subs: SubIntent[] = []
   510→  for (const r of raws) {
   511→    if (r.intent === 'read' && r.tool && READ_TOOLS.has(r.tool)) {
   512→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   513→      subs.push({ intent: 'read', tool: r.tool as ReadTool, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   514→    } else if (r.intent === 'write' && r.opType && VALID_OP_TYPES.has(r.opType)) {
   515→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   516→      subs.push({ intent: 'write', opType: r.opType as OpType, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   517→    }
   518→  }
   519→  const merged: SubIntent[] = []
   520→  for (const s of subs) {
   521→    const last = merged[merged.length - 1]
   522→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   523→      // consecutive same-type writes = ONE basket; full-text scan finds the lines
   524→      last.sourceText = fullText
   525→      last.fields = { ...last.fields, ...s.fields }
   526→      continue
   527→    }
   528→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   529→      continue // duplicate read
   530→    }
   531→    merged.push(s)
   532→  }
   533→  return merged
   534→}

   596→function deterministicMultiReply(parts: string[], overflow: number): string {

   633→/**
   634→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   635→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   636→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   637→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   638→ * customer — the ل/من marker is what marks the ROLE.
   639→ */
   640→function findPartyMention<T extends { name: string }>(
   641→  text: string,
   642→  parties: T[],
   643→  marker: 'ل' | 'من' = 'ل',
   644→): T | null {
   645→  const norm = normalizeAr(text)
   646→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   647→  for (const p of parties) {
   648→    const words = normalizeAr(p.name)
   649→      .split(' ')
   650→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   651→      .filter((w) => w.length >= 2)
   652→    for (const w of words) {
   653→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   654→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   655→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   656→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   657→    }
   658→  }
   659→  return null
   660→}

   685→/** Paid amount: LLM field → explicit «دفع 20» clause → آجل = 0 → default cash (total). */
   686→function resolvePaid(norm: string, f: Record<string, unknown>, total: number): number {
   687→  if (typeof f.paid === 'number' && f.paid > 0) return Math.min(Math.round(f.paid * 100), total)
   688→  const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   689→  if (paidWord?.[1]) {
   690→    const n = parseQuantity(paidWord[1], null).qty
   691→    if (n != null && n > 0) return Math.min(Math.round(n * 100), total)
   692→  }
   693→  if (/(?:بال)?اجل|علي الحساب|نسيون/.test(norm)) return 0
   694→  return total
   695→}

   697→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit, memCtx: MemoryContext): Promise<ResolvedWrite> {
   698→  const { business, user } = ctx
   699→  const type = (intent.opType ?? 'sale') as OpType
   700→  const f = intent.fields ?? {}
   701→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   702→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   703→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   704→
   705→  const text = ctx.text
   706→  const norm = normalizeAr(text)
   707→  const normW = toWesternDigits(norm)
   708→
   709→  switch (type) {
   710→    case 'sale':
   711→    case 'purchase': {
   712→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   713→      // «دفع ٢٠ بس» can never be misread as the sold quantity.

  1013→      const summary = `تحصيل ${egp(amount)} من ${customer.name}`
  1014→      const card = fieldsToCard('', type, risk, summary, [
  1015→        { label: 'العميل', value: customer.name },
  1016→        { label: 'المبلغ', value: egp(amount) },
  1017→        { label: 'المديونية الحالية', value: egp(customer.balance) },
  1018→        { label: 'بعد التحصيل', value: egp(customer.balance - amount) },
  1019→      ], [{ label: 'الخزنة', value: egp(amount), tone: 'good' }], RISK_META[risk].hint)
  1020→      return { ok: true, opType: type, payload, summary, card }
  1021→    }
  1022→
  1023→    case 'payment': {
  1024→      const supplierName = String(f.supplierName ?? '')
  1025→      const supplier = matchEntity(supplierName || text, suppliers)?.entity ?? null
  1026→      if (!supplier) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المورد', failMessage: 'هسدد لمين؟' }
  1027→      let amount = (f.amount as number | undefined) && (f.amount as number) > 0 ? Math.round((f.amount as number) * 100) : null
  1028→      if (amount == null) {
  1029→        const m = norm.match(/([٠-٩0-9]+(?:[.,][٠-٩0-9]+)?)/)
  1030→        if (m) {
  1031→          const western = m[1].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
  1032→          amount = Math.round(parseFloat(western.replace(',', '.')) * 100)
  1033→        }
  1034→      }
  1035→      if (amount == null || amount <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `هسدد قد إيه لـ${supplier.name}؟` }
  1036→      if (amount > supplier.balance) {
  1037→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المستحق لـ${supplier.name} ${egp(supplier.balance)} بس` }
  1038→      }
  1039→      const payload = { supplierId: supplier.id, supplierName: supplier.name, amount }
  1040→      const risk = computeRisk(type, amount)
  1041→      const summary = `سداد ${egp(amount)} لـ${supplier.name}`
  1042→      const card = fieldsToCard('', type, risk, summary, [
  1043→        { label: 'المورد', value: supplier.name },
  1044→        { label: 'المبلغ', value: egp(amount) },
  1045→        { label: 'المستفق الحالي', value: egp(supplier.balance) },
  1046→        { label: 'بعد السداد', value: egp(supplier.balance - amount) },
  1047→      ], [{ label: 'الخزنة', value: egp(-amount), tone: 'bad' }], RISK_META[risk].hint)
  1048→      return { ok: true, opType: type, payload, summary, card }
  1049→    }
  1050→
  1051→    case 'membership': {
  1052→      const isRenewal = norm.includes('تجديد') || norm.includes('جدد')
  1053→      const explicitName = String(f.customerName ?? '').trim()
  1054→      const customer = explicitName ? matchEntity(explicitName, customers)?.entity ?? null : matchEntity(text, customers)?.entity ?? null
  1055→      if (!customer && isRenewal) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'تجديد لمين؟' }
  1056→      // A NEW subscription with NO name at all (rule path, question-like text)
  1057→      // must never draft «عضو جديد» — ask for the name first.
  1058→      if (!customer && !explicitName) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'اشتراك جديد لمين؟ اكتبلي اسم العضو' }
  1059→      const plan = String(f.plan ?? 'شهرية')
  1060→      const months = Math.max(1, Math.min(12, Math.round(Number(f.months ?? 1))))
  1061→      // plan price from items (gym plans are items) or fallback 35000
  1062→      const planItem = items.find((i) => normalizeAr(i.name).includes(normalizeAr(plan)))
  1063→      const total = Math.round((planItem?.sellPrice ?? 35000) * months)
  1064→      let paid = total
  1065→      if (typeof f.paid === 'number' && f.paid > 0) paid = Math.round(f.paid * 100)
  1066→      paid = Math.min(paid, total)
  1067→      const name = customer?.name ?? explicitName
  1068→      const payload = {
  1069→        ...(customer ? { customerId: customer.id } : { customerName: name }),
  1070→        plan,
  1071→        months,
  1072→        total,
  1073→        paid,
  1074→      }
  1075→      const risk = computeRisk(type, total)
  1076→      const summary = `${isRenewal ? 'تجديد' : 'اشتراك جديد'} ${plan} لـ${name} — ${egp(total)}`
  1077→      const card = fieldsToCard('', type, risk, summary, [
  1078→        { label: 'العضو', value: name },
  1079→        { label: 'الخطة', value: plan },
  1080→        { label: 'المدة', value: fmtQty(months, 'شهر') },
  1081→        { label: 'الإجمالي', value: egp(total) },
  1082→        { label: 'مدفوع', value: egp(paid) },
  1083→      ], [
  1084→        { label: 'الخزنة', value: egp(paid), tone: 'good' },
  1085→        ...(total - paid > 0 ? [{ label: 'باقي على العضو', value: egp(total - paid), tone: 'bad' as const }] : []),
  1086→      ], RISK_META[risk].hint)
  1087→      return { ok: true, opType: type, payload, summary, card }
  1088→    }
  1089→
  1090→    case 'attendance': {
  1091→      const customerName = String(f.customerName ?? '')
  1092→      const customer = matchEntity(customerName || text, customers)?.entity ?? null
  1093→      if (!customer) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'العضو', failMessage: 'حضور مين؟' }
  1094→      const payload = { customerId: customer.id, customerName: customer.name }
  1095→      const risk = computeRisk(type)
  1096→      const summary = `حضور ${customer.name}`
  1097→      const card = fieldsToCard('', type, risk, summary, [
  1098→        { label: 'العضو', value: customer.name },
  1099→        { label: 'اليوم', value: 'النهاردة' },
  1100→        ...(customer.expiresAt ? [{ label: 'الاشتراك لحد', value: arDate(customer.expiresAt) }] : []),
  1101→      ], [], RISK_META[risk].hint)
  1102→      return { ok: true, opType: type, payload, summary, card }
  1103→    }
  1104→
  1105→    case 'stock_adjustment': {
  1106→      const rawItemName = String(f.itemName ?? '')
  1107→      const item = matchEntity(rawItemName || text, items)?.entity ?? null
  1108→      if (!item) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: 'تسوية مخزون لأنهي صنف؟' }
  1109→      let delta = (f.delta as number | undefined) ?? null
  1110→      if (delta == null && typeof f.qty === 'number') delta = f.qty - item.stockQty
  1111→      if (delta == null) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية الجديدة/الفرق', failMessage: `المخزون الحالي ${fmtQty(item.stockQty, arUnit(item.unit))} — عايزه يبقى قد إيه؟` }
  1112→      const reason = String(f.reason ?? '') || 'تسوية جرد'
  1113→      const payload = { itemId: item.id, itemName: item.name, delta, reason, unit: item.unit }
  1114→      const risk = computeRisk(type)
  1115→      const summary = `تسوية مخزون ${item.name}: ${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`
  1116→      const card = fieldsToCard('', type, risk, summary, [
  1117→        { label: 'الصنف', value: item.name },
  1118→        { label: 'الحالي', value: fmtQty(item.stockQty, arUnit(item.unit)) },
  1119→        { label: 'الجديد', value: fmtQty(item.stockQty + delta, arUnit(item.unit)) },
  1120→        { label: 'السبب', value: reason },
  1121→      ], [{ label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty + delta, arUnit(item.unit))}`, tone: delta >= 0 ? 'good' : 'bad' }], 'تسوية المخزون عملية حساسة (R3) — محتاجة سبب واعتماد')
  1122→      return { ok: true, opType: type, payload, summary, card }
  1123→    }
  1124→
  1125→    case 'reversal': {
  1126→      const ref = String(f.ref ?? '') || (norm.match(/WKL-\d+/)?.[0] ?? '')
  1127→      if (!ref) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'رقم العملية', failMessage: 'عايز تعكس أنهي عملية؟ اكتب رقمها (زي WKL-1006)' }
  1128→      const original = await db.operation.findFirst({ where: { businessId: business.id, ref } })
  1129→      if (!original) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `ملقيتش عملية رقم ${ref}` }
  1130→      if (original.reversedByRef) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `العملية ${ref} معكوسة قبل كده بـ${original.reversedByRef}` }
  1131→      const reason = String(f.reason ?? '') || 'تصحيح خطأ'
  1132→      const payload = { originalRef: ref, reason }
  1133→      const risk = computeRisk(type)
  1134→      const summary = `عكس العملية ${ref} (${original.summary})`
  1135→      const card = fieldsToCard('', type, risk, summary, [
  1136→        { label: 'العملية الأصلية', value: `${ref} — ${TYPE_AR[original.type] ?? original.type}` },
  1137→        { label: 'ملخصها', value: original.summary },
  1138→        { label: 'السبب', value: reason },
  1139→      ], [{ label: 'الأثر', value: 'هيتلغى بالكامل بعملية عكسية موثقة', tone: 'neutral' }], 'عكس عملية مالية حساس (R3) — التاريخ مش بيتمسح، بيتصحح')
  1140→      return { ok: true, opType: type, payload, summary, card }
  1141→    }
  1142→
  1143→    default:
  1144→      return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: 'نوع العملية مش مدعوم' }
  1145→  }
  1146→}

  1161→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
  1162→  const { business, user, text } = ctx
  1163→  const v = getVertical(business.type)
  1164→  const model = await resolveModel(ctx.modelKey)
  1165→  const modelLabel = modelLabelFor(model)
  1166→
  1167→  // Load context once (memories raised to 8; pending ops count for the persona)
  1168→  const [items, customers, suppliers, memories, pendingCount] = await Promise.all([
  1169→    db.item.findMany({ where: { businessId: business.id } }),
  1170→    db.customer.findMany({ where: { businessId: business.id } }),
  1171→    db.supplier.findMany({ where: { businessId: business.id } }),
  1172→    db.agentMemory.findMany({ where: { businessId: business.id, userId: user.id, active: true }, orderBy: { createdAt: 'desc' }, take: 12 }),
  1173→    db.operation.count({ where: { businessId: business.id, status: 'AWAITING_APPROVAL' } }),
  1174→  ])
  1175→  const rules = memories.filter((m) => m.kind === 'rule')
  1176→  const memCtx: MemoryContext = {
  1177→    lexicon: parseLexicon(memories),
  1178→    rules,
  1179→    referencedIds: [],
  1180→  }
  1181→  // Prompt memories: preferences + facts (rules are injected in their own block)
  1182→  const promptMemories = memories.filter((m) => m.kind === 'preference' || m.kind === 'fact')
  1183→

  1185→  // «بفهم طلبك» whenever the agent actually remembers something about this user.
  1186→  const memStep: ThinkingStep[] =
  1187→    memories.length > 0
  1188→      ? [

  1190→            label: 'فتكرت من ذاكرتي',

  1194→          },
  1195→        ]
  1196→      : []

  1198→    { label: 'بفهم طلبك', detail: 'بحلل الجملة وأستخرج القصد والحقول', state: 'active', icon: 'understand' },
  1199→    ...memStep,
  1200→    { label: 'بدوّر في سجلك', detail: 'بدوّر على الأصناف والأسماء الحقيقية', state: 'pending', icon: 'search' },
  1201→    { label: 'بتحقق من الأرقام', detail: 'بحسب من الأسعار الفعلية مش من دماغي', state: 'pending', icon: 'verify' },

  1206→  const done = (i: number, detail?: string) => {
  1207→    if (steps[i]) {
  1208→      steps[i].state = 'done'
  1209→      if (detail) steps[i].detail = detail
  1210→    }

  1212→  }

  1214→
  1215→  const norm = normalizeAr(text)
  1216→
  1217→  // ---- PASS 0a: Identity question → instant, deterministic, zero LLM ----
  1218→  if (isIdentityQuestion(norm)) {
  1219→    const lowStockCount = items.filter((i) => i.stockQty <= i.lowStockThreshold).length
  1220→    const receivablesTotal = customers.reduce((s, c) => s + Math.max(0, c.balance), 0)

  1222→    done(IDX.search, 'جبت بياناتي من سجلك الحقيقي')

  1227→    if (identityMemory) memCtx.referencedIds.push(identityMemory.id)
  1228→    const replyText = buildIdentityReply({
  1229→      business,
  1230→      user,
  1231→      verticalLabel: v.label,
  1232→      memoryText: identityMemory?.text ?? null,
  1233→      lowStockCount,

  1238→      { id: 'act-today', label: 'لوحة اليوم', view: 'today', icon: 'today', tone: 'primary' },

  1244→    emit({ event: 'actions', actions })
  1245→    await trackUsage(business.id, user.id, 'chat', 'chat')
  1246→    await bumpMemoryUsage(memCtx.referencedIds)
  1247→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions }
  1248→  }
  1249→
  1250→  // ---- PASS 0b: Memory save (preference OR BUSINESS_DNA rule) → instant, persisted, zero LLM ----
  1251→  const saveRequest = extractSaveRequest(text, norm) ?? extractRuleRequest(text, norm)
  1252→  if (saveRequest) {
  1253→    // Dedupe: re-saving the exact same memory must NOT clutter the memory
  1254→    // list — the owner repeating himself deserves a calm «already saved».
  1255→    const isDuplicate = memories.some((m) => normalizeAr(m.text) === normalizeAr(saveRequest.text))
  1256→    if (!isDuplicate) {
  1257→      await db.agentMemory.create({
  1258→        data: {
  1259→          businessId: business.id,
  1260→          userId: user.id,

  1264→          confidence: saveRequest.confidence,

  1268→    const isRule = saveRequest.kind === 'rule'
  1269→    done(IDX.understand, isRule ? 'قانون من صاحب الشغل — مش محتاج موديل' : 'طلب حفظ في الذاكرة — مش محتاج موديل')
  1270→    done(IDX.search, 'بتخزنها في ذاكرتي عنك')
  1271→    done(IDX.verify, isDuplicate ? 'موجودة أصلًا — من غير تكرار' : 'اتحفظت فعليًا في قاعدة البيانات')

  1275→      ? `دي محفوظة عندي أصلًا ✅ — «${saveRequest.text}». مطمن، مش هنسيها.`
  1276→      : isRule
  1277→        ? `اتحفظت قانون من قوانين المحل ✅ — «${saveRequest.text}».\nمن دلوقتي أي عملية تخص الموضوع ده هتتنبه عليه قبل الاعتماد.`
  1278→        : `اتحفظت في ذاكرتي ✅ — هفتكرها: «${saveRequest.text}». لما الموقف ده يتكرر هرفعها على طول.`
  1279→    await streamDeterministic(replyText, emit, 1000)
  1280→    await trackUsage(business.id, user.id, 'chat', 'chat')
  1281→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions: null }
  1282→  }
  1283→
  1284→  // ---- PASS 0c: FORGET («انسى…») → deactivate, instant, zero LLM ----
  1285→  const forgetRequest = extractForgetRequest(text, norm)
  1286→  if (forgetRequest) {
  1287→    const target = findMemoryByReference(forgetRequest.reference, memories)
  1288→    done(IDX.understand, 'طلب مسح من الذاكرة — مش محتاج موديل')
  1289→    done(IDX.search, 'بدوّر عليها في ذاكرتي المحفوظة')
  1290→    let replyText: string
  1291→    if (target) {
  1292→      await db.agentMemory.update({ where: { id: target.id }, data: { active: false, updatedAt: new Date() } })
  1293→      done(IDX.verify, 'اتمسحت فعليًا (متفعّلة لأ)')
  1294→      emit({ event: 'step', label: 'بامسح من ذاكرتي', state: 'done' })
  1295→      replyText = `اتمسحت من ذاكرة وكيل ✅ — الكلام ده: «${target.text.slice(0, 100)}».\nلو احتجته تاني في يوم من الأيام قولي «احفظ» وأرجعه لك.`
  1296→    } else {
  1297→      done(IDX.verify, 'ملقيتش حاجة بالشكل ده')
  1298→      emit({ event: 'step', label: 'بامسح من ذاكرتي', state: 'done' })
  1299→      replyText = `بصراحة ملقيتش في ذاكرتي حاجة بالشكل ده 🤔 — قولي جزء من الكلام اللي حفظته بالظبط وأمسحه لك.`
  1300→    }
  1301→    done(IDX.reply)
  1302→    await streamDeterministic(replyText, emit, 1000)
  1303→    await trackUsage(business.id, user.id, 'chat', 'chat')
  1304→    return { replyText, card: null, cards: null, thinking: steps, modelLabel, actions: null }

  1306→
  1307→  // ---- PASS 1: Understand (LLM with deterministic multi-split + rule fallback) ----
  1308→  let llmRaws: RawIntentJson[] = []
  1309→  let llmSingle: ExtractedIntent | null = null
  1310→  try {
  1311→    const system = buildSystemPrompt({ business, user, items, customers, suppliers, memories: promptMemories, rules, history: ctx.history })
  1312→    const raw = await completeChat(model, [
  1313→      { role: 'system', content: system },
  1314→      { role: 'user', content: text },
  1315→    ])
  1316→    const parsed = parseIntentJsonMulti(raw)
  1317→    llmRaws = parsed.intents
  1318→    llmSingle = parsed.single
  1319→    done(IDX.understand, llmSingle?.replyHint ? `فهمت: ${llmSingle.replyHint}` : llmRaws.length >= 2 ? `فهمت ${llmRaws.length} طلبات في رسالة واحدة` : 'فهمت الطلب')
  1320→  } catch (err) {
  1321→    console.error('[harness] intent pass failed → rule-based fallback:', err)
  1322→    done(IDX.understand, 'فهمت الطلب بالقواعد المحلية (الموديل مش متاح)')
  1323→  }
  1324→
  1325→  // Deterministic compound split — beats a single-LLM read when the user
  1326→  // stacked requests («ملخص اليوم ومين عليها فلوس وايه الناقص»).
  1327→  const detSubs = splitSubIntents(text, business.type)
  1328→
  1329→  let subs: SubIntent[] = []
  1330→  if (llmRaws.length >= 2) {
  1331→    subs = normalizeMultiIntents(llmRaws, text)
  1332→    if (subs.length === 1) {
  1333→      // the array collapsed to one valid request → single path with it
  1334→      const only = subs[0]
  1335→      llmSingle = { intent: only.intent, tool: only.tool, opType: only.opType, fields: only.fields, replyHint: only.replyHint }
  1336→      subs = []
  1337→    }
  1338→  } else if (detSubs.length >= 2) {
  1339→    subs = detSubs.map((s) => ({ ...s }))
  1340→    // if the LLM single intent is the same write as the leading sub, prefer its fields
  1341→    if (llmSingle?.intent === 'write' && subs[0].intent === 'write') {

  1376→      detail: s.sourceText.slice(0, 40),

  1395→    const usedTools: string[] = []

  1397→

  1413→        replyParts.push(deterministicReadReply(subCards))
  1414→        await trackUsage(business.id, user.id, 'tool', sub.tool === 'search_catalog' || sub.tool === 'get_item_stock' ? `${sub.tool}:${String(sub.fields?.query ?? sub.fields?.name ?? '')}` : sub.tool)
  1415→      } else if (sub.intent === 'write' && sub.opType) {
  1416→        const opType = sub.opType
  1417→        if (!canDraft(user.role, opType)) {
  1418→          steps[1 + memStep.length + i].state = 'done'
  1419→          pushThinking()
  1420→          emit({ event: 'step', label: stepLabel, state: 'done' })
  1421→          situationParts.push(`الطلب ${AR_ORDINALS[i]}/${Math.min(N, 4)} — عملية ${TYPE_AR[opType] ?? opType} محتاجة صلاحية أعلى من صلاحية ${user.name} (${roleTitle(user)}) — ولا حاجة اتنفذت.`)

  1426→        if (!resolved.ok) {
  1427→          steps[1 + memStep.length + i].state = 'done'
  1428→          pushThinking()
  1429→          emit({ event: 'step', label: stepLabel, state: 'done' })
  1430→          situationParts.push(`الطلب ${AR_ORDINALS[i]}/${Math.min(N, 4)} — ناقص ${resolved.missing ?? ''}: ${resolved.failMessage ?? ''}`)
  1431→          replyParts.push(`طلب رقم ${AR_ORDINALS[i]} ناقص بيانات: ${resolved.failMessage ?? resolved.missing ?? ''}`)
  1432→          continue
  1433→        }
  1434→        const idem = `${business.id}:${user.id}:${Date.now()}:${Math.random().toString(36).slice(2, 8)}`
  1435→        const draft = await createDraft({
  1436→          businessId: business.id,
  1437→          userId: user.id,
  1438→          type: resolved.opType,
  1439→          summary: resolved.summary,
  1440→          payload: resolved.payload,
  1441→          conversationId: ctx.conversationId,
  1442→          idempotencyKey: idem,
  1443→        })
  1444→        if (!opId) {
  1445→          // the SSE contract has ONE draft-card slot per message — the first
  1446→          // draft owns it; later drafts stay in the journal + reply text
  1447→          opId = draft.id
  1448→          card = { ...resolved.card, ref: draft.ref, opId: draft.id }
  1449→          emit({ event: 'card', card })
  1450→        }
  1451→        anyDraft = true
  1452→        await trackUsage(business.id, user.id, 'tool', resolved.opType)
  1453→        situationParts.push(`الطلب ${AR_ORDINALS[i]}/${Math.min(N, 4)} — اتعملت مسودة رقم ${draft.ref} (${resolved.summary}) مستنية اعتماد المستخدم${resolved.card.dangerNote ? `، وفيها تحذير: ${resolved.card.dangerNote}` : ''}. مفيش حاجة اتنفذت لسه.`)
  1454→        replyParts.push(`جهزت مسودة «${TYPE_AR[resolved.opType] ?? resolved.opType}» (${draft.ref}): ${resolved.summary} — مستنية اعتمادك.`)
  1455→      }
  1456→      steps[1 + memStep.length + i].state = 'done'
  1457→      pushThinking()
  1458→      emit({ event: 'step', label: stepLabel, state: 'done' })
  1459→    }
  1460→
  1461→    if (allCards.length > 0) cards = allCards
  1462→    multiReplyParts = replyParts
  1463→    multiOverflow = overflow
  1464→    done(verifyIdx, N > 4 ? `${Math.min(N, 4)} طلبات اتجاوب عليهم — والباقي ${overflow}` : 'كل الأرقام من قاعدة البيانات مباشرة')
  1465→    done(replyIdx)
  1466→
  1467→    situation = `المستخدم بعت رسالة فيها ${N} طلبات. ${situationParts.join('\n\n')}`
  1468→    replyInstruction = `جاوب على كل طلب منهم بالترتيب${overflow > 0 ? ` (خدت أول ٤ — قول له إن ${overflow} طلب${overflow === 1 ? '' : 'ات'} تاني${overflow === 1 ? '' : 'ن'} مستنيين)` : ''}. ابدأ كل جزء برقم عربي ونقطة (١. ٢. ٣.) وسيب سطر فاضي بين كل جزء والتاني. كل الأرقام من «الوضع الحالي» بس. متختلقش أي رقم.`
  1469→    // merged actions: max 3, most relevant (dedup by view; drafts first)
  1470→    const merged = new Map<string, ChatActionDTO>()
  1471→    if (anyDraft) merged.set('journal', DRAFT_ACTIONS[0])
  1472→    for (const t of usedTools) {
  1473→      for (const a of TOOL_ACTIONS[t] ?? []) {
  1474→        if (!merged.has(a.view)) merged.set(a.view, a)
  1475→      }
  1476→    }
  1477→    actions = [...merged.values()].slice(0, 3)
  1478→  } else if (llmSingle && llmSingle.intent === 'read' && llmSingle.tool) {
  1479→    // ================= SINGLE READ =================
  1480→    const tool = llmSingle.tool as ReadTool
  1481→    emit({ event: 'step', label: `بشغّل أداة ${tool}`, state: 'active' })
  1482→    done(IDX.search, 'لقيت البيانات في السجل')
  1483→    cards = await runReadTool(tool, business.id, llmSingle.fields ?? {})
  1484→    emit({ event: 'cards', cards })
  1485→    emit({ event: 'step', label: `بشغّل أداة ${tool}`, state: 'done' })

  1488→    replyInstruction = 'لخّص النتيجة للمستخدم بالمصري بأرقامها الصحيحة من الجدول فوق. لو مفيش بيانات قول كده بصراحة.'
  1489→    const hint = proactiveHintFor(tool, cards)
  1490→    if (hint) replyInstruction += ` ${hint}`
  1491→    actions = TOOL_ACTIONS[tool] ?? null
  1492→    await trackUsage(business.id, user.id, 'tool', tool === 'search_catalog' || tool === 'get_item_stock' ? `${tool}:${String(llmSingle.fields?.query ?? llmSingle.fields?.name ?? '')}` : tool)

  1495→    const opType = llmSingle.opType as OpType
  1496→    if (!canDraft(user.role, opType)) {
  1497→      permissionDenied = true
  1498→      permissionMsg = `عملية «${TYPE_AR[opType] ?? opType}» محتاجة صلاحية أعلى — صلاحية ${user.name} (${roleTitle(user)}) مش بتسمح بتسجيلها. دي من اختصاص صاحب المنشأة أو المشرف بس، ولا حاجة اتنفذت.`

  1556→  let replyText = ''
  1557→  const pacer: PacerState = { t0: null, emitted: 0 }
  1558→  // NOTE: the reply pass MUST carry a trailing user turn — providers reject a
  1559→  // conversation that is only a system/assistant prompt (z-ai: code 1214).
  1560→  const messages: LLMMessage[] = advisoryMode
  1561→    ? [
  1562→        { role: 'system', content: advisorySystem },
  1563→        { role: 'user', content: text },

  1571→      if (chunk.type !== 'content') continue
  1572→      replyText += chunk.text
  1573→      // Re-pace into tiny letter-by-letter deltas — adaptively: total pacing
  1574→      // is capped by the budget; a slow provider gets the 3ms floor instead.
  1575→      // [\s\S] keeps the model's own line breaks alive in the stream
  1576→      // (plain `.` drops \n; ES2017 target forbids the dotAll `s` flag).
  1577→      const pieces = chunk.text.match(/[\s\S]{1,2}/g) ?? []
  1578→      for (let i = 0; i < pieces.length; i++) {
  1579→        emit({ event: 'delta', text: pieces[i] })
  1580→        pacer.emitted++
  1581→        await sleep(adaptivePieceDelay(pacer, pieces.length - i - 1))
  1582→      }
  1583→    }
  1584→    // Multi-intent replies MUST answer every part with numbered sections —
  1585→    // if the model ignored the numbering (weak/canned provider), the
  1586→    // deterministic numbered reply is the guaranteed-correct answer.
  1587→    if (subs.length >= 2 && multiReplyParts && !replyAnswersAllParts(replyText, Math.min(subs.length, 4))) {
  1588→      replyText = deterministicMultiReply(multiReplyParts, multiOverflow)
  1589→    }
  1590→  } catch (err) {
  1591→    console.error('[harness] reply stream failed:', err)
  1592→    if (!replyText) {
  1593→      const fallback = advisoryMode
  1594→        ? advisoryFallback(v)
  1595→        : multiReplyParts
  1596→          ? deterministicMultiReply(multiReplyParts, multiOverflow)
  1597→          : card
  1598→            ? `جهزتلك مسودة «${card.title}» — ${card.summary}. راجعها واعتمدها لو تمام.${draftIssues ? ` بس خد بالك: ${draftIssues.join('؛ ')}.` : ''}`
  1599→            : cards
  1600→              ? deterministicReadReply(cards)
  1601→              : questionMode
  1602→                ? (situation.match(/— (.+)$/)?.[1] ?? 'محتاج تفاصيل أكتر لو سمحت.')
  1603→                : permissionDenied
  1604→                  // The cashier must NEVER read «تمام» on a denied operation —
  1605→                  // that reads like «it's done». Say the permission truth instead.
  1606→                  ? permissionMsg
  1607→                  : chatFallbackWithChips(v)
  1608→      await streamDeterministic(fallback, emit, 900)
  1609→      replyText = fallback
  1610→    }
  1611→  }
  1612→
  1613→  if (!replyText.trim()) {
  1614→    const fallback = advisoryMode
  1615→      ? advisoryFallback(v)
  1616→      : multiReplyParts
  1617→        ? deterministicMultiReply(multiReplyParts, multiOverflow)
  1618→        : card ? `جهزت المسودة — ${card.summary}` : permissionDenied ? permissionMsg : chatFallbackWithChips(v)
  1619→    await streamDeterministic(fallback, emit, 900)
  1620→    replyText = fallback
  1621→  }
  1622→
  1623→  // ---- PASS 5: Deep-link action buttons (after deltas, before done) ----
  1624→  if (actions && actions.length > 0) {
  1625→    emit({ event: 'actions', actions })
  1626→  }
  1627→
  1628→  await bumpMemoryUsage(memCtx.referencedIds)
  1629→  return { replyText, card, cards, thinking: steps, modelLabel, opId, actions: actions && actions.length > 0 ? actions : null }
  1630→}

  1632→/** Legacy chip fallback for non-advisory plain-chat turns (R13, kept for safety). */
  1633→function chatFallbackWithChips(v: VerticalDef): string {
  1634→  const three = v.chips.slice(0, 3)
  1635→  if (three.length === 0) return 'تحت أمرك — قوللي أي حاجة وهظبطها.'
  1636→  return `تحت أمرك. تحب نجرب: ${three.join(' · ')}؟`
  1637→}
