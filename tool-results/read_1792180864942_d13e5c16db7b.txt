   107→  // while a NEW key has no cached data yet (so stale messages never linger).
   108→  // (deps include the conversation key — a cache-hit key switch can return a
   109→  // deep-equal object via structural sharing, which alone wouldn't re-fire.)
   110→  useEffect(() => {
   111→    if (!data) {
   112→      setTurns([])
   113→      setRevealId(null)
   114→      setFailedId(null)
   115→      return
   116→    }
   117→    if (sendingRef.current) return
   118→    setTurns(data.messages.map(messageToTurn))
   119→    setLoadedConvId(data.conversation.id)

   121→    setFailedId(null)
   122→  }, [data, business.id, userId, activeConversationId])

   133→    const el = scrollRef.current
   134→    if (!el) return

   139→
   140→  useEffect(() => {
   141→    if (nearBottomRef.current) scrollToBottom(false)
   142→  }, [turns, scrollToBottom])
   143→
   144→  // ---- providers for the model picker ----
   145→  const { data: providersData } = useQuery({
   146→    queryKey: QUERY_KEYS.providers,
   147→    queryFn: () => api.providers(),

   159→    .flatMap((p) => p.models.map((m) => ({ ...m, providerName: p.name })))
   160→    .filter((m) => m.enabled)
   161→
   162→  // ---- dynamic suggestions (the adaptive demos — بتتغير مع شغلك) ----
   163→  const suggestionsQ = useQuery({
   164→    queryKey: QUERY_KEYS.suggestions(business.id, userId),
   165→    queryFn: () => api.suggestions(business.id, userId),
   166→    refetchOnMount: true,
   167→    retry: 1,
   168→  })

   171→  const commandsQ = useQuery({
   172→    queryKey: QUERY_KEYS.quickCommands(business.id),
   173→    queryFn: () => api.quickCommands(business.id),
   174→  })
   175→
   176→  // ---- «ورقة وكيل» count (the agent's scratchpad for THIS conversation) ----
   177→  // same key as the sheet's own query → one fetch, badge + sheet stay in sync
   178→  const notesQ = useQuery({
   179→    queryKey: QUERY_KEYS.notes(business.id, loadedConvId ?? 'none'),
   180→    queryFn: () => api.notes(business.id, loadedConvId as string),
   181→    enabled: !!loadedConvId,
   182→    retry: 1,
   183→    refetchOnWindowFocus: false,
   184→  })
   185→  const notesCount = notesQ.data?.length ?? 0
   186→
   187→  // ---- send a message → stream the turn (ChatGPT order guaranteed) ----
   188→  // STORM-SAFE QUEUE (R15 CEO fix): sends that arrive mid-stream are NOT
   189→  // dropped — they queue and auto-flush the instant the live turn finishes
   190→  // (the silent-loss bug the adversarial audit caught with 3 rapid messages).
   191→  const pendingSendsRef = useRef<Array<{ text: string; opts?: { suggestionId?: string; deepResearch?: boolean } }>>([])
   192→  const [queuedCount, setQueuedCount] = useState(0)
   193→
   194→  const send = useCallback(
   195→    async (text: string, opts?: { suggestionId?: string; deepResearch?: boolean }) => {
   196→      if (!text.trim()) return
   197→      const trimmed = text.trim()
   198→      if (sendingRef.current) {
   199→        // a turn is live → queue this one; it sends automatically on done
   200→        pendingSendsRef.current.push({ text: trimmed, opts })
   201→        setQueuedCount(pendingSendsRef.current.length)
   202→        return
   203→      }
   204→      sendingRef.current = true
   205→      setSending(true)
   206→      setFailedId(null)
   207→      setRevealId(null)
   208→      retryRef.current = trimmed
   209→      // the conversation the scratchpad invalidation targets at done time
   210→      let convId = activeConversationId ?? loadedConvId
   211→
   212→      const userLocalId = `local-user-${Date.now()}`
   213→      const liveId = `local-live-${Date.now()}`
   214→
   215→      setTurns((prev) => [
   216→        ...prev,
   217→        { id: userLocalId, role: 'user', text: trimmed, createdAt: new Date().toISOString() },
   218→        {
   219→          id: liveId,
   220→          role: 'assistant',
   221→          text: '',
   222→          thinking: [],
   223→          createdAt: new Date().toISOString(),
   224→          streaming: true,
   225→        },
   226→      ])
   227→      nearBottomRef.current = true
   228→      setNearBottom(true)
   229→
   230→      // BUFFERS: card/cards/actions stay out of render until the stream ends
   231→      const buf = {
   232→        card: null as ActionCardDTO | null,
   233→        cards: null as ResultCardDTO[] | null,
   234→        actions: null as ChatActionDTO[] | null,
   235→      }
   236→      let errored = false
   237→      let doneId: string | null = null
   238→      // IMPORTANT: capture the target id at HANDLER-CALL time — closures that
   239→      // resolve it inside the setTurns updater would read `doneId` at flush
   240→      // time and silently drop the final deltas that arrive in the same batch
   241→      // as the done event.
   242→      const patchTurn = (patch: Partial<ChatTurn>) => {
   243→        const id = doneId ?? liveId
   244→        setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, ...patch } : t)))
   245→      }
   246→
   247→      try {
   248→        await streamChat(
   249→          {
   250→            businessId: business.id,
   251→            userId,
   252→            text: trimmed,
   253→            modelKey,
   254→            conversationId: activeConversationId ?? loadedConvId,
   255→            suggestionId: opts?.suggestionId ?? null,
   256→            deepResearch: opts?.deepResearch ?? false,
   257→          },
   258→          {
   259→            onStart: (cid, userMessageId) => {
   260→              // a new conversation may be created server-side when none was
   261→              // passed — track it so later sends stay in the same thread
   262→              setLoadedConvId(cid)
   263→              convId = cid
   264→              // swap optimistic user id → the real persisted id so any later
   265→              // refetch reconciles by key (no remount → no flicker)
   266→              setTurns((prev) => prev.map((t) => (t.id === userLocalId ? { ...t, id: userMessageId } : t)))
   267→            },
   268→            onThinking: (steps) => patchTurn({ thinking: steps }),
   269→            onStep: (label, state) => {
   270→              const id = doneId ?? liveId
   271→              setTurns((prev) =>
   272→                prev.map((t) =>
   273→                  t.id === id && t.thinking
   274→                    ? { ...t, thinking: t.thinking.map((s) => (s.label === label ? { ...s, state } : s)) }
   275→                    : t,
   276→                ),
   277→              )
   278→            },
   279→            onCard: (card) => {
   280→              buf.card = card
   281→            },
   282→            onCards: (cards) => {
   283→              buf.cards = cards
   284→            },
   285→            onActions: (actions) => {
   286→              buf.actions = actions
   287→            },
   288→            onDelta: (d) => {
   289→              const id = doneId ?? liveId
   290→              setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, text: t.text + d } : t)))
   291→            },
   292→            onDone: (messageId, modelLabel, meta) => {
   293→              doneId = messageId
   294→              // finalize IN PLACE (same DOM node) + reveal the buffers below text
   295→              // (R15: done carries the verification ledger + research metadata)
   296→              setTurns((prev) =>
   297→                prev.map((t) =>
   298→                  t.id === liveId
   299→                    ? {
   300→                        ...t,
   301→                        id: messageId,
   302→                        streaming: false,
   303→                        modelLabel,
   304→                        card: buf.card,
   305→                        cards: buf.cards,
   306→                        actions: buf.actions,
   307→                        verified: meta?.verified ?? null,
   308→                        research: meta?.research ?? null,
   309→                      }
   310→                    : t,
   311→                ),
   312→              )
   313→              setRevealId(messageId)
   314→            },
   315→            onError: (message) => {
   316→              errored = true
   317→              patchTurn({ streaming: false })
   318→              setFailedId(doneId ?? liveId)
   319→              toast({ title: 'مشكلة', description: message, variant: 'destructive' })
   320→            },
   321→          },
   322→        )
   323→      } catch {
   324→        errored = true
   325→        patchTurn({ streaming: false })
   326→        setFailedId(doneId ?? liveId)
   327→        toast({ title: 'مشكلة', description: 'الاتصال اتقطع — جرّب تاني', variant: 'destructive' })
   328→      } finally {
   329→        sendingRef.current = false
   330→        setSending(false)
   331→        if (errored) {
   332→          // keep the failed bubble (retry button) — the server never persisted
   333→          // a reply, so a refetch here would wipe the retry UI
   334→        } else if (doneId) {
   335→          // chat state is already exact locally — refresh the app-wide caches
   336→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   337→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.operations(business.id) })
   338→          // the conversation list (title/preview/counts) + suggestions follow
   339→          // the user's work and update live after every turn
   340→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.conversations(business.id, userId) })
   341→          void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.suggestions(business.id, userId) })
   342→          // «ورقة وكيل» — the agent jots notes while it works; refresh the
   343→          // scratchpad (header badge + sheet) after every completed turn
   344→          if (convId) {
   345→            void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.notes(business.id, convId) })
   346→          }
   347→        } else {
   348→          // stream closed without done/error — treat as a failed turn
   349→          patchTurn({ streaming: false })
   350→          setFailedId(doneId ?? liveId)
   351→          toast({ title: 'مشكلة', description: 'الرد مشكتلش — جرّب تاني', variant: 'destructive' })
   352→        }
   353→        // STORM-SAFE QUEUE: flush the next queued message (if any) now that
   354→        // the live turn finished — the audit's rapid-fire test stays lossless.
   355→        const next = pendingSendsRef.current.shift()
   356→        setQueuedCount(pendingSendsRef.current.length)
   357→        if (next) void send(next.text, next.opts)
   358→      }
   359→    },
   360→    [business.id, userId, modelKey, activeConversationId, loadedConvId, toast, queryClient],
   361→  )
   362→
   363→  const retry = useCallback(() => {
   364→    if (retryRef.current) void send(retryRef.current)
   365→  }, [send])
   366→

   368→  const sendSuggestion = useCallback(
   369→    (s: SuggestionDTO) => {
   370→      void send(s.prompt, { suggestionId: s.id })
   371→    },
   372→    [send],
   373→  )
   374→
   375→  const handleCommandTap = useCallback(
   376→    (cmd: QuickCommandDTO) => {
   377→      void send(cmd.prompt)
   378→      // silent usage ping — updates «استخدمته X مرة» in the manager

   380→        .touchQuickCommand(cmd.id)
   381→        .then((updated) => {
   382→          queryClient.setQueryData<QuickCommandDTO[]>(QUERY_KEYS.quickCommands(business.id), (old) =>
   383→            old?.map((c) => (c.id === updated.id ? updated : c)),
   384→          )
   385→        })
   386→        .catch(() => {})
   387→    },
   388→    [send, business.id, queryClient],
   389→  )
   390→
   391→  // ---- «محادثة جديدة» → create + activate + empty hero ----
   392→  const startNewChat = useCallback(async () => {
   393→    if (creatingRef.current) return
   394→    creatingRef.current = true
   395→    setCreating(true)
   396→    try {
   397→      const summary = await api.createConversation(business.id, userId)
   398→      setActiveConversation(summary.id)
   399→      setTurns([])
   400→      setRevealId(null)
   401→      setFailedId(null)
   402→      setDrawerOpen(false)
   403→      void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.conversations(business.id, userId) })
   404→    } catch (err) {
   405→      toast({
   406→        title: 'مشكلة',
   407→        description: err instanceof Error ? err.message : 'جرّب تاني',
   408→        variant: 'destructive',
   409→      })
   410→    } finally {
   411→      creatingRef.current = false
   412→      setCreating(false)
   413→    }
   414→  }, [business.id, userId, setActiveConversation, queryClient, toast])
   415→
   416→  // ---- approvals (R2+ / R3 reason flow) ----
   417→  const approve = useCallback(
   418→    async (card: ActionCardDTO, reason?: string) => {
   419→      if (!card.opId) return
   420→      setBusyOpId(card.opId)
   421→      try {
   422→        const res = await api.confirmOperation(card.opId, 'approve', userId, reason)
   423→        setTurns((prev) =>
   424→          prev.map((t) =>
   425→            t.card?.opId === card.opId
   426→              ? { ...t, card: { ...t.card!, status: res.operation.status }, outcome: res.outcome ?? undefined }
   427→              : t,
   428→          ),
   429→        )
   430→        toast({
   431→          title: res.outcome?.ok ? 'تم — موثّق ومتحقق' : 'العملية فشلت بأمان',
   432→          description: res.outcome?.ok
   433→            ? `${res.operation.ref} — ${res.operation.summary}`
   434→            : (res.outcome?.failReason ?? 'راجع التفاصيل'),
   435→          variant: res.outcome?.ok ? 'default' : 'destructive',
   436→        })
   437→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   438→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.operations(business.id) })
   439→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.items(business.id) })
   440→      } catch (err) {
   441→        toast({
   442→          title: 'مشكلة في الاعتماد',
   443→          description: err instanceof Error ? err.message : 'جرّب تاني',
   444→          variant: 'destructive',
   445→        })
   446→      } finally {
   447→        setBusyOpId(null)
   448→      }
   449→    },
   450→    [userId, business.id, toast, queryClient],
   451→  )
   452→
   453→  const reject = useCallback(
   454→    async (card: ActionCardDTO) => {
   455→      if (!card.opId) return
   456→      setBusyOpId(card.opId)
   457→      try {
   458→        const res = await api.confirmOperation(card.opId, 'reject', userId)
   459→        setTurns((prev) =>
   460→          prev.map((t) => (t.card?.opId === card.opId ? { ...t, card: { ...t.card!, status: 'CANCELLED' } } : t)),
   461→        )
   462→        toast({ title: 'اتلغت المسودة', description: `${res.operation.ref} اتلغت بلا أي أثر` })
   463→        void queryClient.invalidateQueries({ queryKey: QUERY_KEYS.overview(business.id, userId) })
   464→      } catch (err) {
   465→        toast({ title: 'مشكلة', description: err instanceof Error ? err.message : 'جرّب تاني', variant: 'destructive' })
   466→      } finally {
   467→        setBusyOpId(null)
   468→      }
   469→    },
   470→    [userId, business.id, toast, queryClient],
   471→  )
   472→
   473→  // ---- derived ----
   474→  const lastTurn = turns.length > 0 ? turns[turns.length - 1] : null
   475→  const headerAvatarState: AgentState =
   476→    failedId && lastTurn?.id === failedId
   477→      ? 'error'
   478→      : lastTurn?.role === 'assistant' && lastTurn.streaming
   479→        ? lastTurn.text
   480→          ? 'writing'
   481→          : 'thinking'
   482→        : 'idle'
   483→  const showWelcome = !isLoading && turns.length === 0
   484→  const showJump = !nearBottom && !isLoading
   485→
   486→  // hero chips: dynamic when the endpoint answers, else 2 vertical fallbacks
   487→  const heroSuggestions = !suggestionsQ.isPending && (suggestionsQ.data?.length ?? 0) > 0 ? suggestionsQ.data : null
   488→  const heroFallback = vertical.examples.slice(0, 2)
   489→
   490→  return (
   491→    <MotionConfig reducedMotion="user">
   492→      <div className="relative flex h-full min-h-0 flex-col">
   493→        {/* ===== compact header: agent + business | new chat + history | model ===== */}
   494→        <div className="flex shrink-0 items-center justify-between gap-2 border-b border-border/70 bg-card/60 px-3 py-2 backdrop-blur-sm sm:px-4">
   495→          <div className="flex min-w-0 items-center gap-1.5 sm:gap-2">
   496→            <AgentAvatar state={headerAvatarState} size="sm" />

   604→                    ))
   605→                  ) : (
   606→                    // the suggestion engine is unreachable → silent static fallback
   607→                    heroFallback.map((ex, i) => (
   608→                      <motion.button
   609→                        key={ex}
   610→                        type="button"
   611→                        initial={{ opacity: 0, y: 8 }}
   612→                        animate={{ opacity: 1, y: 0 }}
   613→                        transition={{ delay: 0.15 + i * 0.05, duration: 0.22, ease: 'easeOut' }}
   614→                        onClick={() => void send(ex)}
   615→                        className="press rounded-full border border-border bg-card px-3.5 py-2 text-xs transition-all hover:-translate-y-0.5 hover:border-wakil-amber/50 hover:bg-accent hover:text-accent-foreground focus-visible:outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50"

   617→                        {ex}

   619→                    ))
   620→                  )}

   622→              </motion.div>
   623→            )}
   624→
   625→            <div className="mx-auto flex max-w-2xl flex-col gap-5">
   626→              {turns.map((t) => (
   627→                <MessageBubble
   628→                  key={t.id}
   629→                  turn={t}
   630→                  onApprove={approve}
   631→                  onReject={reject}
   632→                  busyOpId={busyOpId}
   633→                  onNavigate={onNavigate}
   634→                  onRetry={t.id === failedId ? retry : undefined}
   635→                  reveal={t.id === revealId}
   636→                  error={t.id === failedId}
   637→                />
   638→              ))}
   639→            </div>
   640→          </div>
   641→
   642→          {/* floating scroll-to-bottom (appears when scrolled up) */}
   643→          {showJump && (
   644→            <div className="pointer-events-none absolute inset-x-0 bottom-4 flex justify-center">
   645→              <motion.button
   646→                type="button"
   647→                initial={{ scale: 0.6, opacity: 0, y: 4 }}
   648→                animate={{ scale: 1, opacity: 1, y: 0 }}
   649→                transition={{ duration: 0.15, ease: 'easeOut' }}

   653→              >
   654→                <ChevronDown className="size-4" aria-hidden="true" />

   656→            </div>
   657→          )}
   658→        </div>
   659→
   660→        {/* ===== composer ===== */}
   661→        <div className="shrink-0 border-t border-border/70 bg-card/60 px-3 pb-[max(0.5rem,env(safe-area-inset-bottom))] pt-2 backdrop-blur-sm sm:px-4 sm:pb-3">
   662→          <div className="mx-auto max-w-2xl">
   663→            <Composer
   664→              commands={commandsQ.data ?? []}
   665→              suggestions={suggestionsQ.data ?? []}
   666→              onSend={send}
   667→              onCommandTap={handleCommandTap}
   668→              onSuggestionTap={sendSuggestion}
   669→              disabled={sending}
   670→              queuedCount={queuedCount}
   671→            />
   672→            <p className="mt-1.5 hidden text-center text-[10px] text-muted-foreground sm:block">
   673→              وكيل بيسأل قبل ما ينفذ — كل عملية بتتحقق بعد التنفيذ
   674→            </p>
   675→          </div>
   676→        </div>
   677→
   678→        {/* ===== conversations drawer (سجل المحادثات القديمة) ===== */}
   679→        {drawerOpen && (
   680→          <ConversationDrawer
   681→            businessId={business.id}
   682→            userId={userId}
   683→            activeId={activeConversationId ?? loadedConvId}
   684→            onClose={() => setDrawerOpen(false)}
   685→            onSwitch={(id) => {
   686→              if (id !== activeConversationId) setActiveConversation(id)
   687→              setDrawerOpen(false)
   688→            }}
   689→            onNewChat={() => void startNewChat()}
   690→            creating={creating}
   691→          />
   692→        )}
   693→
   694→        {/* ===== «ورقة وكيل» — the scratchpad sheet (read-only) ===== */}
   695→        {notesOpen && loadedConvId && (
   696→          <NotesSheet
   697→            businessId={business.id}
   698→            conversationId={loadedConvId}
   699→            onClose={() => setNotesOpen(false)}
   700→          />
   701→        )}
   702→      </div>
   703→    </MotionConfig>
   704→  )

   748→function ModelPicker({
   749→  modelKey,
   750→  setModel,
   751→  enabledExternal,
   752→  currentLabel,

   773→          variant="outline"

   777→          <Cpu className="size-3.5 text-wakil-amber" aria-hidden="true" />

   781→            aria-hidden="true"
   782→          />
   783→        </Button>
   784→      </PopoverTrigger>
   785→
   786→      <PopoverContent
   787→        align="end"
   788→        // R17 fix: long provider lists overflow small screens with no way to
   789→        // scroll («القايمة بتاعت الموديلات مش بتفتح كاملة ومبقدرش اسطرول») —
   790→        // cap the height and make the list itself scrollable.

   792→        dir="rtl"
   793→      >
   794→        <div className="p-1.5">
   795→          <p className="px-2.5 pb-1 pt-1.5 text-[11px] font-semibold text-muted-foreground">

   799→            <ModelRow
   800→              key={m.key}
   801→              icon={m.icon}
   802→              label={m.label}
   803→              sub={m.sub}
   804→              active={modelKey === m.key}
   805→              onSelect={() => select(m.key, m.label)}

   807→          ))}
   808→
   809→          {enabledExternal.length > 0 && (
   810→            <>
   811→              <p className="mt-1.5 border-t border-border/60 px-2.5 pb-1 pt-2 text-[11px] font-semibold text-muted-foreground">

   815→                <ModelRow
   816→                  key={m.id}
   817→                  icon={TIER_ICONS[m.tier] ?? Brain}
   818→                  label={m.label}
   819→                  sub={`${TIER_AR[m.tier] ?? m.tier} · ${m.providerName}`}
   820→                  active={modelKey === `ext:${m.id}`}
   821→                  onSelect={() => select(`ext:${m.id}`, `${m.label} — ${m.providerName}`)}
   822→                />
   823→              ))}
   824→            </>
   825→          )}
   826→
   827→          <p className="mt-1.5 border-t border-border/60 px-2.5 py-2 text-[10px] leading-relaxed text-muted-foreground">
   828→            تضيف مزوّدين من الإعدادات ↗
   829→          </p>
   830→        </div>
   831→      </PopoverContent>
   832→    </Popover>
