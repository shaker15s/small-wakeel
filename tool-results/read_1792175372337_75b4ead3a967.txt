   110→// ---------------------------------------------------------------------------
   111→// Pacing — ADAPTIVE: a fixed budget from the first delta, 1-2 char pieces,
   112→// and a hard floor so a slow provider never gets delay stacked on top of it.
   113→// ---------------------------------------------------------------------------
   114→const REPLY_PACING_BUDGET_MS = 1500
   115→const PIECE_DELAY_MIN_MS = 3
   116→const PIECE_DELAY_MAX_MS = 26
   117→
   118→interface PacerState {
   119→  t0: number | null
   120→  emitted: number
   121→}
   122→
   123→/** Per-piece delay = clamp(remaining_budget / remaining_pieces, 3, 26)ms.
   124→ * The moment elapsed time eats the whole budget (the provider itself is slow),
   125→ * the delay collapses to the 3ms floor — never stack delay on a slow provider. */
   126→function adaptivePieceDelay(p: PacerState, remainingInChunk: number): number {
   127→  const now = Date.now()
   128→  if (p.t0 == null) p.t0 = now
   129→  const remainingBudget = REPLY_PACING_BUDGET_MS - (now - p.t0)
   130→  if (remainingBudget <= 0) return PIECE_DELAY_MIN_MS
   131→  // Estimate the pieces still coming: at least the rest of this chunk, and at
   132→  // least as many as we've already emitted (replies rarely stop abruptly).
   133→  const estimatedRemaining = Math.max(remainingInChunk, p.emitted, 1)
   134→  return Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / estimatedRemaining))
   135→}

   137→/** Stream a fully-known deterministic reply in 1-2 char pieces within `budgetMs`
   138→ * (identity/memory replies: whole text known upfront, exact remaining count). */
   139→async function streamDeterministic(text: string, emit: TurnEmit, budgetMs = REPLY_PACING_BUDGET_MS): Promise<void> {
   140→  // [\s\S] (not `.` + dotAll `s` flag) is CRITICAL: plain `.` never matches
   141→  // \n, so a multi-line template would lose its line breaks mid-stream (and the
   142→  // project targets ES2017, where the `s` flag is a tsc error).
   143→  const pieces = text.match(/[\s\S]{1,2}/g) ?? []
   144→  const t0 = Date.now()
   145→  for (let i = 0; i < pieces.length; i++) {
   146→    emit({ event: 'delta', text: pieces[i] })
   147→    const remainingPieces = pieces.length - i - 1
   148→    if (remainingPieces <= 0) break
   149→    const remainingBudget = budgetMs - (Date.now() - t0)
   150→    if (remainingBudget <= 0) {
   151→      await sleep(PIECE_DELAY_MIN_MS)
   152→      continue
   153→    }
   154→    await sleep(Math.min(PIECE_DELAY_MAX_MS, Math.max(PIECE_DELAY_MIN_MS, remainingBudget / remainingPieces)))
   155→  }
   156→}

   341→function ruleBasedIntent(text: string, businessType: string): ExtractedIntent {
   342→  const s = normalizeAr(text)
   343→  const isGym = businessType === 'gym'
   344→
   345→  if (/(السلام|اهلا|هلا|صباح|مساء|ازيك|عامل ايه|شكرا)/.test(s) && s.length < 25) {
   346→    return { intent: 'chat', fields: {}, replyHint: 'تحية' }
   347→  }
   348→  // Opinion/decision-seeking → ADVISORY free-chat (the real-employee mode).
   349→  // Must beat the price-query rule below so «رأيك أزود سعر الكمون؟» is a
   350→  // conversation about the decision, never a bare catalog scan.
   351→  if (/(رايك|رأيك|ايه رايك|تنصحني|نصيحه|اعمل ايه لو|تشوف ايه|رايك ايه|استشاره)/.test(s) && !/(بعت|بيع|سجل بيع|تحصيل|اشتري)/.test(s)) {
   352→    return { intent: 'chat', fields: {}, replyHint: 'استشارة شغل' }
   353→  }
   354→  // Stock ADJUSTMENT write («عدل مخزون البيض يبقى ٥٠») must beat the low-stock READ
   355→  if (/(مخزون|استوك)/.test(s) && /(عدل|ظبط|خلي|زود|نقص|صحح|سوي|يبقى|تسويه|اضبط)/.test(s)) {
   356→    return { intent: 'write', opType: 'stock_adjustment', fields: {} }
   357→  }
   358→  if (/(وضع|رصيد|حاله) الخزنه|الخزنه|كاش/.test(s)) return { intent: 'read', tool: 'get_cash_position', fields: {} }
   359→  if (/(مخزون|ناقص|استوك)/.test(s) && !/(بعت|بيع|شرا|اشتري)/.test(s)) return { intent: 'read', tool: 'list_low_stock', fields: {} }
   360→  if (/(صلاحيت|صلاحيه|منتهي|قربت|تنتهي|قرب تخلص)/.test(s)) return { intent: 'read', tool: 'list_expiring', fields: {} }
   361→  if (/(مديون|اللي ليا|تحصيلات|مين عليها|عليها فلوس|عليهم فلوس|العملاء)/.test(s)) return { intent: 'read', tool: 'list_receivables', fields: {} }
   362→  if (/(اللي علينا|للمورد|الموردين|مستحق)/.test(s)) return { intent: 'read', tool: 'list_payables', fields: {} }
   363→  if (/(ملخص|كويس اليوم|حال اليوم|عملت ايه النهارده)/.test(s)) return { intent: 'read', tool: 'get_daily_summary', fields: {} }
   364→  // Price query («ايه سعر عود كمبودي؟» / «بكام المسك؟») → catalog search (R0 read)
   365→  if (/(سعر|بكام)/.test(s) && !/(بعت|بيع|بع |خرج|سلف|دفع|سجل)/.test(s)) {
   366→    const q = s
   367→      .replace(/^.*?(?:سعر|بكام)\s*/, '')
   368→      .replace(/[؟?].*$/, '')
   369→      .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   370→      .replace(/^\s*(?:في|من)\s*/, '')
   371→      .trim()
   372→    if (q.length >= 2) return { intent: 'read', tool: 'search_catalog', fields: { query: q.slice(0, 40) } }
   373→  }
   374→  // Attendance (gym): noun «حضور» + Egyptian verb forms («كريم حضر»,
   375→  // «لعب حصة PT», «خلص تدريب») — a member WORKED OUT today.
   376→  if (/(حضور|(^| )حضر( |$)|لعب حصه?|خلص (تدريب|حصه?)|اتمرن|اتدرب)/.test(s)) return { intent: 'write', opType: 'attendance', fields: {} }
   377→  // Membership STATUS question — a READ, must beat the renewal WRITE below:
   378→  // «عضوية عمر باقي قد ايه؟» → get_member_status for عمر (name word follows
   379→  // عضويه/اشتراك); renewal verbs (جدد/تجديد/سجل) keep the WRITE path.
   380→  if (isGym && !/(جدد|تجديد|سجل|اعمل)/.test(s)) {
   381→    const mMember = s.match(/(?:عضويه|اشتراك)\s+([^ ،,؟?]+)/)
   382→    const name = mMember && !/^(اللي|التي|بتاع|دي|ده|الاشتراكات)$/.test(mMember[1]) ? mMember[1] : ''
   383→    const isQuestion = /(باقي|قد ايه|متبقي|فاضل|تنتهي|خلص|حاله|وضع|اوضه)/.test(s)
   384→    if (name && (isQuestion || /عضويه/.test(s))) {
   385→      return { intent: 'read', tool: 'get_member_status', fields: { name } }
   386→    }
   387→  }
   388→  // Word-boundary forms only: «اشتراك عمر شهرية» matches, but a QUESTION like
   389→  // «الاشتراكات اللي هتنتهي» (plural الاشتراكات) must NOT fire a membership write.
   390→  if (isGym && (/(^| )(اشتراك|تجديد|اشتراك جديد|جديد لـ)( |$)/.test(s) || /جدد/.test(s))) return { intent: 'write', opType: 'membership', fields: {} }
   391→  if (/(مصروف|صرفت)/.test(s)) return { intent: 'write', opType: 'expense', fields: { label: 'مصروف' } }
   392→  // Sale FIRST so a partial payment («بع لبن لأحمد بس دفع ٢٠ بس») never looks
   393→  // like a collection. «بع» is the Egyptian imperative — standalone word.
   394→  if (/(^| )(بعت|بيع|بع|خرج|سلفت)( |$)/.test(s)) return { intent: 'write', opType: 'sale', fields: {} }
   395→  if (/(^| )(اشتريت|شرا|جبت|دخلت)( |$)/.test(s)) return { intent: 'write', opType: 'purchase', fields: {} }
   396→  // Collection: «تحصيل ١٠٠ من أحمد» / «حصلت ١٠٠ من أحمد» / «أحمد دفع ١٠٠»
   397→  // NOTE: «خلص» must be word-initial — «تخلص صلاحيته» (expiry) must NEVER
   398→  // fire a collection (caught by the expiring rule above instead).
   399→  if (/(تحصيل|(?:^| )خلص|سدد|دفع|حصل|قبض|استلم)/.test(s) && /(من|لـ|ل)/.test(s)) {
   400→    return { intent: 'write', opType: 'collection', fields: {} }
   401→  }
   402→  if (/^(حصلت|قبضت|استلمت|خدت)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   403→  if (/^(\S+)\s+(دفع|سدد|خلص|حصل)/.test(s)) return { intent: 'write', opType: 'collection', fields: {} }
   404→  return { intent: 'chat', fields: {} }
   405→}

   435→/** Parse the LLM JSON contract: single intent OR the "intents" array (multi). */
   436→function parseIntentJsonMulti(raw: string): { intents: RawIntentJson[]; single: ExtractedIntent | null } {
   437→  const cleaned = raw.replace(/```json|```/g, '').trim()
   438→  const start = cleaned.indexOf('{')
   439→  const end = cleaned.lastIndexOf('}')
   440→  if (start < 0 || end <= start) return { intents: [], single: null }
   441→  try {
   442→    const json = JSON.parse(cleaned.slice(start, end + 1)) as Partial<ExtractedIntent> & { intents?: RawIntentJson[] }
   443→    if (Array.isArray(json.intents)) {
   444→      const valid = json.intents.filter((r) => r && typeof r.intent === 'string')
   445→      if (valid.length >= 2) return { intents: valid, single: null }
   446→      if (valid.length === 1) return { intents: [], single: rawToSingle(valid[0]) }
   447→      return { intents: [], single: null }
   448→    }
   449→    if (!json.intent) return { intents: [], single: null }
   450→    return { intents: [], single: rawToSingle(json as RawIntentJson) }
   451→  } catch {
   452→    return { intents: [], single: null }
   453→  }
   454→}

   456→function rawToSingle(json: RawIntentJson): ExtractedIntent | null {
   457→  const intent = json.intent === 'read' || json.intent === 'write' || json.intent === 'chat' ? json.intent : null
   458→  if (!intent) return null
   459→  return {
   460→    intent,
   461→    tool: json.tool,
   462→    opType: json.opType as OpType | undefined,
   463→    fields: (json.fields as Record<string, unknown>) ?? {},
   464→    question: json.question,
   465→    replyHint: json.replyHint,
   466→  }

   469→/** Read-request trigger keywords → tool (used by the deterministic splitter). */
   470→const READ_SPLITTERS: [RegExp, string][] = [
   471→  [/ملخص|حال اليوم|كويس اليوم|عملت ايه النهارده|اقفل الورديه/, 'get_daily_summary'],
   472→  [/وضع الخزنه|رصيد الخزنه|الخزنه|الكاش/, 'get_cash_position'],
   473→  [/النواقص|المخزون الناقص|الناقص|المخزون|استوك/, 'list_low_stock'],
   474→  [/صلاحيه?|منتهي|قربت تخلص|تنتهي/, 'list_expiring'],
   475→  [/مين عليها|عليها فلوس|عليهم فلوس|مديون|تحصيلات|اللي ليا/, 'list_receivables'],
   476→  [/اللي علينا|للموردين|الموردين|مستحقات/, 'list_payables'],
   477→  [/آخر الحركات|اخر الحركات|سجل العمليات|آخر العمليات|اخر العمليات|العمليات المستنيه/, 'get_activity_history'],
   478→  [/سعر|بكام/, 'search_catalog'],
   479→]
   480→
   481→/** Sequence verbs that START a new request mid-message (never paid-clauses). */
   482→const WRITE_CUT_RE = /بعت|اشتريت|اشتري|سجل|حصلت|قبضت|استلمت|سدد|صرفت|مصروف|تسويه|عكسها|اعكس/
   483→
   484→/**
   485→ * Deterministic compound splitter — «ملخص اليوم ومين عليها فلوس وايه الناقص
   486→ * في المخزون» → 3 read sub-intents. Cuts at read-trigger keywords, sequence
   487→ * verbs (بعت/سجل/…) and connectors (و بعدين/كمان) — never inside a multi-item
   488→ * basket («بعت ٢ لبن و ١٠ بيض» stays ONE write). Zero LLM.
   489→ */
   490→function splitSubIntents(text: string, businessType: string): SubIntent[] {
   491→  const norm = toWesternDigits(normalizeAr(text))
   492→  if (norm.length < 8) return []
   493→  type Hit = { at: number; len: number; tool: string | null }
   494→  const hits: Hit[] = []
   495→  for (const [re, tool] of READ_SPLITTERS) {
   496→    const g = new RegExp(re.source, 'g')
   497→    let m: RegExpExecArray | null
   498→    while ((m = g.exec(norm)) != null) {
   499→      hits.push({ at: m.index, len: m[0].length, tool })
   500→      if (m.index === g.lastIndex) g.lastIndex++
   501→    }
   502→  }
   503→  // sequence verbs — only when a و/، actually precedes them (mid-message cut)
   504→  {
   505→    const g = new RegExp(WRITE_CUT_RE.source, 'g')
   506→    let m: RegExpExecArray | null
   507→    while ((m = g.exec(norm)) != null) {
   508→      const at = m.index
   509→      const prev = at > 0 ? norm[at - 1] : ''
   510→      const prev2 = at > 1 ? norm[at - 2] : ''
   511→      const prev3 = at > 2 ? norm[at - 3] : ''
   512→      const isBoundary =
   513→        prev === 'و' || prev === '،' || prev === ',' || prev === '؛' || (prev === ' ' && (prev2 === 'و' || prev2 === '،')) || (prev === ' ' && prev2 === ' ' && prev3 === 'و')
   514→      if (isBoundary) hits.push({ at, len: m[0].length, tool: null })
   515→      if (m.index === g.lastIndex) g.lastIndex++
   516→    }
   517→  }
   518→  // connectors «و بعدين» — cut at the connector itself
   519→  {
   520→    const g = /(بعدين|بعد كده|كمان)/g
   521→    let m: RegExpExecArray | null
   522→    while ((m = g.exec(norm)) != null) {
   523→      hits.push({ at: m.index, len: m[0].length, tool: null })
   524→      if (m.index === g.lastIndex) g.lastIndex++
   525→    }
   526→  }
   527→  hits.sort((a, b) => a.at - b.at)
   528→  if (hits.length === 0) return []
   529→
   530→  // dedupe overlaps + same-tool continuations, build cut points
   531→  const cuts: { at: number; tool: string | null }[] = []
   532→  let coveredEnd = -1
   533→  let currentTool: string | null = null
   534→  for (const h of hits) {
   535→    if (h.at < coveredEnd) continue // overlapping keyword (الخزنه inside وضع الخزنه)
   536→    if (h.tool != null && h.tool === currentTool) {
   537→      // same tool still talking — «سعر العنبر بكام؟» is ONE search, and
   538→      // «سعر العود وبكام المسك» is one question about two items (the catalog
   539→      // matcher is bidirectional-includes, so both still show in one card)
   540→      coveredEnd = Math.max(coveredEnd, h.at + h.len)
   541→      continue
   542→    }
   543→    // cut position: include a glued و in the PREVIOUS segment
   544→    const cutAt = h.at > 0 && norm[h.at - 1] === 'و' ? h.at - 1 : h.at
   545→    cuts.push({ at: cutAt, tool: h.tool })
   546→    coveredEnd = h.at + h.len
   547→    currentTool = h.tool
   548→  }
   549→  if (cuts.length === 0) return []
   550→
   551→  // build segments
   552→  const segments: { text: string; tool: string | null }[] = []
   553→  const seg0 = norm.slice(0, cuts[0].at).replace(/[\s،,؛]+$/, '').trim()
   554→  if (seg0.length > 0) segments.push({ text: seg0, tool: null })
   555→  for (let i = 0; i < cuts.length; i++) {
   556→    const end = i + 1 < cuts.length ? cuts[i + 1].at : norm.length
   557→    const seg = norm.slice(cuts[i].at, end).replace(/^[\s،,؛و]+/, '').replace(/[\s،,؛]+$/, '').trim()
   558→    if (seg.length > 0) segments.push({ text: seg, tool: cuts[i].tool })
   559→  }
   560→
   561→  // map segments → sub-intents
   562→  const subs: SubIntent[] = []
   563→  for (let i = 0; i < segments.length; i++) {
   564→    const seg = segments[i]
   565→    // politeness prefixes («قوللي», «اط») — skip chat-y tiny segments
   566→    if (seg.tool == null) {
   567→      if (i === 0 && seg.text.length <= 6) continue
   568→      const mapped = ruleBasedIntent(seg.text, businessType)
   569→      if (mapped.intent === 'read' && mapped.tool) subs.push({ ...mapped, intent: 'read', tool: mapped.tool, sourceText: seg.text })
   570→      else if (mapped.intent === 'write' && mapped.opType) subs.push({ ...mapped, intent: 'write', sourceText: seg.text })
   571→      continue
   572→    }
   573→    // tool-owned segment
   574→    if (seg.tool === 'search_catalog') {
   575→      const q = seg.text
   576→        .replace(/^(?:ايه|ايه هو|هو)?\s*(?:سعر|بكام)\s*/, '')
   577→        .replace(/[؟?].*$/, '')
   578→        .replace(/\s*(?:سعر|بكام|ايه|هو)\s*/g, ' ')
   579→        .replace(/^\s*(?:في|من)\s*/, '')
   580→        .trim()
   581→      subs.push({ intent: 'read', tool: 'search_catalog', fields: q.length >= 2 ? { query: q.slice(0, 40) } : {}, sourceText: seg.text })
   582→    } else {
   583→      subs.push({ intent: 'read', tool: seg.tool, fields: {}, sourceText: seg.text })
   584→    }
   585→  }
   586→
   587→  // merge CONSECUTIVE same-opType writes into one basket (multi-item scan)
   588→  const merged: SubIntent[] = []
   589→  for (const s of subs) {
   590→    const last = merged[merged.length - 1]
   591→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   592→      last.sourceText = `${last.sourceText} و ${s.sourceText}`
   593→      last.fields = { ...last.fields, ...s.fields }
   594→      continue
   595→    }
   596→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   597→      continue // duplicate read
   598→    }
   599→    merged.push({ ...s })
   600→  }
   601→  return merged
   602→}

   604→/** Normalize LLM multi-intents: validate, merge consecutive same-opType writes. */
   605→function normalizeMultiIntents(raws: RawIntentJson[], fullText: string): SubIntent[] {
   606→  const subs: SubIntent[] = []
   607→  for (const r of raws) {
   608→    if (r.intent === 'read' && r.tool && READ_TOOLS.has(r.tool)) {
   609→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   610→      subs.push({ intent: 'read', tool: r.tool as ReadTool, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   611→    } else if (r.intent === 'write' && r.opType && VALID_OP_TYPES.has(r.opType)) {
   612→      const source = typeof r.sourceText === 'string' && r.sourceText.trim().length > 0 ? r.sourceText.trim() : fullText
   613→      subs.push({ intent: 'write', opType: r.opType as OpType, fields: (r.fields as Record<string, unknown>) ?? {}, replyHint: r.replyHint, sourceText: source })
   614→    }
   615→  }
   616→  const merged: SubIntent[] = []
   617→  for (const s of subs) {
   618→    const last = merged[merged.length - 1]
   619→    if (s.intent === 'write' && last && last.intent === 'write' && last.opType === s.opType) {
   620→      // consecutive same-type writes = ONE basket; full-text scan finds the lines
   621→      last.sourceText = fullText
   622→      last.fields = { ...last.fields, ...s.fields }
   623→      continue
   624→    }
   625→    if (s.intent === 'read' && last && last.intent === 'read' && last.tool === s.tool && s.tool !== 'search_catalog') {
   626→      continue // duplicate read
   627→    }
   628→    merged.push(s)
   629→  }
   630→  return merged
   631→}

   634→// Proactive hints — when a read tool surfaced real counts worth attention,
   635→// the reply closes with one short heads-up (numbers only from real data).

   637→function tableFooterCount(cards: ResultCardDTO[]): number | null {
   638→  for (const c of cards) {
   639→    if (c.kind !== 'table' || !c.footer) continue
   640→    for (const f of c.footer) {
   641→      const m = String(f.value).match(/[\d٠-٩]+/)
   642→      if (m) {
   643→        const west = m[0].replace(/[٠-٩]/g, (d) => String('٠١٢٣٤٥٦٧٨٩'.indexOf(d)))
   644→        const n = Number(west)
   645→        if (Number.isFinite(n)) return n
   646→      }
   647→    }
   648→  }
   649→  return null
   650→}

   652→function proactiveHintFor(tool: string, cards: ResultCardDTO[]): string {
   653→  if (tool === 'list_low_stock') {
   654→    const n = tableFooterCount(cards)
   655→    if (n != null && n > 0) return `قفل جملة قصيرة تنبّه على ${n} صنف ناقص من غير ما تكرر الأرقام كلها.`
   656→  }
   657→  if (tool === 'list_expiring') {
   658→    const n = tableFooterCount(cards)
   659→    if (n != null && n > 0) return `قفل جملة قصيرة تنبّه على ${n} صنف صلاحيته قربت تخلص من غير ما تكرر الأرقام كلها.`
   660→  }
   661→  if (tool === 'get_activity_history') {
   662→    const pending = cards
   663→      .flatMap((c) => (c.kind === 'table' ? c.rows : []))
   664→      .filter((r) => String(r.status ?? '').includes('مستنية')).length
   665→    if (pending > 0) return `قفل جملة قصيرة تنبّه على ${pending} عملية مستنية اعتماد من غير ما تكرر الأرقام كلها.`
   666→  }
   667→  return ''
   668→}

   670→/**
   671→ * Deterministic reply for read turns when the LLM reply stream fails
   672→ * (rate-limit / network). NEVER generic — quotes the REAL numbers the tools
   673→ * just fetched, so the owner still gets a useful answer with the data below.
   674→ */
   675→function deterministicReadReply(cards: ResultCardDTO[]): string {
   676→  for (const c of cards) {
   677→    if (c.kind === 'kv' && c.rows.length > 0) {
   678→      const parts = c.rows.slice(0, 4).map((r) => `${r.label} ${r.value}`).join(' · ')
   679→      return `دي أرقام «${c.title}» من سجلك: ${parts}.`
   680→    }
   681→  }
   682→  const table = cards.find((c) => c.kind === 'table') as Extract<ResultCardDTO, { kind: 'table' }> | undefined
   683→  if (table) {
   684→    const n = table.rows.length
   685→    return `دي «${table.title}» من سجلك — ${toArabicNum(n)} ${n === 1 ? 'صف' : n === 2 ? 'صفين' : 'صفوف'}، التفاصيل في الكارت تحت.`
   686→  }
   687→  const text = cards.find((c) => c.kind === 'text') as Extract<ResultCardDTO, { kind: 'text' }> | undefined
   688→  if (text && text.body.trim()) return text.body.trim().slice(0, 220)
   689→  return 'دي النتايج من سجل الشغل، التفاصيل في الكارت تحت.'
   690→}

   692→/** Deterministic numbered multi-intent reply — answers EVERY sub-request, no LLM. */
   693→function deterministicMultiReply(parts: string[], overflow: number): string {
   694→  const sections = parts.map((p, i) => `${AR_ORDINALS[i] ?? i + 1}. ${p}`)
   695→  const overflowNote = overflow > 0 ? `\n\n${toArabicNum(overflow)} ${overflow === 1 ? 'طلب' : 'طلبات'} تانية بعد دول — قولي نكمل فيهم؟` : ''
   696→  const closer = sections.some((s) => s.includes('مسودة') || s.includes('مستنية'))
   697→    ? '\n\nالمسودات تحت مستنية اعتمادك.'
   698→    : '\n\nتحب تفاصيل أكتر على أي جزء؟'
   699→  return sections.join('\n\n') + overflowNote + closer
   700→}

   702→/** Multi-intent LLM reply must answer every part with numbered sections — verify. */
   703→function replyAnswersAllParts(reply: string, expected: number): boolean {
   704→  const numbered = (reply.match(/^[٠-٩][.،)]/gm) ?? []).length + (reply.match(/^[1-4][.،)]/gm) ?? []).length
   705→  if (expected >= 2 && numbered < 2) return false
   706→  return true
   707→}

   722→  summary: string
   723→  card: ActionCardDTO
   724→  missing?: string
   725→  failMessage?: string
   726→  /** Multi-item drafts: lines that could NOT be drafted (stock/unit), shown to the user. */
   727→  issues?: string[]
   728→}

   730→/**
   731→ * Deterministic Egyptian-phrasing party scan for sales/purchases on the
   732→ * rule-fallback path (no LLM): «بع لبن لأحمد», «بيع للعميل أحمد»,
   733→ * «اشتريت من شركة الألبان». Only a preposition/role-word directly attached
   734→ * to a name-word counts, so an item brand like «قهوة أحمد» never becomes a
   735→ * customer — the ل/من marker is what marks the ROLE.
   736→ */
   737→function findPartyMention<T extends { name: string }>(
   738→  text: string,
   739→  parties: T[],
   740→  marker: 'ل' | 'من' = 'ل',
   741→): T | null {
   742→  const norm = normalizeAr(text)
   743→  const hasRoleWord = marker === 'ل' ? norm.includes('عميل') || norm.includes('زبون') : norm.includes('مورد')
   744→  for (const p of parties) {
   745→    const words = normalizeAr(p.name)
   746→      .split(' ')
   747→      .flatMap((w) => (w.startsWith('ال') && w.length > 4 ? [w, w.slice(2)] : [w]))
   748→      .filter((w) => w.length >= 2)
   749→    for (const w of words) {
   750→      // glued («لأحمد» / «منشركة») or spaced («من محمد») marker
   751→      if (norm.includes(marker + w) || norm.includes(marker + ' ' + w)) return p
   752→      // «العميل أحمد» — role word anywhere in the message + the name standalone
   753→      if (w.length >= 3 && hasRoleWord && norm.includes(' ' + w + ' ')) return p
   754→    }
   755→  }
   756→  return null
   757→}

   759→function fieldsToCard(ref: string, type: OpType, risk: Risk, summary: string, fields: { label: string; value: string; hint?: string }[], effects: { label: string; value: string; tone?: 'good' | 'bad' | 'neutral' }[], dangerNote?: string): ActionCardDTO {
   760→  return {
   761→    kind: 'draft',
   762→    ref,
   763→    type,
   764→    risk,
   765→    title: TYPE_AR[type] ?? type,
   766→    summary,
   767→    fields,
   768→    effects,
   769→    status: 'AWAITING_APPROVAL',
   770→    dangerNote,
   771→  }
   772→}

   774→/** عطارة «البيع بالمبلغ»: «بـ20 جنيه فلفل أسود» → (amount, in EGP) or null.
   775→ * Runs on WESTERN-digit normalized text (normalizeAr already ate the ـ tatweel
   776→ * in «بـ20»). Requires a real money word (جنيه/ج) after the number so a bare
   777→ * quantity («بعت 20 فلفل») can never fire it. */
   778→function matchReverseSellAmount(normWestern: string): number | null {
   779→  const m = normWestern.match(/(?:^|\s)(?:بال|بمبلغ|ب)([0-9]+(?:[.,][0-9]+)?)\s*(?:جنيه|جنيهات|ج)(?=\s|$)/)
   780→  if (!m) return null
   781→  const amount = parseFloat(m[1].replace(',', '.'))
   782→  return Number.isFinite(amount) && amount > 0 ? amount : null
   783→}
   784→
   785→/** Remove the «بـN جنيه» clause so the money number can never masquerade as a
   786→ * weight in ANY quantity parser downstream (segment scan / parseQuantity). */
   787→function stripReverseSellAmount(normWestern: string): string {
   788→  return normWestern.replace(/(?:^|\s)(?:بال|بمبلغ|ب)[0-9]+(?:[.,][0-9]+)?\s*(?:جنيه|جنيهات|ج)(?=\s|$)/g, ' ')
   789→}

   791→/** Paid amount: LLM field → explicit «دفع 20» clause → آجل = 0 → default cash (total). */
   792→function resolvePaid(norm: string, f: Record<string, unknown>, total: number): number {
   793→  if (typeof f.paid === 'number' && f.paid > 0) return Math.min(Math.round(f.paid * 100), total)
   794→  const paidWord = norm.match(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?([٠-٩0-9.,]+|\S+)?/)
   795→  if (paidWord?.[1]) {
   796→    const n = parseQuantity(paidWord[1], null).qty
   797→    if (n != null && n > 0) return Math.min(Math.round(n * 100), total)
   798→  }
   799→  if (/(?:بال)?اجل|علي الحساب|نسيون/.test(norm)) return 0
   800→  return total
   801→}

   803→async function resolveWrite(ctx: TurnContext, intent: ExtractedIntent, emit: TurnEmit, memCtx: MemoryContext): Promise<ResolvedWrite> {
   804→  const { business, user } = ctx
   805→  const type = (intent.opType ?? 'sale') as OpType
   806→  const f = intent.fields ?? {}
   807→  const items: Item[] = await db.item.findMany({ where: { businessId: business.id } })
   808→  const customers: Customer[] = await db.customer.findMany({ where: { businessId: business.id } })
   809→  const suppliers: Supplier[] = await db.supplier.findMany({ where: { businessId: business.id } })
   810→
   811→  const text = ctx.text
   812→  const norm = normalizeAr(text)
   813→  const normW = toWesternDigits(norm)
   814→
   815→  switch (type) {
   816→    case 'sale':
   817→    case 'purchase': {
   818→      // Paid clauses are stripped BEFORE quantity parsing so a payment like
   819→      // «دفع ٢٠ بس» can never be misread as the sold quantity.
   820→      const textNoPaid = norm.replace(/(?:دفع|دفعت|خلاص|ادفع)\s*(?:الي\s*)?[٠-٩0-9][٠-٩0-9.,]*\s*(?:بس|فقط|تمام)?/g, ' ')
   821→      // ---- عطارة reverse-selling («بعت بـ20 جنيه فلفل») ----
   822→      // The money clause MUST be detected and stripped BEFORE any quantity
   823→      // parsing: a bare «20» in «بـ20 جنيه» is the AMOUNT, not a weight. The
   824→      // deterministic scanners and the LLM both tend to read it as qty=20 —
   825→      // the pattern match on «بـN جنيه» is the ground truth.
   826→      const reverseAmountEgp = business.type === 'attar' && type === 'sale' ? matchReverseSellAmount(normW) : null
   827→      const scanText = reverseAmountEgp != null ? stripReverseSellAmount(toWesternDigits(textNoPaid)) : textNoPaid
   828→      // ---- Multi-item scan (deterministic — works with or without the LLM) ----
   829→      // «سجّل بيع ٢ لبن و ١٠ بيض» → segments → per-item quantity. When TWO or
   830→      // more distinct catalog items are mentioned we draft ONE multi-line
   831→      // operation (a single basket the cashier approves once), like the real POS.
   832→      // Longest-alias-first + overlap rejection: «بانادول» alone can never fan
   833→      // out into «بانادول أكتيف» + «بانادول شرائط» — the more specific name wins.
   834→      const segments = scanText.split(/\s*(?:،|,|\+|؛|;)\s*|\s+و\s+|\s+و(?=[٠-٩0-9])/).filter(Boolean)
   835→      const byItem = new Map<string, { item: Item; qty: number | null; unit: string | null }>()
   836→      for (const seg of segments) {
   837→        const segNorm = normalizeAr(seg)
   838→        const hits: { item: Item; token: string; at: number }[] = []
   839→        for (const it of items) {
   840→          // canonical names + column aliases + LEXICON dialect aliases (boost)
   841→          const lexAl = memCtx.lexicon.filter((l) => l.entityRef === it.id).map((l) => l.alias)
   842→          const names = [it.name, ...(it.aliases ? it.aliases.split(',').map((a) => a.trim()) : []), ...lexAl]
   843→          let best: { token: string; at: number } | null = null
   844→          for (const n of names) {
   845→            const nn = normalizeAr(n)
   846→            if (nn.length >= 3) {
   847→              const at = segNorm.indexOf(nn)
   848→              if (at >= 0 && (!best || nn.length > best.token.length)) best = { token: nn, at }
   849→            }
   850→          }
   851→          if (best) hits.push({ item: it, ...best })
   852→        }
   853→        hits.sort((a, b) => b.token.length - a.token.length)
   854→        const taken: Array<[number, number]> = []
   855→        for (const h of hits) {
   856→          if (taken.some(([s, e]) => h.at < e && h.at + h.token.length > s)) continue
   857→          taken.push([h.at, h.at + h.token.length])
   858→          const q = parseQuantity(seg, null)
   859→          const prev = byItem.get(h.item.id)
   860→          byItem.set(h.item.id, {
   861→            item: h.item,
   862→            qty: q.qty != null && q.qty > 0 ? (prev?.qty ?? 0) + q.qty : prev?.qty ?? null,
   863→            unit: prev?.unit ?? unitToCode(q.unit) ?? null,
   864→          })
   865→        }
   866→      }
   867→      const multi = byItem.size >= 2
   868→
   869→      if (multi) {
   870→        // ---- Multi-line draft: one basket, real prices per line ----
   871→        const lines: { item: Item; qty: number; unitPrice: number; total: number }[] = []
   872→        const issues: string[] = []
   873→        for (const { item, qty, unit } of byItem.values()) {
   874→          const rawUnit = unit ?? item.unit // default = the item's own canonical unit
   875→          const baseQty = convertToBaseUnit(qty ?? 1, rawUnit, item.unit)
   876→          if (baseQty == null || !(baseQty > 0)) {
   877→            issues.push(`الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})`)
   878→            continue
   879→          }
   880→          const qtyN = Math.round(baseQty * 1000) / 1000
   881→          if (type === 'sale' && qtyN > item.stockQty) {
   882→            issues.push(`المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — انت طلبت ${fmtQty(qtyN, arUnit(item.unit))}`)
   883→            continue
   884→          }
   885→          const unitPrice = type === 'sale' ? item.sellPrice : item.costPrice
   886→          lines.push({ item, qty: qtyN, unitPrice, total: Math.round(qtyN * unitPrice) })
   887→        }
   888→        if (lines.length === 0) {
   889→          return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: issues.join('؛ ') || 'الكمية قد إيه لكل صنف؟', issues }
   890→        }
   891→        const total = lines.reduce((s, l) => s + l.total, 0)
   892→        // Paid (clamped to the basket total; آجل = 0)
   893→        const paid = resolvePaid(norm, f, total)
   894→        // Party (lexicon first, then LLM field, then deterministic phrasing)
   895→        const lexCustomer = type === 'sale' ? lexiconResolve(norm, memCtx.lexicon, customers)?.entity : null
   896→        if (lexCustomer) memCtx.referencedIds.push(...memCtx.lexicon.filter((l) => l.entityRef === lexCustomer.id).map((l) => l.id))
   897→        const customer = lexCustomer ?? (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? (type === 'sale' ? findPartyMention(text, customers) : null)
   898→        const supplier = (f.supplierName ? matchEntity(String(f.supplierName), suppliers)?.entity : null) ?? (type === 'purchase' ? findPartyMention(text, suppliers, 'من') : null)
   899→
   900→        const payload: Record<string, unknown> = {
   901→          lines: lines.map((l) => ({ itemId: l.item.id, name: l.item.name, qty: l.qty, unit: l.item.unit, unitPrice: l.unitPrice, total: l.total })),
   902→          total,
   903→          paid,
   904→          ...(type === 'sale' && customer ? { customerId: customer.id, customerName: customer.name } : {}),
   905→          ...(type === 'purchase' && supplier ? { supplierId: supplier.id, supplierName: supplier.name } : {}),
   906→        }
   907→        const ref = `WKL-DRAFT` // replaced by real ref from createDraft
   908→        const risk = computeRisk(type, total)
   909→        const summary = `${type === 'sale' ? 'بيع' : 'شرا'} ${lines.map((l) => `${fmtQty(l.qty, arUnit(l.item.unit))} ${l.item.name}`).join(' + ')} بـ${egp(total)}${type === 'sale' && customer ? ` — ${customer.name}` : ''}${type === 'purchase' && supplier ? ` — من ${supplier.name}` : ''}`
   910→        // BUSINESS_DNA credit-limit enforcement (آجل on a capped party)
   911→        let dangerNote: string | undefined
   912→        let creditField: { label: string; value: string; hint?: string } | undefined
   913→        if (type === 'sale' && customer && total - paid > 0 && memCtx.rules.length > 0) {
   914→          const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
   915→          if (check) {
   916→            dangerNote = check.warning
   917→            creditField = { label: '⚠️ قانون المحل', value: `هيوصل رصيد ${customer.name} لـ${egp(check.projectedPiastres)} — القانون: ${egp(check.limitPiastres)}` }
   918→            memCtx.referencedIds.push(check.rule.id)
   919→          }
   920→        }
   921→        const card = fieldsToCard(ref, type, risk, summary, [
   922→          ...lines.map((l) => ({ label: l.item.name, value: `${fmtQty(l.qty, arUnit(l.item.unit))} × ${egp(l.unitPrice)} = ${egp(l.total)}` })),
   923→          { label: 'الإجمالي', value: egp(total) },
   924→          { label: 'مدفوع', value: egp(paid) },
   925→          ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
   926→          ...(type === 'sale' && customer ? [{ label: 'العميل', value: customer.name }] : []),
   927→          ...(type === 'purchase' && supplier ? [{ label: 'المورد', value: supplier.name }] : []),
   928→          ...(creditField ? [creditField] : []),
   929→          ...issues.map((i) => ({ label: '⚠ مش ضمن المسودة', value: i })),
   930→        ], [
   931→          ...lines.map((l) => ({ label: `مخزون ${l.item.name}`, value: `${fmtQty(l.item.stockQty, arUnit(l.item.unit))} ← ${fmtQty(l.item.stockQty + (type === 'sale' ? -l.qty : l.qty), arUnit(l.item.unit))}`, tone: 'neutral' as const })),
   932→          { label: 'الخزنة', value: egp(type === 'sale' ? paid : -paid), tone: type === 'sale' ? 'good' as const : 'bad' as const },
   933→        ], dangerNote ?? RISK_META[risk].hint)
   934→        return { ok: true, opType: type, payload, summary, card, issues: issues.length > 0 ? issues : undefined }
   935→      }
   936→
   937→      // ---- Single-item draft (original path) ----
   938→      // Item: lexicon (dialect alias) → LLM field → text mention → fuzzy whole-text match
   939→      const rawItemName = String(f.itemName ?? '')
   940→      const mentioned = byItem.size === 1 ? [...byItem.values()][0] : null
   941→      const lexHit = lexiconResolve(norm, memCtx.lexicon, items)
   942→      const item = lexHit?.entity ?? (rawItemName ? matchEntity(rawItemName, items)?.entity : null) ?? mentioned?.item ?? matchEntity(text, items)?.entity ?? null
   943→      if (lexHit) {
   944→        memCtx.referencedIds.push(lexHit.entry.id)
   945→      } else if (item) {
   946→        // LEXICON learning: a dialect token that overlaps the canonical name
   947→        const alias = learnableAlias(text, item)
   948→        if (alias) await saveLexiconMemory({ businessId: business.id, userId: user.id }, alias, item)
   949→      }
   950→      if (!item) {
   951→        return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الصنف', failMessage: `مش لاقي الصنف — ممكن تكتبه بالاسم اللي في الكتالوج؟` }
   952→      }
   953→
   954→      // ---- عطارة reverse selling: «بعت بـ20 جنيه فلفل أسود» → وزن بالجرام ----
   955→      if (reverseAmountEgp != null) {
   956→        // scanText already has the money clause stripped, so ANY number still
   957→        // left in it is a real weight («بعت 100 جرام كركديه بـ20 جنيه» → qty wins)
   958→        const qtyAfterStrip = parseQuantity(scanText, null).qty
   959→        const digitsAfterStrip = /[0-9]/.test(scanText)
   960→        // The LLM tends to echo the money as qty — only a qty that ALSO has a
   961→        // real number left in the stripped text (or a word-number like نص) counts
   962→        const llmQtyIsReal =
   963→          typeof f.qty === 'number' && f.qty > 0 && f.qty !== reverseAmountEgp && (digitsAfterStrip || (f.unit && unitToCode(String(f.unit)) != null))
   964→        const noExplicitQty = qtyAfterStrip == null && mentioned?.qty == null && !llmQtyIsReal
   965→        if (noExplicitQty) {
   966→          if (item.unit !== 'gram' && item.unit !== 'kg') {
   967→            return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `«${item.name}» بيتاع بـ${arUnit(item.unit)} مش بالوزن — قولي الكمية بالوحدة بتاعته وأنا أحسبها لك` }
   968→          }
   969→          const amountEgp = reverseAmountEgp
   970→          const perGram = item.unit === 'gram' ? item.sellPrice : Math.round(item.sellPrice / 1000) // kg → per gram
   971→          const amountPiastres = Math.round(amountEgp * 100)
   972→          if (perGram > 0 && amountPiastres > 0) {
   973→            const grams = Math.round(amountPiastres / perGram)
   974→            if (grams < 1) {
   975→              return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'المبلغ', failMessage: `${egp(amountPiastres)} أقل من سعر الجرام (${egp(perGram)}) — الزبون ده عايز ياخد حاجة تانية؟` }
   976→            }
   977→            if (grams > item.stockQty) {
   978→              return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `المتاح من ${item.name} ${fmtQty(item.stockQty, arUnit(item.unit))} بس — ${egp(amountPiastres)} يعني ${fmtQty(grams, 'جرام')}` }
   979→            }
   980→            // revenue = the money the customer asked to buy with (integer piastres)
   981→            const total = amountPiastres
   982→            const paid = resolvePaid(norm, f, total)
   983→            const customer = (f.customerName ? matchEntity(String(f.customerName), customers)?.entity : null) ?? findPartyMention(text, customers)
   984→            const payload: Record<string, unknown> = {
   985→              itemId: item.id,
   986→              itemName: item.name,
   987→              qty: grams,
   988→              unit: 'gram',
   989→              unitPrice: perGram,
   990→              total,
   991→              paid,
   992→              reverseSell: true,
   993→              requestedAmount: amountPiastres,
   994→              ...(customer ? { customerId: customer.id, customerName: customer.name } : {}),
   995→            }
   996→            const risk = computeRisk(type, total)
   997→            const summary = `بيع بالوزن: ${fmtQty(grams, 'جرام')} ${item.name} بمبلغ ${egp(total)}${customer ? ` — ${customer.name}` : ''}`
   998→            let dangerNote: string | undefined
   999→            let creditField: { label: string; value: string; hint?: string } | undefined
  1000→            if (customer && total - paid > 0 && memCtx.rules.length > 0) {
  1001→              const check = creditLimitFor(memCtx.rules, customer, customer.balance + (total - paid))
  1002→              if (check) {
  1003→                dangerNote = check.warning
  1004→                creditField = { label: '⚠️ قانون المحل', value: `هيوصل رصيد ${customer.name} لـ${egp(check.projectedPiastres)} — القانون: ${egp(check.limitPiastres)}` }
  1005→                memCtx.referencedIds.push(check.rule.id)
  1006→              }
  1007→            }
  1008→            const card = fieldsToCard('', type, risk, summary, [
  1009→              { label: 'الصنف', value: item.name },
  1010→              { label: 'الوزن المحسوب', value: fmtQty(grams, 'جرام'), hint: `⚖️ اوزن ${fmtQty(grams, 'جرام')} للزبون` },
  1011→              { label: 'سعر الجرام', value: egp(perGram) },
  1012→              { label: 'المبلغ (بيع بـ)', value: egp(total) },
  1013→              { label: 'مدفوع', value: egp(paid) },
  1014→              ...(total - paid > 0 ? [{ label: 'باقي (آجل)', value: egp(total - paid) }] : []),
  1015→              ...(customer ? [{ label: 'العميل', value: customer.name }] : []),
  1016→              ...(creditField ? [creditField] : []),
  1017→            ], [
  1018→              { label: `مخزون ${item.name}`, value: `${fmtQty(item.stockQty, arUnit(item.unit))} ← ${fmtQty(item.stockQty - grams, arUnit(item.unit))}`, tone: 'neutral' },
  1019→              { label: 'الخزنة', value: egp(paid), tone: 'good' },
  1020→            ], dangerNote ?? RISK_META[risk].hint)
  1021→            return { ok: true, opType: type, payload, summary, card }
  1022→          }
  1023→        }
  1024→      }
  1025→
  1026→      // Quantity: LLM → mention → text (paid clause already stripped) → default 1
  1027→      // («بع لبن» = one unit of the item's own canonical kind — the cashier's shorthand)
  1028→      // (when a reverse-sell amount was detected but a real qty survived the
  1029→      // strip, the normal path runs — the explicit weight is the ground truth)
  1030→      const textQ = parseQuantity(reverseAmountEgp != null ? scanText : textNoPaid, null)
  1031→      const llmQty = typeof f.qty === 'number' && f.qty > 0 ? f.qty : null
  1032→      // reverse-sell candidate: an LLM qty equal to the money is the echo — drop it
  1033→      const qtySource = reverseAmountEgp != null && llmQty === reverseAmountEgp ? null : llmQty
  1034→      const qty = qtySource ?? mentioned?.qty ?? textQ.qty ?? 1
  1035→      if (qty <= 0) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, missing: 'الكمية', failMessage: 'الكمية قد إيه بالظبط؟' }
  1036→      // Unit: LLM → mention → text word → the item's OWN canonical unit (الافتراضي)
  1037→      const rawUnit = unitToCode(f.unit as string) ?? mentioned?.unit ?? unitToCode(textQ.unit) ?? item.unit
  1038→      const baseQty = convertToBaseUnit(qty, rawUnit, item.unit)
  1039→      if (baseQty == null || !(baseQty > 0)) return { ok: false, opType: type, payload: {}, summary: '', card: null as never, failMessage: `الوحدة «${rawUnit}» مش متوافقة مع وحدة ${item.name} (${arUnit(item.unit)})` }
  1040→      const qtyN = Math.round(baseQty * 1000) / 1000
  1041→      if (type === 'sale' && qtyN > item.stockQty) {

  1304→/**
  1305→ * The public turn runner — a telemetry shell around the real engine:
  1306→ * measures total latency + TTFT (first delta/step on the wire), resets the
  1307→ * LLM health flags (retry/fallback ladder), writes exactly ONE UsageEvent per
  1308→ * turn {kind, label = أول ٤٠ حرف من رسالة المستخدم, latencyMs, ttftMs,
  1309→ * tokens ≈ len/3, fallback}, plus a 'verify' event when verification did work.
  1310→ */
  1311→export async function runAgentTurn(ctx: TurnContext, emit: TurnEmit): Promise<TurnResult> {
  1312→  ensurePrismaFresh() // dev-server prisma hot-reload seam (no-op on fresh processes)
  1313→  const t0 = Date.now()
  1314→  let ttftMs: number | null = null
  1315→  let sawFirstToken = false
  1316→  const meta: TurnMeta = { kind: 'chat', fallback: false, notesJotted: 0 }
  1317→  llmHealth.retried = false
  1318→  llmHealth.fellBack = false
  1319→  const trackedEmit: TurnEmit = (ev) => {
  1320→    if (!sawFirstToken && (ev.event === 'delta' || ev.event === 'step')) {
  1321→      sawFirstToken = true
  1322→      ttftMs = Date.now() - t0
  1323→    }
  1324→    emit(ev)
  1325→  }
  1326→  let result: TurnResult
  1327→  try {
  1328→    result = await runAgentTurnInner(ctx, trackedEmit, meta)
  1329→  } catch (err) {
  1330→    // the engine itself exploded — record the failure, then let the route's
  1331→    // catch send the SSE error (the user never sees a half-turn silently die)
  1332→    void trackTurnUsage(ctx.business.id, ctx.user.id, meta.kind, ctx.text, {
  1333→      latencyMs: Date.now() - t0,
  1334→      ttftMs,
  1335→      tokens: 0,
  1336→      fallback: true,
  1337→    })
  1338→    throw err
  1339→  }
  1340→  void trackTurnUsage(ctx.business.id, ctx.user.id, meta.kind, ctx.text, {
  1341→    latencyMs: Date.now() - t0,
  1342→    ttftMs,
  1343→    tokens: Math.ceil(result.replyText.length / 3),
  1344→    fallback: meta.fallback || llmHealth.retried || llmHealth.fellBack,
  1345→  })
  1346→  if (result.verified && (result.verified.status === 'corrected' || result.verified.checks.length > 0)) {
  1347→    void trackKindEvent(ctx.business.id, ctx.user.id, 'verify', result.verified.status)
  1348→  }
  1349→  result.notesJotted = meta.notesJotted
  1350→  return result
  1351→}

  1353→async function runAgentTurnInner(ctx: TurnContext, emit: TurnEmit, meta: TurnMeta): Promise<TurnResult> {
  1354→  const { business, user, text } = ctx

  1356→  const model = await resolveModel(ctx.modelKey)
  1357→  const modelLabel = modelLabelFor(model)
  1358→
  1359→  // Load context once (memories raised to 8; pending ops count for the persona;
  1360→  // R15: «ورقة وكيل» scratchpad notes for prompt injection)
  1361→  const [items, customers, suppliers, memories, pendingCount, notesSection] = await Promise.all([
