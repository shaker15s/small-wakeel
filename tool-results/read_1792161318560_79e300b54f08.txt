   107→function ModelRow({
   108→  selected,
   109→  onSelect,
   110→  disabled,
   111→  children,
   112→  ariaLabel,
   113→}: {
   114→  selected: boolean
   115→  onSelect: () => void
   116→  disabled?: boolean
   117→  children: React.ReactNode
   118→  ariaLabel: string
   119→}) {
   120→  return (
   121→    <button
   122→      type="button"
   123→      onClick={onSelect}
   124→      disabled={disabled}
   125→      aria-label={ariaLabel}
   126→      aria-pressed={selected}
   127→      className={cn(
   128→        'press focus-warm flex min-h-11 w-full items-center gap-2.5 rounded-lg border px-3 py-2.5 text-start transition-colors',
   129→        selected
   130→          ? 'border-wakil-amber/50 bg-accent'
   131→          : 'border-transparent hover:bg-muted disabled:cursor-not-allowed disabled:opacity-60',
   132→      )}
   133→    >
   134→      {children}
   135→      {selected && (
   136→        <motion.span
   137→          initial={{ scale: 0.3, opacity: 0 }}
   138→          animate={{ scale: 1, opacity: 1 }}
   139→          transition={{ type: 'spring', stiffness: 500, damping: 24 }}
   140→          className="ms-auto flex shrink-0"
   141→        >
   142→          <Check className="size-4 text-wakil-amber" aria-hidden />
   143→        </motion.span>
   144→      )}
   145→    </button>
   146→  )
   147→}

   153→  const setModel = useWakil((s) => s.setModel)
   154→
   155→  // — queries

   159→  })
   160→  const tasksQ = useQuery({
   161→    queryKey: QUERY_KEYS.tasks(businessId),

   166→    queryFn: () => api.memories(businessId),
   167→  })
   168→  const providers = providersQ.data?.providers ?? []
   169→  // api.tasks types the payload as ScheduledTaskDTO[] but the route actually
   170→  // returns { tasks: [...] } — accept both shapes until lib/api.ts is aligned.
   171→  const tasks: ScheduledTaskDTO[] = Array.isArray(tasksQ.data)
   172→    ? tasksQ.data

   175→  // — dialog state (إضافة مزوّد)

   179→  const [pApiKey, setPApiKey] = useState('')

   181→  const [addError, setAddError] = useState<string | null>(null)
   182→  const [fetched, setFetched] = useState<ProviderDTO | null>(null)
   183→  const [edits, setEdits] = useState<EditableModel[]>([])
   184→  const [saveBusy, setSaveBusy] = useState(false)

   188→  // — tasks form

   190→  const [taskTool, setTaskTool] = useState('list_low_stock')
   191→  const [taskRec, setTaskRec] = useState<'daily' | 'weekly' | 'monthly'>('daily')
   192→  const [taskBusy, setTaskBusy] = useState(false)
   193→  const [tickBusy, setTickBusy] = useState(false)
   194→
   195→  // — memory form
   196→  const [memText, setMemText] = useState('')
   197→  const [memKind, setMemKind] = useState<'fact' | 'preference'>('fact')
   198→  const [memBusy, setMemBusy] = useState(false)

   200→  function resetDialog() {
   201→    setAddOpen(false)
   202→    setPName('')
   203→    setPBaseUrl('')
   204→    setPApiKey('')
   205→    setAddError(null)
   206→    setFetched(null)
   207→    setEdits([])
   208→  }
   209→
   210→  async function handleTestProvider() {
   211→    setAddError(null)
   212→    if (!pName.trim() || !pBaseUrl.trim() || !pApiKey.trim()) {
   213→      setAddError('املأ الاسم والرابط والمفتاح الأول')
   214→      return
   215→    }
   216→    setTestBusy(true)
   217→    try {
   218→      const { provider } = await api.addProvider({
   219→        name: pName.trim(),
   220→        baseUrl: pBaseUrl.trim(),
   221→        apiKey: pApiKey.trim(),
   222→      })
   223→      setFetched(provider)
   224→      setEdits(
   225→        provider.models.map((m) => ({
   226→          remoteId: m.remoteId,
   227→          label: m.label,
   228→          tier: m.tier,
   229→          enabled: m.enabled,
   230→          selected: true,
   231→        })),
   232→      )
   233→      toast({
   234→        title: 'وصلنا للمزوّد',
   235→        description: `جبنا ${arNum(provider.models.length)} موديل — اختار اللي يعجبك`,
   236→      })
   237→    } catch (e) {
   238→      setAddError(e instanceof Error ? e.message : 'حصل خطأ غير متوقع')
   239→    } finally {
   240→      setTestBusy(false)
   241→    }
   242→  }
   243→
   244→  async function handleSaveModels() {
   245→    if (!fetched) return
   246→    setSaveBusy(true)
   247→    setAddError(null)
   248→    try {
   249→      const chosen = edits.filter((m) => m.selected)
   250→      await api.saveProviderModels(
   251→        fetched.id,
   252→        chosen.map(({ remoteId, label, tier, enabled }) => ({ remoteId, label, tier, enabled })),
   253→      )
   254→      await qc.invalidateQueries({ queryKey: QUERY_KEYS.providers })
   255→      toast({ title: 'تم الحفظ', description: `حفظنا ${arNum(chosen.length)} موديل مع «${fetched.name}»` })
   256→      resetDialog()
   257→    } catch (e) {
   258→      setAddError(e instanceof Error ? e.message : 'حصل خطأ غير متوقع')
   259→    } finally {
   260→      setSaveBusy(false)
   261→    }
   262→  }
   263→
   264→  async function handleToggleModel(provider: ProviderDTO, model: ModelConfigDTO, next: boolean) {
   265→    const rowKey = `${provider.id}:${model.id}`
   266→    setBusyModelKey(rowKey)
   267→    try {
   268→      await api.saveProviderModels(
   269→        provider.id,
   270→        provider.models.map((m) => ({
   271→          remoteId: m.remoteId,
   272→          label: m.label,
   273→          tier: m.tier,
   274→          enabled: m.id === model.id ? next : m.enabled,
   275→        })),
   276→      )
   277→      await qc.invalidateQueries({ queryKey: QUERY_KEYS.providers })
   278→      toast({ title: next ? 'فعّلنا الموديل' : 'وقفنا الموديل', description: model.label })
   279→    } catch (e) {
   280→      toast({
   281→        variant: 'destructive',
   282→        title: 'مش قادرين نحفظ',
   283→        description: e instanceof Error ? e.message : undefined,
   284→      })
   285→    } finally {
   286→      setBusyModelKey(null)
   287→    }
   288→  }
   289→
   290→  async function handleDeleteProvider(provider: ProviderDTO) {
   291→    setDeletingId(provider.id)
   292→    try {
   293→      const res = await api.deleteProvider(provider.id)
   294→      if (!res.ok) {
   295→        const body = (await res.json().catch(() => ({}))) as { error?: string }
   296→        throw new Error(body.error ?? 'فشل المسح')
   297→      }
   298→      await qc.invalidateQueries({ queryKey: QUERY_KEYS.providers })
   299→      toast({ title: 'مسحنا المزوّد', description: provider.name })
   300→    } catch (e) {
   301→      toast({
   302→        variant: 'destructive',
   303→        title: 'مش قادرين نمسح المزوّد',
   304→        description: e instanceof Error ? e.message : undefined,
   305→      })
   306→    } finally {
   307→      setDeletingId(null)
   308→    }
   309→  }
   310→
   311→  function handleSelectModel(key: string, label: string) {
   312→    setModel(key)
   313→    toast({ title: 'بدّلنا الموديل', description: label })
   314→  }
   315→
   316→  async function handleCreateTask() {

   318→      toast({ variant: 'destructive', title: 'اكتب عنوان المهمة الأول' })

   323→      await api.createTask({
   324→        businessId,
   325→        userId,
   326→        title: taskTitle.trim(),
   327→        tool: taskTool,
   328→        recurrence: taskRec,
   329→        whenLabel: RECURRENCE_WHEN[taskRec] ?? 'كل يوم الصبح',

   332→      setTaskTitle('')
   333→      toast({ title: 'أضفنا المهمة', description: taskTitle.trim() })
   334→    } catch (e) {
   335→      toast({
   336→        variant: 'destructive',
   337→        title: 'مش قادرين نضيف المهمة',
   338→        description: e instanceof Error ? e.message : undefined,
   339→      })
   340→    } finally {
   341→      setTaskBusy(false)
   342→    }
   343→  }
   344→
   345→  async function handleTickTasks() {
   346→    setTickBusy(true)
   347→    try {
   348→      const r = await api.tickTasks()
   349→      await qc.invalidateQueries({ queryKey: QUERY_KEYS.tasks(businessId) })
   350→      toast({
   351→        title: r.due === 0 ? 'مفيش مهام مستحقة دلوقتي' : `شغّلنا ${arNum(r.ran)} مهمة`,
   352→        description: r.due > 0 ? `كان فيه ${arNum(r.due)} مهمة مستحقة` : undefined,
   353→      })
   354→    } catch (e) {
   355→      toast({
   356→        variant: 'destructive',
   357→        title: 'مش قادرين نشغّل المهام',
   358→        description: e instanceof Error ? e.message : undefined,
   359→      })
   360→    } finally {
   361→      setTickBusy(false)
   362→    }
   363→  }
   364→
   365→  async function handleAddMemory() {
   366→    if (!memText.trim()) {
   367→      toast({ variant: 'destructive', title: 'اكتب اللي عايز وكيل يفتكره' })

   369→    }
   370→    setMemBusy(true)
   371→    try {
   372→      await api.addMemory({ businessId, userId, text: memText.trim(), kind: memKind })
   373→      await qc.invalidateQueries({ queryKey: QUERY_KEYS.memories(businessId) })
   374→      setMemText('')
   375→      toast({ title: 'حفظناها في ذاكرة وكيل' })
   376→    } catch (e) {
   377→      toast({
   378→        variant: 'destructive',
   379→        title: 'مش قادرين نحفظها',
   380→        description: e instanceof Error ? e.message : undefined,
   381→      })
   382→    } finally {
   383→      setMemBusy(false)
   384→    }
   385→  }
   386→
   387→  return (
   388→    <motion.div
   389→      initial={{ opacity: 0, y: 8 }}
   390→      animate={{ opacity: 1, y: 0 }}
   391→      transition={{ duration: 0.2, ease: 'easeOut' }}
   392→      className="mx-auto w-full max-w-5xl space-y-4 px-4 pb-8 pt-2 sm:px-6"
   393→    >
   394→      <ViewHeader title="الإعدادات" subtitle="الموديلات والمهام والذاكرة — كل حاجة تحت سيطرتك" />
   395→
   396→      {/* (أ) الموديلات والمزوّدين */}

   409→            <p className="text-xs font-medium text-muted-foreground">
   410→              الموديلات الجاهزة — شغالة على حساب وكيل من غير أي إعداد

   447→                icon={Server}
   448→                title="لسه مفيش مزوّدين خارجيين"
   449→                hint="عندك مفتاح OpenAI أو أي مزوّد متوافق؟ ضيفه من زر «إضافة مزوّد جديد» وهنجيب موديلاته بنفسنا"
   450→              />
   451→            ) : (
   452→              providers.map((p) => (
   453→                <Card
   454→                  key={p.id}
   455→                  className="card-enter gap-0 py-0 transition-[border-color,box-shadow] duration-200 hover:border-wakil-amber/40 hover:shadow-md"
   456→                >
   457→                  <CardHeader className="border-b px-4 py-3">
   458→                    <CardTitle className="flex w-full flex-wrap items-center justify-between gap-2 text-sm">
   459→                      <span className="min-w-0 break-words font-semibold">{p.name}</span>
   460→                      <span className="flex items-center gap-1">
   461→                        {deletingId === p.id && (

   464→                        <AlertDialog>
   465→                          <AlertDialogTrigger asChild>
   466→                            <Button
   467→                              variant="ghost"
   468→                              size="icon"
   469→                              className="size-10 text-destructive hover:bg-destructive/10"
   470→                              aria-label={`امسح مزوّد ${p.name}`}
   471→                            >
   472→                              <Trash2 aria-hidden />
   473→                            </Button>
   474→                          </AlertDialogTrigger>
   475→                          <AlertDialogContent dir="rtl">
   476→                            <AlertDialogHeader>
   477→                              <AlertDialogTitle>تمسح «{p.name}»؟</AlertDialogTitle>
   478→                              <AlertDialogDescription>
   479→                                الموديلات المرتبطة بيه هتتمسح معاه. لو الموديل الحالي منه، ارجع
   480→                                للموديلات الجاهزة.
   481→                              </AlertDialogDescription>
   482→                            </AlertDialogHeader>
   483→                            <AlertDialogFooter dir="rtl">
   484→                              <AlertDialogCancel className="min-h-10">إلغاء</AlertDialogCancel>
   485→                              <AlertDialogAction
   486→                                className="min-h-10"
   487→                                onClick={() => handleDeleteProvider(p)}
   488→                              >
   489→                                امسح المزوّد
   490→                              </AlertDialogAction>
   491→                            </AlertDialogFooter>
   492→                          </AlertDialogContent>
   493→                        </AlertDialog>
   494→                      </span>
   495→                    </CardTitle>
   496→                    <p dir="ltr" className="num truncate font-mono text-xs text-muted-foreground">
   497→                      {p.baseUrl}
   498→                    </p>
   499→                  </CardHeader>
   500→                  <CardContent className="p-2">
   501→                    {p.models.length === 0 ? (
   502→                      <p className="px-3 py-4 text-center text-sm text-muted-foreground">
   503→                        مفيش موديلات محفوظة للمزوّد ده

   507→                        {p.models.map((m) => {
   508→                          const selected = modelKey === `ext:${m.id}`
   509→                          const busy = busyModelKey === `${p.id}:${m.id}`
   510→                          return (
   511→                            <li key={m.id} className="flex items-center gap-1">
   512→                              <div className="min-w-0 flex-1">

   515→                                  disabled={!m.enabled}
   516→                                  onSelect={() => m.enabled && handleSelectModel(`ext:${m.id}`, m.label)}
   517→                                  ariaLabel={`اختار موديل ${m.label}`}
   518→                                >
   519→                                  <span className="flex min-w-0 flex-1 flex-wrap items-center gap-2">
   520→                                    <span
   521→                                      className={cn(
   522→                                        'truncate text-sm font-medium',
   523→                                        !m.enabled && 'text-muted-foreground',

   525→                                    >
   526→                                      {m.label}
   527→                                    </span>
   528→                                    <TierBadge tier={m.tier} />
   529→                                    {busy && (
   530→                                      <Loader2
   531→                                        className="size-3.5 animate-spin text-muted-foreground"
   532→                                        aria-hidden
   533→                                      />
   534→                                    )}
   535→                                  </span>
   536→                                </ModelRow>
   537→                              </div>
   538→                              <div className="ps-1">
   539→                                <SwitchHit

   942→/** 44×44 hit-area switch — the visual Switch is pointer-events-none,
   943→ *  this wrapper carries the semantics and the touch target. */
   944→function SwitchHit({
   945→  checked,
   946→  onToggle,
   947→  ariaLabel,
   948→  disabled,
   949→}: {
   950→  checked: boolean
   951→  onToggle: (next: boolean) => void
   952→  ariaLabel: string
   953→  disabled?: boolean
   954→}) {
   955→  return (
   956→    <span
   957→      role="switch"
   958→      aria-checked={checked}
   959→      aria-label={ariaLabel}
   960→      tabIndex={disabled ? -1 : 0}
   961→      onClick={() => !disabled && onToggle(!checked)}
   962→      onKeyDown={(e) => {
   963→        if (disabled) return
   964→        if (e.key === 'Enter' || e.key === ' ') {
   965→          e.preventDefault()
   966→          onToggle(!checked)
   967→        }
   968→      }}
   969→      className={cn(
   970→        'flex min-h-11 min-w-11 cursor-pointer items-center justify-center rounded-md outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50',
   971→        disabled && 'cursor-not-allowed opacity-50',
   972→      )}
   973→    >
   974→      <Switch checked={checked} className="pointer-events-none" tabIndex={-1} />
   975→    </span>
   976→  )
   977→}
